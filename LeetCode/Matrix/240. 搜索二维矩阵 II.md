# [240. 搜索二维矩阵 II](https://leetcode.cn/problems/search-a-2d-matrix-ii/)

- 🟡 **难度**：中等
- 🔲 **分类**：矩阵
- 🏷️ **标签**：`数组`、`二分查找`、`分治`、`矩阵`

## 题目描述

编写一个高效的算法来搜索 `m x n` 矩阵 `matrix` 中的一个目标值 `target` 。该矩阵具有以下特性：
- 每行的元素从左到右升序排列。
- 每列的元素从上到下升序排列。

**示例 1：**

![](https://assets.leetcode.cn/aliyun-lc-upload/uploads/2020/11/25/searchgrid2.jpg)

```
输入：matrix = [[1,4,7,11,15],[2,5,8,12,19],[3,6,9,16,22],[10,13,14,17,24],[18,21,23,26,30]], target = 5
输出：true
```

**示例 2：**

![](https://assets.leetcode.cn/aliyun-lc-upload/uploads/2020/11/25/searchgrid.jpg)

```
输入：matrix = [[1,4,7,11,15],[2,5,8,12,19],[3,6,9,16,22],[10,13,14,17,24],[18,21,23,26,30]], target = 20
输出：false
```

**提示：**
- `m == matrix.length`
- `n == matrix[i].length`
- `1 <= n, m <= 300`
- `-10^9 <= matrix[i][j] <= 10^9`
- 每行的所有元素从左到右升序排列
- 每列的所有元素从上到下升序排列
- `-10^9 <= target <= 10^9`

## 题解：右上角阶梯搜索（O(m + n)）

### ☕ Java

```java
import java.util.Scanner;

class Solution {
    public boolean searchMatrix(int[][] matrix, int target) {
        int m=matrix.length,n=matrix[0].length;
        // 从右上角出发：它是所在行的最大值、所在列的最小值，
        // 每比较一次可排除一整行或一整列
        int x=0,y=n-1;
        while(x<m&&y>=0){
            if(matrix[x][y]==target){
                return true;
            }else if(matrix[x][y]>target){
                // 当前值太大：所在列的下方只会更大，整列排除，往左走
                y--;
            }else{
                // 当前值太小：所在行的左边只会更小，整行排除，往下走
                x++;
            }
        }
        return false;
    }

    public static void main(String[] args) {
        // ACM 输入：第一行 m n target，接下来 m 行每行 n 个整数
        Scanner sc = new Scanner(System.in);
        int m = sc.nextInt();
        int n = sc.nextInt();
        int target = sc.nextInt();
        int[][] matrix = new int[m][n];
        for (int i = 0; i < m; i++) {
            for (int j = 0; j < n; j++) {
                matrix[i][j] = sc.nextInt();
            }
        }
        System.out.println(new Solution().searchMatrix(matrix, target));
        sc.close();
    }
}
```

### 🐍 Python

```python
import sys
from typing import List


class Solution:
    def searchMatrix(self, matrix: List[List[int]], target: int) -> bool:
        m, n = len(matrix), len(matrix[0])
        # 从右上角出发：所在行的最大值、所在列的最小值
        x, y = 0, n - 1
        while x < m and y >= 0:
            if matrix[x][y] == target:
                return True
            elif matrix[x][y] > target:
                # 整列排除，往左走
                y -= 1
            else:
                # 整行排除，往下走
                x += 1
        return False


if __name__ == "__main__":
    # ACM 输入：第一行 m n target，接下来 m 行每行 n 个整数
    data = sys.stdin.read().split()
    m, n, target = int(data[0]), int(data[1]), int(data[2])
    matrix = [[int(x) for x in data[3 + i * n: 3 + (i + 1) * n]] for i in range(m)]
    print(Solution().searchMatrix(matrix, target))
```

### ➕ C++

```cpp
#include <vector>
#include <iostream>
using namespace std;

class Solution {
public:
    bool searchMatrix(vector<vector<int>>& matrix, int target) {
        int m = matrix.size(), n = matrix[0].size();
        // 从右上角出发：所在行的最大值、所在列的最小值
        int x = 0, y = n - 1;
        while (x < m && y >= 0) {
            if (matrix[x][y] == target) {
                return true;
            } else if (matrix[x][y] > target) {
                // 整列排除，往左走
                y--;
            } else {
                // 整行排除，往下走
                x++;
            }
        }
        return false;
    }
};

int main() {
    // ACM 输入：第一行 m n target，接下来 m 行每行 n 个整数
    int m, n, target;
    cin >> m >> n >> target;
    vector<vector<int>> matrix(m, vector<int>(n));
    for (int i = 0; i < m; i++) {
        for (int j = 0; j < n; j++) {
            cin >> matrix[i][j];
        }
    }
    cout << (Solution().searchMatrix(matrix, target) ? "true" : "false") << endl;
    return 0;
}
```

### ⚙️ C

```c
#include <stdio.h>
#include <stdlib.h>
#include <stdbool.h>

bool searchMatrix(int** matrix, int matrixSize, int* matrixColSize, int target) {
    int m = matrixSize, n = matrixColSize[0];
    // 从右上角出发：所在行的最大值、所在列的最小值
    int x = 0, y = n - 1;
    while (x < m && y >= 0) {
        if (matrix[x][y] == target) {
            return true;
        } else if (matrix[x][y] > target) {
            // 整列排除，往左走
            y--;
        } else {
            // 整行排除，往下走
            x++;
        }
    }
    return false;
}

int main(void) {
    // ACM 输入：第一行 m n target，接下来 m 行每行 n 个整数
    int m, n, target;
    if (scanf("%d %d %d", &m, &n, &target) != 3) {
        return 0;
    }
    int** matrix = (int**)malloc(m * sizeof(int*));
    int* matrixColSize = (int*)malloc(m * sizeof(int));
    for (int i = 0; i < m; i++) {
        matrix[i] = (int*)malloc(n * sizeof(int));
        matrixColSize[i] = n;
        for (int j = 0; j < n; j++) {
            scanf("%d", &matrix[i][j]);
        }
    }

    printf("%s\n", searchMatrix(matrix, m, matrixColSize, target) ? "true" : "false");

    for (int i = 0; i < m; i++) {
        free(matrix[i]);
    }
    free(matrix);
    free(matrixColSize);
    return 0;
}
```
