# [76. 最小覆盖子串](https://leetcode.cn/problems/minimum-window-substring/)

- 🔴 **难度**：困难
- 🔤 **分类**：子串
- 🏷️ **标签**：`哈希表`、`字符串`、`滑动窗口`

## 题目描述

给定两个字符串 `s` 和 `t`，长度分别是 `m` 和 `n`，返回 s 中的 **最短窗口 子串**，使得该子串包含 `t` 中的每一个字符（**包括重复字符**）。如果没有这样的子串，返回空字符串 `""`。

测试用例保证答案唯一。

**示例 1：**

```
输入：s = "ADOBECODEBANC", t = "ABC"
输出："BANC"
解释：最小覆盖子串 "BANC" 包含来自字符串 t 的 'A'、'B' 和 'C'。
```

**示例 2：**

```
输入：s = "a", t = "a"
输出："a"
解释：整个字符串 s 是最小覆盖子串。
```

**示例 3:**

```
输入: s = "a", t = "aa"
输出: ""
解释: t 中两个字符 'a' 均应包含在 s 的子串中，
因此没有符合条件的子字符串，返回空字符串。
```

**提示：**
- `m == s.length`
- `n == t.length`
- `1 <= m, n <= 10^5`
- `s` 和 `t` 由英文字母组成

**进阶：** 你能设计一个在 `O(m + n)` 时间内解决此问题的算法吗？

## 题解：滑动窗口（欠债计数）

### ☕ Java

```java
import java.util.Scanner;

class Solution {
    public String minWindow(String str, String tar) {
        if (str.length() < tar.length()) {
            return "";
        } else if (str.length() == tar.length() && str.equals(tar)) {
            return str;
        }
        char[] s = str.toCharArray();
        char[] t = tar.toCharArray();
        int[] cnt = new int[128];
        for (char c : t) {
            cnt[c]--;
        }
        int start = -1;
        int debt = t.length;
        int len = Integer.MAX_VALUE;
        for (int l = 0, r = 0; r < s.length; r++) {
            if (cnt[s[r]]++ < 0) {
                debt--;
            }
            if (debt == 0) {
                while (cnt[s[l]] > 0) {
                    cnt[s[l]]--;
                    l++;
                }
                if (r - l + 1 < len) {
                    len = r - l + 1;
                    start = l;
                }
            }
        }
        return len == Integer.MAX_VALUE ? "" : str.substring(start, start + len);
    }

    public static void main(String[] args) {
        // ACM 输入：第一行 s，第二行 t（均为英文字母，无空格）
        Scanner sc = new Scanner(System.in);
        String s = sc.next();
        String t = sc.next();
        System.out.println(new Solution().minWindow(s, t));
        sc.close();
    }
}
```

### 🐍 Python

```python
import sys


class Solution:
    def minWindow(self, s: str, t: str) -> str:
        if len(s) < len(t):
            return ""
        elif len(s) == len(t) and s == t:
            return s

        # t 中字符需要的次数记为负数，欠多少一目了然
        cnt = [0] * 128
        for c in t:
            cnt[ord(c)] -= 1

        start = -1
        debt = len(t)
        length = float("inf")
        l = 0
        for r in range(len(s)):
            if cnt[ord(s[r])] < 0:
                debt -= 1
            cnt[ord(s[r])] += 1
            if debt == 0:
                # 收缩左边界，直到再缩就会缺字符
                while cnt[ord(s[l])] > 0:
                    cnt[ord(s[l])] -= 1
                    l += 1
                if r - l + 1 < length:
                    length = r - l + 1
                    start = l
        return "" if length == float("inf") else s[start:start + length]


if __name__ == "__main__":
    # ACM 输入：第一行 s，第二行 t（均为英文字母，无空格）
    data = sys.stdin.read().split()
    s, t = data[0], data[1]
    print(Solution().minWindow(s, t))
```

### ➕ C++

```cpp
#include <string>
#include <climits>
#include <iostream>
using namespace std;

class Solution {
public:
    string minWindow(string str, string tar) {
        if (str.size() < tar.size()) {
            return "";
        } else if (str.size() == tar.size() && str == tar) {
            return str;
        }
        int cnt[128] = {0};
        for (char c : tar) {
            cnt[(unsigned char)c]--;
        }
        int start = -1;
        int debt = tar.size();
        int len = INT_MAX;
        int l = 0;
        for (int r = 0; r < (int)str.size(); r++) {
            unsigned char c = (unsigned char)str[r];
            if (cnt[c] < 0) {
                debt--;
            }
            cnt[c]++;
            if (debt == 0) {
                while (cnt[(unsigned char)str[l]] > 0) {
                    cnt[(unsigned char)str[l]]--;
                    l++;
                }
                if (r - l + 1 < len) {
                    len = r - l + 1;
                    start = l;
                }
            }
        }
        return len == INT_MAX ? "" : str.substr(start, len);
    }
};

int main() {
    // ACM 输入：第一行 s，第二行 t（均为英文字母，无空格）
    string s, t;
    cin >> s >> t;
    cout << Solution().minWindow(s, t) << endl;
    return 0;
}
```

### ⚙️ C

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <limits.h>

static char* empty(void) {
    char* r = (char*)malloc(1);
    r[0] = '\0';
    return r;
}

char* minWindow(char* str, char* tar) {
    int sLen = (int)strlen(str);
    int tLen = (int)strlen(tar);
    if (sLen < tLen) {
        return empty();
    } else if (sLen == tLen && strcmp(str, tar) == 0) {
        char* r = (char*)malloc(sLen + 1);
        strcpy(r, str);
        return r;
    }

    // t 中字符需要的次数记为负数，欠多少一目了然
    int cnt[128] = {0};
    for (int i = 0; i < tLen; i++) {
        cnt[(unsigned char)tar[i]]--;
    }

    int start = -1;
    int debt = tLen;
    int len = INT_MAX;
    int l = 0;
    for (int r = 0; r < sLen; r++) {
        unsigned char c = (unsigned char)str[r];
        if (cnt[c] < 0) {
            debt--;
        }
        cnt[c]++;
        if (debt == 0) {
            // 收缩左边界，直到再缩就会缺字符
            while (cnt[(unsigned char)str[l]] > 0) {
                cnt[(unsigned char)str[l]]--;
                l++;
            }
            if (r - l + 1 < len) {
                len = r - l + 1;
                start = l;
            }
        }
    }

    if (len == INT_MAX) {
        return empty();
    }
    char* res = (char*)malloc(len + 1);
    memcpy(res, str + start, len);
    res[len] = '\0';
    return res;
}

int main(void) {
    // ACM 输入：第一行 s，第二行 t（均为英文字母，无空格）
    char s[100005], t[100005];
    if (scanf("%s", s) != 1) {
        return 0;
    }
    if (scanf("%s", t) != 1) {
        return 0;
    }
    char* res = minWindow(s, t);
    printf("%s\n", res);
    free(res);
    return 0;
}
```
