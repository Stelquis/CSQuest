# [239. 滑动窗口最大值](https://leetcode.cn/problems/sliding-window-maximum/)

- 🔴 **难度**：困难
- 🔤 **分类**：子串
- 🏷️ **标签**：`队列`、`数组`、`滑动窗口`、`单调队列`、`堆（优先队列）`、`区间最值查询`

## 题目描述

给你一个整数数组 `nums`，有一个大小为 `k` 的滑动窗口从数组的最左侧移动到数组的最右侧。你只可以看到在滑动窗口内的 `k` 个数字。滑动窗口每次只向右移动一位。

返回 *滑动窗口中的最大值* 。

**示例 1：**

```
输入：nums = [1,3,-1,-3,5,3,6,7], k = 3
输出：[3,3,5,5,6,7]
解释：
滑动窗口的位置                最大值
---------------               -----
[1  3  -1] -3  5  3  6  7       3
 1 [3  -1  -3] 5  3  6  7       3
 1  3 [-1  -3  5] 3  6  7       5
 1  3  -1 [-3  5  3] 6  7       5
 1  3  -1  -3 [5  3  6] 7       6
 1  3  -1  -3  5 [3  6  7]      7
```

**示例 2：**

```
输入：nums = [1], k = 1
输出：[1]
```

**提示：**
- `1 <= nums.length <= 10^5`
- `-10^4 <= nums[i] <= 10^4`
- `1 <= k <= nums.length`

## 题解：单调队列

### ☕ Java

```java
import java.util.ArrayDeque;
import java.util.Deque;
import java.util.Scanner;

class Solution {
    public int[] maxSlidingWindow(int[] nums, int k) {
        if (k == 1) return nums;
        int[] ans = new int[nums.length - k + 1];
        Deque<Integer> deque = new ArrayDeque<>();
        int max = Integer.MIN_VALUE;
        for (int i = 0; i < nums.length; i++) {
            while (!deque.isEmpty() && nums[deque.peekLast()] <= nums[i]) {
                deque.pollLast();
            }
            deque.offerLast(i);
            if (deque.peekFirst() < i - k + 1) {
                deque.pollFirst();
            }
            if (i >= k - 1) {
                ans[i - k + 1] = nums[deque.peekFirst()];
            }
        }
        return ans;
    }

    public static void main(String[] args) {
        // ACM 输入：第一行 n k，第二行 n 个整数
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        int[] nums = new int[n];
        for (int i = 0; i < n; i++) {
            nums[i] = sc.nextInt();
        }
        int[] res = new Solution().maxSlidingWindow(nums, k);
        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < res.length; i++) {
            if (i > 0) sb.append(' ');
            sb.append(res[i]);
        }
        System.out.println(sb);
        sc.close();
    }
}
```

### 🐍 Python

```python
import sys
from typing import List
from collections import deque


class Solution:
    def maxSlidingWindow(self, nums: List[int], k: int) -> List[int]:
        if k == 1:
            return nums
        ans = [0] * (len(nums) - k + 1)
        dq = deque()
        for i in range(len(nums)):
            # 队尾比当前元素小的都弹掉，保持队列单调不增
            while dq and nums[dq[-1]] <= nums[i]:
                dq.pop()
            dq.append(i)
            # 队首已经滑出窗口就弹掉
            if dq[0] < i - k + 1:
                dq.popleft()
            if i >= k - 1:
                ans[i - k + 1] = nums[dq[0]]
        return ans


if __name__ == "__main__":
    # ACM 输入：第一行 n k，第二行 n 个整数
    data = sys.stdin.read().split()
    n, k = int(data[0]), int(data[1])
    nums = [int(x) for x in data[2:2 + n]]
    res = Solution().maxSlidingWindow(nums, k)
    print(" ".join(map(str, res)))
```

### ➕ C++

```cpp
#include <vector>
#include <deque>
#include <iostream>
using namespace std;

class Solution {
public:
    vector<int> maxSlidingWindow(vector<int>& nums, int k) {
        if (k == 1) return nums;
        vector<int> ans(nums.size() - k + 1);
        deque<int> dq;
        for (int i = 0; i < (int)nums.size(); i++) {
            while (!dq.empty() && nums[dq.back()] <= nums[i]) {
                dq.pop_back();
            }
            dq.push_back(i);
            if (dq.front() < i - k + 1) {
                dq.pop_front();
            }
            if (i >= k - 1) {
                ans[i - k + 1] = nums[dq.front()];
            }
        }
        return ans;
    }
};

int main() {
    // ACM 输入：第一行 n k，第二行 n 个整数
    int n, k;
    cin >> n >> k;
    vector<int> nums(n);
    for (int i = 0; i < n; i++) {
        cin >> nums[i];
    }
    vector<int> res = Solution().maxSlidingWindow(nums, k);
    for (size_t i = 0; i < res.size(); i++) {
        if (i) cout << ' ';
        cout << res[i];
    }
    cout << endl;
    return 0;
}
```

### ⚙️ C

```c
#include <stdio.h>
#include <stdlib.h>

/**
 * Note: The returned array must be malloced, assume caller calls free().
 */
int* maxSlidingWindow(int* nums, int numsSize, int k, int* returnSize) {
    if (k == 1) {
        int* res = (int*)malloc(numsSize * sizeof(int));
        for (int i = 0; i < numsSize; i++) {
            res[i] = nums[i];
        }
        *returnSize = numsSize;
        return res;
    }

    int* ans = (int*)malloc((numsSize - k + 1) * sizeof(int));
    // 单调队列（存下标）：head 为队首，tail 为队尾后一位
    // 每个下标只入队一次，故数组开 numsSize 足够
    int* dq = (int*)malloc(numsSize * sizeof(int));
    int head = 0, tail = 0;

    for (int i = 0; i < numsSize; i++) {
        // 队尾比当前元素小的都弹掉，保持队列单调不增
        while (tail > head && nums[dq[tail - 1]] <= nums[i]) {
            tail--;
        }
        dq[tail++] = i;
        // 队首已经滑出窗口就弹掉
        if (dq[head] < i - k + 1) {
            head++;
        }
        if (i >= k - 1) {
            ans[i - k + 1] = nums[dq[head]];
        }
    }

    free(dq);
    *returnSize = numsSize - k + 1;
    return ans;
}

int main(void) {
    // ACM 输入：第一行 n k，第二行 n 个整数
    int n, k;
    if (scanf("%d %d", &n, &k) != 2) {
        return 0;
    }
    int* nums = (int*)malloc(n * sizeof(int));
    for (int i = 0; i < n; i++) {
        scanf("%d", &nums[i]);
    }

    int returnSize = 0;
    int* res = maxSlidingWindow(nums, n, k, &returnSize);
    for (int i = 0; i < returnSize; i++) {
        if (i) printf(" ");
        printf("%d", res[i]);
    }
    printf("\n");

    free(res);
    free(nums);
    return 0;
}
```
