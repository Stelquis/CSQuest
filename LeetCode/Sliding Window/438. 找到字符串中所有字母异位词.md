# [438. 找到字符串中所有字母异位词](https://leetcode.cn/problems/find-all-anagrams-in-a-string/)

- 🟡 **难度**：中等
- 🪟 **分类**：滑动窗口
- 🏷️ **标签**：`哈希表`、`字符串`、`滑动窗口`

## 题目描述

给定两个字符串 `s` 和 `p`，找到 `s` 中所有 `p` 的 **异位词** 的子串，返回这些子串的起始索引。不考虑答案输出的顺序。

**示例 1:**

```
输入: s = "cbaebabacd", p = "abc"
输出: [0,6]
解释:
起始索引等于 0 的子串是 "cba", 它是 "abc" 的异位词。
起始索引等于 6 的子串是 "bac", 它是 "abc" 的异位词。
```

 **示例 2:**

```
输入: s = "abab", p = "ab"
输出: [0,1,2]
解释:
起始索引等于 0 的子串是 "ab", 它是 "ab" 的异位词。
起始索引等于 1 的子串是 "ba", 它是 "ab" 的异位词。
起始索引等于 2 的子串是 "ab", 它是 "ab" 的异位词。
```

**提示:**
- `1 <= s.length, p.length <= 3 * 10^4`
- `s` 和 `p` 仅包含小写字母

## 题解一：滑动窗口 + 字符计数

### ☕ Java

```java
import java.util.ArrayList;
import java.util.Arrays;
import java.util.HashMap;
import java.util.List;
import java.util.Map;
import java.util.Scanner;

class Solution {
    public List<Integer> findAnagrams(String s, String p) {
        List<Integer> ans = new ArrayList<>();
        if (s.length() < p.length()) return ans;

        Map<Character, Integer> map = new HashMap<>();
        for (char c : p.toCharArray()) {
            map.put(c, map.getOrDefault(c, 0) + 1);
        }

        int[] str = new int[128];
        int pLen = p.length();

        // 初始化第一个窗口
        for (int i = 0; i < pLen; i++) {
            str[s.charAt(i)]++;
        }

        if (exam(map, str)) {
            ans.add(0);
        }

        // 滑动窗口
        for (int i = pLen; i < s.length(); i++) {
            str[s.charAt(i)]++;
            str[s.charAt(i - pLen)]--;
            if (exam(map, str)) {
                ans.add(i - pLen + 1);
            }
        }

        return ans;
    }

    public boolean exam(Map<Character, Integer> map, int[] str) {
        for (Map.Entry<Character, Integer> en : map.entrySet()) {
            char key = en.getKey();
            int value = en.getValue();
            if (str[key] != value) {
                return false;
            }
        }
        return true;
    }

    public static void main(String[] args) {
        // ACM 输入：第一行 s，第二行 p（均为小写字母，无空格）
        Scanner sc = new Scanner(System.in);
        String s = sc.next();
        String p = sc.next();
        List<Integer> res = new Solution().findAnagrams(s, p);
        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < res.size(); i++) {
            if (i > 0) sb.append(' ');
            sb.append(res.get(i));
        }
        System.out.println(sb);
        sc.close();
    }
}
```

### 🐍 Python

```python
import sys
from typing import List


class Solution:
    def findAnagrams(self, s: str, p: str) -> List[int]:
        ans = []
        if len(s) < len(p):
            return ans

        # p 中每个字母需要出现的次数
        need = {}
        for c in p:
            need[c] = need.get(c, 0) + 1

        window = [0] * 128
        pLen = len(p)

        # 初始化第一个窗口
        for i in range(pLen):
            window[ord(s[i])] += 1
        if self.exam(need, window):
            ans.append(0)

        # 滑动窗口
        for i in range(pLen, len(s)):
            window[ord(s[i])] += 1
            window[ord(s[i - pLen])] -= 1
            if self.exam(need, window):
                ans.append(i - pLen + 1)

        return ans

    def exam(self, need: dict, window: List[int]) -> bool:
        for key, value in need.items():
            if window[ord(key)] != value:
                return False
        return True


if __name__ == "__main__":
    # ACM 输入：第一行 s，第二行 p（均为小写字母，无空格）
    data = sys.stdin.read().split()
    s, p = data[0], data[1]
    res = Solution().findAnagrams(s, p)
    print(" ".join(map(str, res)))
```

### ➕ C++

```cpp
#include <string>
#include <vector>
#include <iostream>
using namespace std;

class Solution {
public:
    vector<int> findAnagrams(string s, string p) {
        vector<int> ans;
        if (s.size() < p.size()) return ans;

        int need[128] = {0};
        for (char c : p) {
            need[(unsigned char)c]++;
        }

        int window[128] = {0};
        int pLen = p.size();

        // 初始化第一个窗口
        for (int i = 0; i < pLen; i++) {
            window[(unsigned char)s[i]]++;
        }
        if (same(need, window)) {
            ans.push_back(0);
        }

        // 滑动窗口
        for (int i = pLen; i < (int)s.size(); i++) {
            window[(unsigned char)s[i]]++;
            window[(unsigned char)s[i - pLen]]--;
            if (same(need, window)) {
                ans.push_back(i - pLen + 1);
            }
        }
        return ans;
    }

private:
    // 窗口长度恒等于 pLen，因此全表比对与只比对 p 中字母等价
    bool same(int* a, int* b) {
        for (int i = 0; i < 128; i++) {
            if (a[i] != b[i]) return false;
        }
        return true;
    }
};

int main() {
    // ACM 输入：第一行 s，第二行 p（均为小写字母，无空格）
    string s, p;
    cin >> s >> p;
    vector<int> res = Solution().findAnagrams(s, p);
    for (size_t i = 0; i < res.size(); i++) {
        if (i) cout << ' ';
        cout << res[i];
    }
    cout << endl;
    return 0;
}
```

## 题解二：滑动窗口 + 数组比对

### ☕ Java

```java
import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;
import java.util.Scanner;

class Solution {
    public List<Integer> findAnagrams(String s, String p) {
        List<Integer> ans = new ArrayList<>();
        if (s.length() < p.length()) return ans;

        int[] need = new int[26];
        int[] window = new int[26];
        for (char c : p.toCharArray()) need[c - 'a']++;

        int pLen = p.length();
        for (int i = 0; i < s.length(); i++) {
            window[s.charAt(i) - 'a']++;
            if (i >= pLen) window[s.charAt(i - pLen) - 'a']--;
            if (i >= pLen - 1 && Arrays.equals(need, window)) {
                ans.add(i - pLen + 1);
            }
        }
        return ans;
    }

    public static void main(String[] args) {
        // ACM 输入：第一行 s，第二行 p（均为小写字母，无空格）
        Scanner sc = new Scanner(System.in);
        String s = sc.next();
        String p = sc.next();
        List<Integer> res = new Solution().findAnagrams(s, p);
        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < res.size(); i++) {
            if (i > 0) sb.append(' ');
            sb.append(res.get(i));
        }
        System.out.println(sb);
        sc.close();
    }
}
```
