# [75. 颜色分类](https://leetcode.cn/problems/sort-colors/)

- 🟡 **难度**：中等
- 🎩 **分类**：技巧
- 🏷️ **标签**：`数组`、`双指针`、`冒泡排序`、`排序`、`快速排序`

## 题目描述

给定一个包含红色、白色和蓝色、共 `n` 个元素的数组 `nums` ，**[原地](https://baike.baidu.com/item/%E5%8E%9F%E5%9C%B0%E7%AE%97%E6%B3%95)** 对它们进行排序，使得相同颜色的元素相邻，并按照红色、白色、蓝色顺序排列。

我们使用整数 0、 1 和 2 分别表示红色、白色和蓝色。

必须在不使用库内置的 sort 函数的情况下解决这个问题。

**示例 1：**

**输入：** nums = [2,0,2,1,1,0]

**输出：**[0,0,1,1,2,2]

**解释：**

该数组包含两个 0、两个 1 和两个 2。将它们原地排序后，所有 0 排在最前面，接着是所有 1，最后是所有 2。

**示例 2：**

**输入：** nums = [2,0,1]

**输出：**[0,1,2]

**解释：**

数组中有且仅有一个 0、一个 1 和一个 2，按 0、1、2 的顺序原地排列。

**提示：**
- `n == nums.length`
- `1 <= n <= 300`
- `nums[i]` 为 0、1 或 2

**进阶：**
- 你能想出一个仅使用常数空间的一趟扫描算法吗？

## 题解：三指针（荷兰国旗 / 三路划分）

### ☕ Java

```java
import java.util.Scanner;

class Solution {
    public void sortColors(int[] nums) {
        int l = 0, r = nums.length - 1;
        for (int i = 0; i <= r; ) {
            if (nums[i] == 0) {
                swap(nums, i++, l++);
            } else if (nums[i] == 2) {
                swap(nums, i, r--);
            } else {
                i++;
            }
        }
    }

    public static void swap(int[] nums, int i, int j) {
        int t = nums[i];
        nums[i] = nums[j];
        nums[j] = t;
    }

    public static void main(String[] args) {
        // ACM 输入：第一行 n，第二行 n 个整数（0/1/2）
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] nums = new int[n];
        for (int i = 0; i < n; i++) {
            nums[i] = sc.nextInt();
        }
        new Solution().sortColors(nums);
        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < n; i++) {
            if (i > 0) sb.append(' ');
            sb.append(nums[i]);
        }
        System.out.println(sb);
        sc.close();
    }
}
```

### 🐍 Python

```python
import sys
from typing import List


class Solution:
    def sortColors(self, nums: List[int]) -> None:
        l, r = 0, len(nums) - 1
        i = 0
        while i <= r:
            if nums[i] == 0:
                # 换到左区，可以放心 i 前进
                nums[i], nums[l] = nums[l], nums[i]
                i += 1
                l += 1
            elif nums[i] == 2:
                # 换到右区，i 不能前进，换过来的值还没看过
                nums[i], nums[r] = nums[r], nums[i]
                r -= 1
            else:
                i += 1


if __name__ == "__main__":
    # ACM 输入：第一行 n，第二行 n 个整数（0/1/2）
    data = sys.stdin.read().split()
    n = int(data[0])
    nums = [int(x) for x in data[1:1 + n]]
    Solution().sortColors(nums)
    print(" ".join(map(str, nums)))
```

```cpp
#include <vector>
#include <iostream>
using namespace std;

class Solution {
public:
    void sortColors(vector<int>& nums) {
        int l = 0, r = (int)nums.size() - 1;
        for (int i = 0; i <= r; ) {
            if (nums[i] == 0) {
                swap(nums, i++, l++);
            } else if (nums[i] == 2) {
                swap(nums, i, r--);
            } else {
                i++;
            }
        }
    }

    static void swap(vector<int>& nums, int i, int j) {
        int t = nums[i];
        nums[i] = nums[j];
        nums[j] = t;
    }
};

int main() {
    // ACM 输入：第一行 n，第二行 n 个整数（0/1/2）
    int n;
    cin >> n;
    vector<int> nums(n);
    for (int i = 0; i < n; i++) {
        cin >> nums[i];

    Solution().sortColors(nums);
    for (int i = 0; i < n; i++) {
        if (i) cout << ' ';
        cout << nums[i];
    }
    cout << endl;
    return 0;
}
```

### ⚙️ C

```c
#include <stdio.h>
#include <stdlib.h>

static void swap(int* nums, int i, int j) {
    int t = nums[i];
    nums[i] = nums[j];
    nums[j] = t;
}

void sortColors(int* nums, int numsSize) {
    int l = 0, r = numsSize - 1;
    for (int i = 0; i <= r; ) {
        if (nums[i] == 0) {
            swap(nums, i, l);
            i++;
            l++;
        } else if (nums[i] == 2) {
            swap(nums, i, r);
            r--;
        } else {
            i++;
        }
    }
}

int main(void) {
    // ACM 输入：第一行 n，第二行 n 个整数（0/1/2）
    int n;
    if (scanf("%d", &n) != 1) {
        return 0;
    }
    int* nums = (int*)malloc(n * sizeof(int));
    for (int i = 0; i < n; i++) {
        scanf("%d", &nums[i]);
    }
    sortColors(nums, n);
    for (int i = 0; i < n; i++) {
        if (i) printf(" ");
        printf("%d", nums[i]);
    }
    printf("\n");
    free(nums);
    return 0;
}
```
