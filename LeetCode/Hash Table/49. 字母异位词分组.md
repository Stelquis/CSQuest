# [49. 字母异位词分组](https://leetcode.cn/problems/group-anagrams/)

- 🟡 **难度**：中等
- 🔑 **分类**：哈希
- 🏷️ **标签**：`数组`、`哈希表`、`字符串`、`排序`

## 题目描述

给你一个字符串数组，请你将 字母异位词 组合在一起。可以按任意顺序返回结果列表。

**示例 1:**

**输入:** strs = ["eat", "tea", "tan", "ate", "nat", "bat"]

**输出:** [["bat"],["nat","tan"],["ate","eat","tea"]]

**解释：**
- 在 strs 中没有字符串可以通过重新排列来形成 `"bat"`。
- 字符串 `"nat"` 和 `"tan"` 是字母异位词，因为它们可以重新排列以形成彼此。
- 字符串 `"ate"` ，`"eat"` 和 `"tea"` 是字母异位词，因为它们可以重新排列以形成彼此。

**示例 2:**

**输入:** strs = [""]

**输出:** [[""]]

**示例 3:**

**输入:** strs = ["a"]

**输出:** [["a"]]

**提示：**
- `1 <= strs.length <= 10^4`
- `0 <= strs[i].length <= 100`
- `strs[i]` 仅包含小写字母

## 题解一：字符排序作 key

### ☕ Java

```java
import java.util.ArrayList;
import java.util.Arrays;
import java.util.HashMap;
import java.util.List;

import java.util.Scanner;

class Solution {
    public List<List<String>> groupAnagrams(String[] strs) {
        String[][] str = new String[strs.length][2];
        HashMap<String,ArrayList<String>> map=new HashMap<>();
        for(int i=0;i<strs.length;i++){
            String temp = sortString(strs[i]);
            if(map.containsKey(temp)){
                map.get(temp).add(strs[i]);
            }else{
                ArrayList<String> list = new ArrayList<>();
                list.add(strs[i]);
                map.put(temp, list);
            }
        }
        List<List<String>> ans =new ArrayList<>();
        for(ArrayList<String> l : map.values()){
            ans.add(l);
        }
        return ans;
    }

    public static String sortString(String s){
        if(s==null){
            return null;
        }
        char[] str=s.toCharArray();
        Arrays.sort(str);
        return new String(str);
    }

    public static void main(String[] args) {
        // ACM 输入：第一行 n，接下来 n 行每行一个字符串（可为空行）
        Scanner sc = new Scanner(System.in);
        int n = Integer.parseInt(sc.nextLine().trim());
        String[] strs = new String[n];
        for (int i = 0; i < n; i++) {
            strs[i] = sc.nextLine();
        }
        List<List<String>> res = new Solution().groupAnagrams(strs);
        for (List<String> group : res) {
            System.out.println(String.join(" ", group));
        }
        sc.close();
    }
}
```

```python
import sys
from typing import List


class Solution:
    def groupAnagrams(self, strs: List[str]) -> List[List[str]]:
        mp = {}
        for i in range(len(strs)):
            temp = self.sortString(strs[i])
            if temp in mp:
                mp[temp].append(strs[i])
            else:
                mp[temp] = [strs[i]]

        ans = []
        for lst in mp.values():
            ans.append(lst)
        return ans

    def sortString(self, s: str) -> str:
        if s is None:
            return None
        return "".join(sorted(s))
```

```cpp
#include <algorithm>
#include <string>
#include <unordered_map>
#include <vector>
#include <iostream>

class Solution {
public:
    vector<vector<string>> groupAnagrams(vector<string>& strs) {
        unordered_map<string, vector<string>> mp;
        for (int i = 0; i < strs.size(); i++) {
            string temp = sortString(strs[i]);
            if (mp.count(temp)) {
                mp[temp].push_back(strs[i]);
            } else {
                mp[temp] = {strs[i]};
            }
        }

        vector<vector<string>> ans;
        for (auto& kv : mp) {
            ans.push_back(kv.second);
        }
        return ans;
    }

    static string sortString(string s) {
        sort(s.begin(), s.end());
        return s;
    }
};
```

### ⚙️ C

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

/**
 * Return an array of arrays of size *returnSize.
 * The sizes of the arrays are returned as *returnColumnSizes array.
 * Note: Both returned array and *columnSizes array must be malloced, assume caller calls free().
 */
static int cmpChar(const void* a, const void* b) {
    return *(const char*)a - *(const char*)b;
}

char*** groupAnagrams(char** strs, int strsLen, int* returnSize, int** returnColumnSizes) {
    int cap = 1;
    while (cap < strsLen * 2) {
        cap <<= 1;
    }
    char** keys = (char**)calloc(cap, sizeof(char*));   // NULL 表示空槽
    int* bucket = (int*)malloc(cap * sizeof(int));      // 槽 -> 分组下标

    char*** groups = (char***)malloc((strsLen ? strsLen : 1) * sizeof(char**));
    int* gSize = (int*)malloc((strsLen ? strsLen : 1) * sizeof(int));
    int gCount = 0;

    for (int i = 0; i < strsLen; i++) {
        // 排序得到 key
        char buf[101];
        strcpy(buf, strs[i]);
        qsort(buf, strlen(buf), sizeof(char), cmpChar);

        unsigned int h = 0;
        for (char* p = buf; *p; p++) {
            h = h * 131u + (unsigned char)*p;
        }
        h &= (cap - 1);

        int slot = -1;
        while (keys[h]) {
            if (strcmp(keys[h], buf) == 0) {
                slot = h;
                break;
            }
            h = (h + 1) & (cap - 1);
        }

        if (slot >= 0) {
            int g = bucket[slot];
            gSize[g]++;
            groups[g] = (char**)realloc(groups[g], gSize[g] * sizeof(char*));
            groups[g][gSize[g] - 1] = strs[i];
        } else {
            keys[h] = (char*)malloc(strlen(buf) + 1);
            strcpy(keys[h], buf);
            bucket[h] = gCount;
            groups[gCount] = (char**)malloc(sizeof(char*));
            groups[gCount][0] = strs[i];
            gSize[gCount] = 1;
            gCount++;
        }
    }

    char*** ans = (char***)malloc(gCount * sizeof(char**));
    int* colSizes = (int*)malloc(gCount * sizeof(int));
    for (int i = 0; i < gCount; i++) {
        ans[i] = groups[i];
        colSizes[i] = gSize[i];
    }

    *returnSize = gCount;
    *returnColumnSizes = colSizes;
    free(keys);
    free(bucket);
    free(gSize);
    free(groups);
    return ans;
}

int main(void) {
    // ACM 输入：第一行 n，接下来 n 行每行一个字符串（可为空行）
    int n, i, ch;
    if (scanf("%d", &n) != 1) {
        return 0;
    }
    while ((ch = getchar()) != '\n' && ch != EOF) {
        // 吃掉第一行剩余的换行
    }
    char** strs = (char**)malloc(n * sizeof(char*));
    for (i = 0; i < n; i++) {
        strs[i] = (char*)malloc(101 * sizeof(char));
        if (fgets(strs[i], 101, stdin) == NULL) {
            strs[i][0] = '\0';
            continue;
        }
        strs[i][strcspn(strs[i], "\r\n")] = '\0';   // 去掉行尾换行
    }

    int returnSize = 0;
    int* returnColumnSizes = NULL;
    char*** res = groupAnagrams(strs, n, &returnSize, &returnColumnSizes);

    for (i = 0; i < returnSize; i++) {
        int j;
        for (j = 0; j < returnColumnSizes[i]; j++) {
            if (j) {
                printf(" ");
            }
            printf("%s", res[i][j]);
        }
        printf("\n");
    }
    return 0;
}
```

## 题解二：计数数组作 key

### ☕ Java

```java
import java.util.ArrayList;
import java.util.HashMap;
import java.util.List;
import java.util.Map;

import java.util.Scanner;

class Solution {
    public List<List<String>> groupAnagrams(String[] strs) {
        Map<String, List<String>> map = new HashMap<>();

        for (int i = 0; i < strs.length; i++) {
            String s = strs[i];

            // 1. 统计每个字母出现次数
            int[] count = new int[26];
            for (int j = 0; j < s.length(); j++) {
                count[s.charAt(j) - 'a']++;
            }

            // 2. 把计数数组拼成唯一 key
            StringBuilder sb = new StringBuilder();
            for (int j = 0; j < 26; j++) {
                sb.append('#');
                sb.append(count[j]);
            }
            String key = sb.toString();

            // 3. 普通 if / else，不用 computeIfAbsent
            if (map.containsKey(key)) {
                map.get(key).add(s);
            } else {
                List<String> list = new ArrayList<>();
                list.add(s);
                map.put(key, list);
            }
        }

        // 4. 收集结果
        List<List<String>> ans = new ArrayList<>();
        for (List<String> list : map.values()) {
            ans.add(list);
        }
        return ans;
    }

    public static void main(String[] args) {
        // ACM 输入：第一行 n，接下来 n 行每行一个字符串（可为空行）
        Scanner sc = new Scanner(System.in);
        int n = Integer.parseInt(sc.nextLine().trim());
        String[] strs = new String[n];
        for (int i = 0; i < n; i++) {
            strs[i] = sc.nextLine();
        }
        List<List<String>> res = new Solution().groupAnagrams(strs);
        for (List<String> group : res) {
            System.out.println(String.join(" ", group));
        }
        sc.close();
    }
}
```

```cpp
class Solution {
public:
    vector<vector<string>> groupAnagrams(vector<string>& strs) {
        unordered_map<string, vector<string>> mp;

        for (int i = 0; i < strs.size(); i++) {
            const string& s = strs[i];

            // 1. 统计每个字母出现次数
            int count[26] = {0};
            for (int j = 0; j < (int)s.size(); j++) {
                count[s[j] - 'a']++;
            }

            // 2. 把计数数组拼成唯一 key
            string key;
            for (int j = 0; j < 26; j++) {
                key += '#';
                key += to_string(count[j]);
            }

            // 3. 普通 if / else，不用 operator[] 隐式插入
            auto it = mp.find(key);
            if (it != mp.end()) {
                it->second.push_back(s);
            } else {
                mp[key] = {s};
            }
        }

        // 4. 收集结果
        vector<vector<string>> ans;
        for (auto& kv : mp) {
            ans.push_back(kv.second);
        }
        return ans;
```
