## 题目描述

给你一个二叉树的根节点 `root` ， 检查它是否轴对称。

```

```

**示例 2：**

![](https://pic.leetcode.cn/1698027008-nPFLbM-image.png)

- 树中节点数目在范围 `[1, 1000]` 内

## 题解一：深度优先搜索（镜像递归）

树对称 ⟺ 左子树与右子树互为镜像。递归函数收**两个**参数同步下潜：每次调用携带一对节点，比较当前对的值，然后**交叉**递归——外侧对外侧（`p.left ↔ q.right`）、内侧对内侧（`p.right ↔ q.left`）。

- 时间复杂度：`O(n)`，每个节点最多被访问两次（作为 left 参数、作为 right 参数各一次）
- 空间复杂度：`O(h)`，递归栈深度为树高，最坏（链状树）为 `O(n)`

### ☕ Java

```java
import java.util.LinkedList;
import java.util.Queue;
import java.util.Scanner;

public class Solution {
    public boolean isSymmetric(TreeNode root) {
        if (root == null) {
            return true;
        }
        return ifSymmetric(root.left, root.right);
    }

    // 判断两棵子树是否互为镜像
    public static boolean ifSymmetric(TreeNode left, TreeNode right) {
        if (left == null && right == null) {
            return true;   // 两侧同时到头
        }
        if (left == null || right == null) {
            return false;  // 一侧缺一块
        }
        if (left.val != right.val) {
            return false;  // 当前对的值不同
        }
        // 交叉下潜：外侧对外侧，内侧对内侧
        return ifSymmetric(left.left, right.right) && ifSymmetric(left.right, right.left);
    }

    public static void main(String[] args) {
        // ACM 输入：一行层序序列，null 表示空节点
        Scanner sc = new Scanner(System.in);
        TreeNode root = buildTree(sc);
        System.out.println(new Solution().isSymmetric(root));
        sc.close();
    }

    // 按层序遍历读入建树，null 表示空节点
    public static TreeNode buildTree(Scanner sc) {
        if (!sc.hasNext()) {
            return null;
        }
        String token = sc.next();
        if ("null".equals(token)) {
            return null;
        }
        TreeNode root = new TreeNode(Integer.parseInt(token));
        Queue<TreeNode> queue = new LinkedList<>();
        queue.offer(root);
        while (!queue.isEmpty()) {
            TreeNode cur = queue.poll();
            if (!sc.hasNext()) {
                break;
            }
            String left = sc.next();
            if (!"null".equals(left)) {
                cur.left = new TreeNode(Integer.parseInt(left));
                queue.offer(cur.left);
            }
            if (!sc.hasNext()) {
                break;
            }
            String right = sc.next();
            if (!"null".equals(right)) {
                cur.right = new TreeNode(Integer.parseInt(right));
                queue.offer(cur.right);
            }
        }
        return root;
    }
}

class TreeNode {
    int val;
    TreeNode left;
    TreeNode right;

    TreeNode() {}

    TreeNode(int val) { this.val = val; }

    TreeNode(int val, TreeNode left, TreeNode right) {
        this.val = val;
        this.left = left;
        this.right = right;
    }
}
```

```python
import sys
from collections import deque
from typing import Optional


class TreeNode:
    def __init__(self, val=0, left=None, right=None):
        self.val = val
        self.left = left
        self.right = right


class Solution:
    def isSymmetric(self, root: Optional[TreeNode]) -> bool:
        if root is None:
            return True
        return self.if_symmetric(root.left, root.right)

    def if_symmetric(self, left: Optional[TreeNode], right: Optional[TreeNode]) -> bool:
        if left is None and right is None:
            return True    # 两侧同时到头
        if left is None or right is None:
            return False   # 一侧缺一块
        if left.val != right.val:
            return False   # 当前对的值不同
        # 交叉下潜：外侧对外侧，内侧对内侧
        return self.if_symmetric(left.left, right.right) and self.if_symmetric(left.right, right.left)


def build_tree(tokens, pos):
    """按层序遍历建树，'null' 表示空节点"""
    if pos[0] >= len(tokens) or tokens[pos[0]] == "null":
        pos[0] += 1
        return None
    root = TreeNode(int(tokens[pos[0]]))
    pos[0] += 1
    queue = deque([root])
    while queue:
        cur = queue.popleft()
        for i in range(2):
            if pos[0] >= len(tokens):
                break
            child = None
            if tokens[pos[0]] != "null":
                child = TreeNode(int(tokens[pos[0]]))
                queue.append(child)
            if i == 0:
                cur.left = child
            else:
                cur.right = child
            pos[0] += 1
    return root


if __name__ == "__main__":
    # ACM 输入：一行层序序列，null 表示空节点
    tokens = sys.stdin.read().split()
    root = build_tree(tokens, [0])
    print(Solution().isSymmetric(root))
```

## 题解二：广度优先搜索（成对入队）

迭代版镜像比较：队列中**成对**存放待比较的节点（初始为根的左右孩子）。每取出 一对，校验值相等后，把它们的四个孩子按“外侧对外侧、内侧对内侧”交叉成两队入队。任何时刻发现不配对即返回 `false`，队列清空则为对称。

- 时间复杂度：`O(n)`，每个节点至多入队两次
- 空间复杂度：`O(w)`，`w` 为树的最大宽度，最坏约为 `O(n/2)`

### ☕ Java

```java
import java.util.LinkedList;
import java.util.Queue;
import java.util.Scanner;

public class Solution {
    public boolean isSymmetric(TreeNode root) {
        if (root == null) {
            return true;
        }
        Queue<TreeNode> queue = new LinkedList<>();
        queue.offer(root.left);
        queue.offer(root.right);
        while (!queue.isEmpty()) {
            TreeNode p = queue.poll();
            TreeNode q = queue.poll();
            if (p == null && q == null) {
                continue;          // 这一对同时到头，继续下一对
            }
            if (p == null || q == null || p.val != q.val) {
                return false;      // 不配对，一票否决
            }
            // 交叉成对入队：外侧对外侧，内侧对内侧
            queue.offer(p.left);
            queue.offer(q.right);
            queue.offer(p.right);
            queue.offer(q.left);
        }
        return true;
    }

    public static void main(String[] args) {
        // ACM 输入：一行层序序列，null 表示空节点
        Scanner sc = new Scanner(System.in);
        TreeNode root = buildTree(sc);
        System.out.println(new Solution().isSymmetric(root));
        sc.close();
    }

    // 按层序遍历读入建树，null 表示空节点
    public static TreeNode buildTree(Scanner sc) {
        if (!sc.hasNext()) {
            return null;
        }
        String token = sc.next();
        if ("null".equals(token)) {
            return null;
        }
        TreeNode root = new TreeNode(Integer.parseInt(token));
        Queue<TreeNode> queue = new LinkedList<>();
        queue.offer(root);
        while (!queue.isEmpty()) {
            TreeNode cur = queue.poll();
            if (!sc.hasNext()) {
                break;
            }
            String left = sc.next();
            if (!"null".equals(left)) {
                cur.left = new TreeNode(Integer.parseInt(left));
                queue.offer(cur.left);
            }
            if (!sc.hasNext()) {
                break;
            }
            String right = sc.next();
            if (!"null".equals(right)) {
                cur.right = new TreeNode(Integer.parseInt(right));
                queue.offer(cur.right);
            }
        }
        return root;
    }
}

class TreeNode {
    int val;
    TreeNode left;
    TreeNode right;

    TreeNode(int val, TreeNode left, TreeNode right) {
        this.val = val;
```

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <stdbool.h>

typedef struct TreeNode {
    int val;
    struct TreeNode* left;
    struct TreeNode* right;
} TreeNode;

// 手写队列：成对存节点指针
typedef struct {
    TreeNode** data;
    int head;
    int tail;
} Queue;

void queueInit(Queue* q, int cap) {
    q->data = (TreeNode**)malloc(cap * sizeof(TreeNode*));
    q->head = q->tail = 0;
}
void queuePush(Queue* q, TreeNode* node) { q->data[q->tail++] = node; }
TreeNode* queuePop(Queue* q) { return q->data[q->head++]; }
int queueEmpty(Queue* q) { return q->head == q->tail; }

bool isSymmetric(struct TreeNode* root) {
    if (root == NULL) {
        return true;
    }
    Queue queue;
    queueInit(&queue, 4096);
    queuePush(&queue, root->left);
    queuePush(&queue, root->right);
    while (!queueEmpty(&queue)) {
        TreeNode* p = queuePop(&queue);
        TreeNode* r = queuePop(&queue);
        if (p == NULL && r == NULL) {
            continue;          // 这一对同时到头，继续下一对
        }
        if (p == NULL || r == NULL || p->val != r->val) {
            free(queue.data);
            return false;      // 不配对，一票否决
        }
        // 交叉成对入队：外侧对外侧，内侧对内侧
        queuePush(&queue, p->left);
        queuePush(&queue, r->right);
        queuePush(&queue, p->right);
        queuePush(&queue, r->left);
    }
    free(queue.data);
    return true;
}
```
