# [94. 二叉树的中序遍历](https://leetcode.cn/problems/binary-tree-inorder-traversal/)

- 🟢 **难度**：简单
- 🌳 **分类**：二叉树
- 🏷️ **标签**：`栈`、`树`、`深度优先搜索`、`二叉树`

## 题目描述

给定一个二叉树的根节点 `root` ，返回 *它的 **中序** 遍历* 。

**示例 1：**

![](https://assets.leetcode.com/uploads/2020/09/15/inorder_1.jpg)

```
输入：root = [1,null,2,3]
输出：[1,3,2]
```

**示例 2：**

```
输入：root = []
输出：[]
```

**示例 3：**

```
输入：root = [1]
输出：[1]
```

**提示：**

- 树中节点数目在范围 `[0, 100]` 内
- `-100 <= Node.val <= 100`

**进阶:** 递归算法很简单，你可以通过迭代算法完成吗？

## 题解一：递归（深度优先搜索）

中序遍历的顺序是「左子树 → 根节点 → 右子树」，对每个节点递归执行同一规则，空节点即为递归出口。用一个共享列表按序收集节点值。

- 时间复杂度：`O(n)`，每个节点恰好访问一次
- 空间复杂度：`O(h)`，递归栈深度为树高，最坏（链状树）为 `O(n)`

### ☕ Java

```java
import java.util.ArrayList;
import java.util.LinkedList;
import java.util.List;
import java.util.Queue;
import java.util.Scanner;

public class Solution {
    public List<Integer> inorderTraversal(TreeNode root) {
        List<Integer> ans = new ArrayList<>();
        inorder(root, ans);
        return ans;
    }

    public static void inorder(TreeNode node, List<Integer> ans) {
        if (node == null) {
            return;
        }
        inorder(node.left, ans);
        ans.add(node.val);
        inorder(node.right, ans);
    }

    public static void main(String[] args) {
        // ACM 输入：一行层序序列，null 表示空节点
        Scanner sc = new Scanner(System.in);
        TreeNode root = buildTree(sc);
        System.out.println(new Solution().inorderTraversal(root));
        sc.close();
    }

    // 按层序遍历读入建树，null 表示空节点
    public static TreeNode buildTree(Scanner sc) {
        if (!sc.hasNext()) {
            return null;
        }
        String token = sc.next();
        if ("null".equals(token)) {
            return null;
        }
        TreeNode root = new TreeNode(Integer.parseInt(token));
        Queue<TreeNode> queue = new LinkedList<>();
        queue.offer(root);
        while (!queue.isEmpty()) {
            TreeNode cur = queue.poll();
            if (!sc.hasNext()) {
                break;
            }
            String left = sc.next();
            if (!"null".equals(left)) {
                cur.left = new TreeNode(Integer.parseInt(left));
                queue.offer(cur.left);
            }
            if (!sc.hasNext()) {
                break;
            }
            String right = sc.next();
            if (!"null".equals(right)) {
                cur.right = new TreeNode(Integer.parseInt(right));
                queue.offer(cur.right);
            }
        }
        return root;
    }
}

class TreeNode {
    int val;
    TreeNode left;
    TreeNode right;

    TreeNode() {}

    TreeNode(int val) { this.val = val; }

    TreeNode(int val, TreeNode left, TreeNode right) {
        this.val = val;
        this.left = left;
        this.right = right;
    }
}
```

```python
import sys
from typing import List, Optional


class TreeNode:
    def __init__(self, val=0, left=None, right=None):
        self.val = val
        self.left = left
        self.right = right


class Solution:
    def inorderTraversal(self, root: Optional[TreeNode]) -> List[int]:
        ans = []
        self.inorder(root, ans)
        return ans

    def inorder(self, node: Optional[TreeNode], ans: List[int]) -> None:
        if node is None:
            return
        self.inorder(node.left, ans)
        ans.append(node.val)
        self.inorder(node.right, ans)


def build_tree(tokens, pos):
    """按层序遍历建树，'null' 表示空节点"""
    if pos[0] >= len(tokens) or tokens[pos[0]] == "null":
        pos[0] += 1
        return None
    root = TreeNode(int(tokens[pos[0]]))
    pos[0] += 1
    queue = [root]
    while queue:
        cur = queue.pop(0)
        if pos[0] >= len(tokens):
            break
        if tokens[pos[0]] != "null":
            cur.left = TreeNode(int(tokens[pos[0]]))
            queue.append(cur.left)
        pos[0] += 1
        if pos[0] >= len(tokens):
            break
        if tokens[pos[0]] != "null":
            cur.right = TreeNode(int(tokens[pos[0]]))
            queue.append(cur.right)
        pos[0] += 1
    return root


if __name__ == "__main__":
    # ACM 输入：一行层序序列，null 表示空节点
```

```c
void inorder(struct TreeNode* node, int* ans, int* size) {
    if (node == NULL) {
        return;
    }
    inorder(node->left, ans, size);
    ans[(*size)++] = node->val;
    inorder(node->right, ans, size);
}

 * Note: The returned array must be malloced, assume caller calls free().
 */
int* inorderTraversal(struct TreeNode* root, int* returnSize) {

    inorder(root, ans, returnSize);
    return ans;

TreeNode* buildTree(char tokens[][16], int n) {
    if (n == 0 || strcmp(tokens[0], "null") == 0) {
        return NULL;
    }
    TreeNode** queue = (TreeNode**)malloc(n * sizeof(TreeNode*));
    int head = 0, tail = 0, i = 1;
    TreeNode* root = (TreeNode*)malloc(sizeof(TreeNode));
    root->val = atoi(tokens[0]);
    root->left = root->right = NULL;
    queue[tail++] = root;
    while (head < tail && i < n) {
        TreeNode* cur = queue[head++];
        if (strcmp(tokens[i], "null") != 0) {
            cur->left = (TreeNode*)malloc(sizeof(TreeNode));
            cur->left->val = atoi(tokens[i]);
            cur->left->left = cur->left->right = NULL;
            queue[tail++] = cur->left;
        }
        i++;
        if (i >= n) {
            break;
        }
        if (strcmp(tokens[i], "null") != 0) {
            cur->right = (TreeNode*)malloc(sizeof(TreeNode));
            cur->right->val = atoi(tokens[i]);
            cur->right->left = cur->right->right = NULL;
            queue[tail++] = cur->right;
        }
        i++;
    }
    free(queue);
    return root;
}

int main(void) {
```

## 题解二：迭代（显式栈）

递归的本质是隐式地使用调用栈，可以改用一个显式栈来模拟：沿左链一路入栈；栈顶弹出并记录（此时其左子树已处理完）；随后转向右子树重复该过程。

- 时间复杂度：`O(n)`，每个节点恰好入栈、出栈各一次
- 空间复杂度：`O(h)`，栈中最多保存一条左链，最坏为 `O(n)`

### ☕ Java

```java
import java.util.ArrayDeque;
import java.util.ArrayList;
import java.util.Deque;
import java.util.LinkedList;
import java.util.List;
import java.util.Queue;
import java.util.Scanner;

public class Solution {
    public List<Integer> inorderTraversal(TreeNode root) {
        List<Integer> ans = new ArrayList<>();
        Deque<TreeNode> stack = new ArrayDeque<>();
        TreeNode cur = root;
        while (cur != null || !stack.isEmpty()) {
            // 沿左链一路入栈
            while (cur != null) {
                stack.push(cur);
                cur = cur.left;
            }
            // 左子树处理完毕，访问栈顶
            cur = stack.pop();
            ans.add(cur.val);
            // 转向右子树
            cur = cur.right;
        }
        return ans;
    }

    public static void main(String[] args) {
        // ACM 输入：一行层序序列，null 表示空节点
        Scanner sc = new Scanner(System.in);
        TreeNode root = buildTree(sc);
        System.out.println(new Solution().inorderTraversal(root));
        sc.close();
    }

    // 按层序遍历读入建树，null 表示空节点
    public static TreeNode buildTree(Scanner sc) {
        if (!sc.hasNext()) {
            return null;
        }
        String token = sc.next();
        if ("null".equals(token)) {
            return null;
        }
        TreeNode root = new TreeNode(Integer.parseInt(token));
        Queue<TreeNode> queue = new LinkedList<>();
        queue.offer(root);
        while (!queue.isEmpty()) {
            TreeNode cur = queue.poll();
            if (!sc.hasNext()) {
                break;
            }
            String left = sc.next();
            if (!"null".equals(left)) {
                cur.left = new TreeNode(Integer.parseInt(left));
                queue.offer(cur.left);
            }
            if (!sc.hasNext()) {
                break;
            }
            String right = sc.next();
            if (!"null".equals(right)) {
                cur.right = new TreeNode(Integer.parseInt(right));
                queue.offer(cur.right);
            }
        }
        return root;
    }
}

class TreeNode {
    int val;
    TreeNode left;
    TreeNode right;

    TreeNode() {}

    TreeNode(int val) { this.val = val; }

    TreeNode(int val, TreeNode left, TreeNode right) {
        this.val = val;
        this.left = left;
        this.right = right;
    }
}
```

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct TreeNode {
    int val;
    struct TreeNode* left;
    struct TreeNode* right;
} TreeNode;

// 手写栈：存节点指针
typedef struct {
    TreeNode** data;
    int top;
} Stack;

void stackInit(Stack* s, int cap) {
    s->data = (TreeNode**)malloc(cap * sizeof(TreeNode*));
    s->top = 0;
}
void stackPush(Stack* s, TreeNode* node) { s->data[s->top++] = node; }
TreeNode* stackPop(Stack* s) { return s->data[--s->top]; }
int stackEmpty(Stack* s) { return s->top == 0; }

/**
 * Note: The returned array must be malloced, assume caller calls free().
 */
int* inorderTraversal(struct TreeNode* root, int* returnSize) {
    int* ans = (int*)malloc(100 * sizeof(int));
    *returnSize = 0;
    Stack stack;
    stackInit(&stack, 100);
    TreeNode* cur = root;
    while (cur != NULL || !stackEmpty(&stack)) {
        // 沿左链一路入栈
        while (cur != NULL) {
            stackPush(&stack, cur);
            cur = cur->left;
        }
        // 左子树处理完毕，访问栈顶
        cur = stackPop(&stack);
        ans[(*returnSize)++] = cur->val;
        // 转向右子树
        cur = cur->right;
    }
    free(stack.data);
    return ans;
}
```
