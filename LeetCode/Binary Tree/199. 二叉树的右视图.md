## 题解一：广度优先搜索（层序遍历取每层末位）

右视图 = 每一层"最靠右"的节点连成的序列。层序遍历时每层从左到右出队，**本层最后一个出队的节点**即答案。复用 102 的 `size` 固化技巧，内层循环里判断 `i == size - 1` 时记录。

注意：右子树缺失时，深处的左孩子会"透过空缺"露出来（示例 2 的 5）——层序法天然覆盖，因为那层最后一个就是它。

- 时间复杂度：`O(n)`
- 空间复杂度：`O(w)`，队列最坏存一整层

### ☕ Java

```java
import java.util.ArrayDeque;
import java.util.ArrayList;
import java.util.LinkedList;
import java.util.List;
import java.util.Queue;
import java.util.Scanner;

public class Solution {
    public List<Integer> rightSideView(TreeNode root) {
        if (root == null) {
            return new ArrayList<>();
        }
        List<Integer> ans = new ArrayList<>();
        Queue<TreeNode> queue = new ArrayDeque<>();
        queue.offer(root);
        while (!queue.isEmpty()) {
            int size = queue.size();              // 固化本层人数
            for (int i = 0; i < size; i++) {
                TreeNode node = queue.poll();
                if (i == size - 1) {              // 本层最后一个出队的
                    ans.add(node.val);
                }
                if (node.left != null) {
                    queue.offer(node.left);       // 先左后右，"最后一个"才有意义
                }
                if (node.right != null) {
                    queue.offer(node.right);
                }
            }
        }
        return ans;
    }

    public static void main(String[] args) {
        // ACM 输入：一行层序序列，null 表示空节点
        Scanner sc = new Scanner(System.in);
        TreeNode root = buildTree(sc);
        System.out.println(new Solution().rightSideView(root));
        sc.close();
    }

    // 按层序遍历读入建树，null 表示空节点
    public static TreeNode buildTree(Scanner sc) {
        if (!sc.hasNext()) {
            return null;
        }
        String token = sc.next();
        if ("null".equals(token)) {
            return null;
        }
        TreeNode root = new TreeNode(Integer.parseInt(token));
        Queue<TreeNode> queue = new LinkedList<>();
        queue.offer(root);
        while (!queue.isEmpty()) {
            TreeNode cur = queue.poll();
            if (!sc.hasNext()) {
                break;
            }
            String left = sc.next();
            if (!"null".equals(left)) {
                cur.left = new TreeNode(Integer.parseInt(left));
                queue.offer(cur.left);
            }
            if (!sc.hasNext()) {
                break;
            }
            String right = sc.next();
            if (!"null".equals(right)) {
                cur.right = new TreeNode(Integer.parseInt(right));
                queue.offer(cur.right);
            }
        }
        return root;
    }
}

class TreeNode {
    int val;
    TreeNode left;
    TreeNode right;

    TreeNode() {}

    TreeNode(int val) { this.val = val; }

    TreeNode(int val, TreeNode left, TreeNode right) {
        this.val = val;
        this.left = left;
        this.right = right;
    }
}
```

```python
        self.val = val
        self.left = left
        self.right = right

    def rightSideView(self, root: Optional[TreeNode]) -> List[int]:
        if root is None:
            return []
        ans = []
        queue = deque([root])
        while queue:
            size = len(queue)
            for i in range(size):
                node = queue.popleft()
                if i == size - 1:       # 本层最后一个出队的

                if node.left:
                    queue.append(node.left)
                if node.right:
                    queue.append(node.right)
        return ans


def build_tree(tokens, pos):
    """按层序遍历建树，'null' 表示空节点"""
    if pos[0] >= len(tokens) or tokens[pos[0]] == "null":
        pos[0] += 1
        return None
    root = TreeNode(int(tokens[pos[0]]))
    pos[0] += 1
    queue = deque([root])
    while queue:
        cur = queue.popleft()
        for i in range(2):
            if pos[0] >= len(tokens):
                break
            child = None
            if tokens[pos[0]] != "null":

            if i == 0:
                cur.left = child
            else:
                cur.right = child
            pos[0] += 1
```

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct TreeNode {
    int val;
    struct TreeNode* left;
    struct TreeNode* right;
} TreeNode;

// 手写队列
typedef struct {
    TreeNode** data;
    int head;
    int tail;
} Queue;

void queueInit(Queue* q, int cap) {
    q->data = (TreeNode**)malloc(cap * sizeof(TreeNode*));
    q->head = q->tail = 0;
}
void queuePush(Queue* q, TreeNode* node) { q->data[q->tail++] = node; }
TreeNode* queuePop(Queue* q) { return q->data[q->head++]; }
int queueEmpty(Queue* q) { return q->head == q->tail; }

int main(void) {
    // ACM 输入：一行层序序列，null 表示空节点
    char tokens[400][16];
    int n = 0;
    while (n < 400 && scanf("%15s", tokens[n]) == 1) {
        n++;
    }
    if (n == 0 || strcmp(tokens[0], "null") == 0) {
        printf("[]\n");
        return 0;
    }

    // 建树
    Queue queue;
    queueInit(&queue, 400);
    TreeNode* root = (TreeNode*)malloc(sizeof(TreeNode));
    root->val = atoi(tokens[0]);
    root->left = root->right = NULL;
    queuePush(&queue, root);
    int i = 1;
    while (!queueEmpty(&queue) && i < n) {
        TreeNode* cur = queuePop(&queue);
        if (strcmp(tokens[i], "null") != 0) {
            cur->left = (TreeNode*)malloc(sizeof(TreeNode));
            cur->left->val = atoi(tokens[i]);
            cur->left->left = cur->left->right = NULL;
            queuePush(&queue, cur->left);
        }
        i++;
        if (i >= n) {
            break;
        }
        if (strcmp(tokens[i], "null") != 0) {
            cur->right = (TreeNode*)malloc(sizeof(TreeNode));
            cur->right->val = atoi(tokens[i]);
            cur->right->left = cur->right->right = NULL;
            queuePush(&queue, cur->right);
        }
        i++;
    }

    // 层序遍历，每层最后一个即右视图
    queue.head = queue.tail = 0;    // 重置队列，从根重新开始
    queuePush(&queue, root);
    printf("[");
    int first = 1;
    while (!queueEmpty(&queue)) {
        int size = queue.tail - queue.head;
        for (int j = 0; j < size; j++) {
            TreeNode* node = queuePop(&queue);
            if (j == size - 1) {
                printf(first ? "%d" : ", %d", node->val);
                first = 0;
            }
            if (node->left != NULL) {
                queuePush(&queue, node->left);
            }
            if (node->right != NULL) {
                queuePush(&queue, node->right);
            }
        }
    }
    printf("]\n");
    return 0;
}
```

## 题解二：深度优先搜索（根 → 右 → 左，首见即收）

另一视角：前序遍历改成"根 → 右子树 → 左子树"，则**每个深度第一个被访问的节点**恰好是该层最右边的。判断"这个深度还没人登记"（`depth == ans.size()`，与 102 题解二同款技巧）即可收下。

- 时间复杂度：`O(n)`
- 空间复杂度：`O(h)`，递归栈

### ☕ Java

```java
import java.util.ArrayList;
import java.util.LinkedList;
import java.util.List;
import java.util.Queue;
import java.util.Scanner;

public class Solution {
    public List<Integer> rightSideView(TreeNode root) {
        List<Integer> ans = new ArrayList<>();
        dfs(root, 0, ans);
        return ans;
    }

    // 根 → 右 → 左：每个深度第一个被访问的就是右视图
    public static void dfs(TreeNode node, int depth, List<Integer> ans) {
        if (node == null) {
            return;
        }
        if (depth == ans.size()) {      // 该层还没人登记
            ans.add(node.val);
        }
        // 注意先右后左
        dfs(node.right, depth + 1, ans);
        dfs(node.left, depth + 1, ans);
    }

    public static void main(String[] args) {
        // ACM 输入：一行层序序列，null 表示空节点
        Scanner sc = new Scanner(System.in);
        TreeNode root = buildTree(sc);
        System.out.println(new Solution().rightSideView(root));
        sc.close();
    }

    // 按层序遍历读入建树，null 表示空节点
    public static TreeNode buildTree(Scanner sc) {
        if (!sc.hasNext()) {
            return null;
        }
        String token = sc.next();
        if ("null".equals(token)) {
            return null;
        }
        TreeNode root = new TreeNode(Integer.parseInt(token));
        Queue<TreeNode> queue = new LinkedList<>();
        queue.offer(root);
        while (!queue.isEmpty()) {
            TreeNode cur = queue.poll();
            if (!sc.hasNext()) {
                break;
            }
            String left = sc.next();
            if (!"null".equals(left)) {
                cur.left = new TreeNode(Integer.parseInt(left));
                queue.offer(cur.left);
            }
            if (!sc.hasNext()) {
                break;
            }
            String right = sc.next();
            if (!"null".equals(right)) {
                cur.right = new TreeNode(Integer.parseInt(right));
                queue.offer(cur.right);
            }
        }
        return root;
    }
}

class TreeNode {
    int val;
    TreeNode left;
    TreeNode right;

    TreeNode() {}

    TreeNode(int val) { this.val = val; }

    TreeNode(int val, TreeNode left, TreeNode right) {
        this.val = val;
        this.left = left;
        this.right = right;
    }
}
```

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct TreeNode {
    int val;
    struct TreeNode* left;
    struct TreeNode* right;
} TreeNode;

// 根 → 右 → 左：每个深度第一个被访问的就是右视图
void dfs(struct TreeNode* node, int depth, int* ans, int* size) {
    if (node == NULL) {
        return;
    }
    if (depth == *size) {       // 该层还没人登记
        ans[(*size)++] = node->val;
    }
    dfs(node->right, depth + 1, ans, size);   // 先右后左
    dfs(node->left, depth + 1, ans, size);
}

// 按层序遍历读入建树，"null" 表示空节点
TreeNode* buildTree(char tokens[][16], int n) {
    if (n == 0 || strcmp(tokens[0], "null") == 0) {
        return NULL;
    }
    TreeNode** queue = (TreeNode**)malloc(n * sizeof(TreeNode*));
    int head = 0, tail = 0, i = 1;
    TreeNode* root = (TreeNode*)malloc(sizeof(TreeNode));
    root->val = atoi(tokens[0]);
    root->left = root->right = NULL;
    queue[tail++] = root;
    while (head < tail && i < n) {
        TreeNode* cur = queue[head++];
        if (strcmp(tokens[i], "null") != 0) {
            cur->left = (TreeNode*)malloc(sizeof(TreeNode));
            cur->left->val = atoi(tokens[i]);
            cur->left->left = cur->left->right = NULL;
            queue[tail++] = cur->left;
        }
        i++;
        if (i >= n) {
            break;
        }
        if (strcmp(tokens[i], "null") != 0) {
            cur->right = (TreeNode*)malloc(sizeof(TreeNode));
            cur->right->val = atoi(tokens[i]);
            cur->right->left = cur->right->right = NULL;
            queue[tail++] = cur->right;
        }
        i++;
    }
    free(queue);
    return root;
}

int main(void) {
    // ACM 输入：一行层序序列，null 表示空节点
    char tokens[400][16];
    int n = 0;
    while (n < 400 && scanf("%15s", tokens[n]) == 1) {
        n++;
    }
    if (n == 0 || strcmp(tokens[0], "null") == 0) {
        printf("[]\n");
        return 0;
    }
    TreeNode* root = buildTree(tokens, n);
    int ans[400];
    int size = 0;
    dfs(root, 0, ans, &size);
    printf("[");

        printf("%d%s", ans[i], i + 1 < size ? ", " : "");
```
