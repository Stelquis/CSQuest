# [141. 环形链表](https://leetcode.cn/problems/linked-list-cycle/)

- 🟢 **难度**：简单
- 🔗 **分类**：链表
- 🏷️ **标签**：`哈希表`、`链表`、`双指针`、`Floyd 判圈算法`

## 题目描述

给你一个链表的头节点 `head` ，判断链表中是否有环。

如果链表中有某个节点，可以通过连续跟踪 `next` 指针再次到达，则链表中存在环。 为了表示给定链表中的环，评测系统内部使用整数 `pos` 来表示链表尾连接到链表中的位置（索引从 0 开始）。**注意：`pos` 不作为参数进行传递** 。仅仅是为了标识链表的实际情况。

*如果链表中存在环* ，则返回 `true` 。 否则，返回 `false` 。

**示例 1：**

![](https://assets.leetcode.cn/aliyun-lc-upload/uploads/2018/12/07/circularlinkedlist.png)

```
输入：head = [3,2,0,-4], pos = 1
输出：true
解释：链表中有一个环，其尾部连接到第二个节点。
```

**示例 2：**

![](https://assets.leetcode.cn/aliyun-lc-upload/uploads/2018/12/07/circularlinkedlist_test2.png)

```
输入：head = [1,2], pos = 0
输出：true
解释：链表中有一个环，其尾部连接到第一个节点。
```

**示例 3：**

![](https://assets.leetcode.cn/aliyun-lc-upload/uploads/2018/12/07/circularlinkedlist_test3.png)

```
输入：head = [1], pos = -1
输出：false
解释：链表中没有环。
```

**提示：**
- 链表中节点的数目范围是 `[0, 10^4]`
- `-10^5 <= Node.val <= 10^5`
- `pos` 为 `-1` 或者链表中的一个 **有效索引** 。

**进阶：** 你能用 `O(1)`（即，常量）内存解决此问题吗？

## 题解：Floyd 快慢指针判圈（O(1) 空间）

思路：慢指针一次走一步、快指针一次走两步。无环时快指针必然率先撞到 `null`；有环时两者都在环内，快指针每轮追近一步，必然与慢指针相遇。时间 `O(n)`，空间 `O(1)`。本实现让 `fast` 先行一步（`head.next`）起步，天然规避了"起点相同立即判相等"的边界问题。

### ☕ Java

```java
import java.util.Scanner;

public class Solution {
    public boolean hasCycle(ListNode head) {
        if (head == null || head.next == null) return false;
        ListNode slow = head, fast = head.next;
        while (slow != fast) {
            if (fast == null || fast.next == null) return false;  // 无环：快指针率先出界
            slow = slow.next;          // 慢指针一步
            fast = fast.next.next;     // 快指针两步
        }
        return true;                   // 相遇即有环
    }

    public static ListNode buildList(int[] vals, int pos) {
        if (vals.length == 0) return null;
        ListNode dummy = new ListNode(0), p = dummy;
        ListNode[] nodes = new ListNode[vals.length];
        for (int i = 0; i < vals.length; i++) {
            p.next = new ListNode(vals[i]);
            p = p.next;
            nodes[i] = p;
        }
        if (pos >= 0) p.next = nodes[pos];   // 尾部接到下标 pos，制造环
        return dummy.next;
    }

    public static void main(String[] args) {
        // ACM 输入：第一行 n pos，第二行 n 个整数（pos = -1 表示无环）
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int pos = sc.nextInt();
        int[] vals = new int[n];
        for (int i = 0; i < n; i++) {
            vals[i] = sc.nextInt();
        }
        System.out.println(new Solution().hasCycle(buildList(vals, pos)));
        sc.close();
    }
}

// 单文件运行时入口类需在前，故 ListNode 定义置于 Solution 之后
class ListNode {
    int val;
    ListNode next;
    ListNode() {}
    ListNode(int val) { this.val = val; }
    ListNode(int val, ListNode next) { this.val = val; this.next = next; }
}
```

### 🐍 Python

```python
import sys
from typing import Optional


class ListNode:
    def __init__(self, x: int):
        self.val = x
        self.next = None


class Solution:
    def hasCycle(self, head: Optional[ListNode]) -> bool:
        if head is None or head.next is None:
            return False
        slow, fast = head, head.next
        while slow is not fast:
            if fast is None or fast.next is None:  # 无环：快指针率先出界
                return False
            slow = slow.next            # 慢指针一步
            fast = fast.next.next       # 快指针两步
        return True                     # 相遇即有环


def build_list(vals, pos):
    if not vals:
        return None
    dummy = ListNode(0)
    p, nodes = dummy, []
    for v in vals:
        p.next = ListNode(v)
        p = p.next
        nodes.append(p)
    if pos >= 0:
        p.next = nodes[pos]  # 尾部接到下标 pos，制造环
    return dummy.next


if __name__ == "__main__":
    # ACM 输入：第一行 n pos，第二行 n 个整数（pos = -1 表示无环）
    data = sys.stdin.read().split()
    n, pos = int(data[0]), int(data[1])
    vals = [int(x) for x in data[2:2 + n]]
    print(Solution().hasCycle(build_list(vals, pos)))
```

### ➕ C++

```cpp
#include <vector>
#include <iostream>
using namespace std;

struct ListNode {
    int val;
    ListNode* next;
    ListNode(int x) : val(x), next(nullptr) {}
};

class Solution {
public:
    bool hasCycle(ListNode* head) {
        if (head == nullptr || head->next == nullptr) return false;
        ListNode* slow = head;
        ListNode* fast = head->next;
        while (slow != fast) {
            if (fast == nullptr || fast->next == nullptr) return false;  // 无环：快指针率先出界
            slow = slow->next;         // 慢指针一步
            fast = fast->next->next;   // 快指针两步
        }
        return true;                   // 相遇即有环
    }
};

ListNode* buildList(const vector<int>& vals, int pos) {
    if (vals.empty()) return nullptr;
    ListNode dummy(0);
    ListNode* p = &dummy;
    vector<ListNode*> nodes;
    for (int v : vals) {
        p->next = new ListNode(v);
        p = p->next;
        nodes.push_back(p);
    }
    if (pos >= 0) p->next = nodes[pos];  // 尾部接到下标 pos，制造环
    return dummy.next;
}

int main() {
    // ACM 输入：第一行 n pos，第二行 n 个整数（pos = -1 表示无环）
    int n, pos;
    cin >> n >> pos;
    vector<int> vals(n);
    for (int i = 0; i < n; i++) {
        cin >> vals[i];
    }
    cout << (Solution().hasCycle(buildList(vals, pos)) ? "true" : "false") << endl;
    return 0;
}
```

### ⚙️ C

```c
#include <stdio.h>
#include <stdlib.h>
#include <stdbool.h>

struct ListNode {
    int val;
    struct ListNode* next;
};

bool hasCycle(struct ListNode* head) {
    if (head == NULL || head->next == NULL) return false;
    struct ListNode* slow = head;
    struct ListNode* fast = head->next;
    while (slow != fast) {
        if (fast == NULL || fast->next == NULL) return false;  // 无环：快指针率先出界
        slow = slow->next;          // 慢指针一步
        fast = fast->next->next;    // 快指针两步
    }
    return true;                    // 相遇即有环
}

struct ListNode* buildList(const int* vals, int n, int pos) {
    if (n == 0) return NULL;
    struct ListNode dummy;
    struct ListNode* p = &dummy;
    dummy.next = NULL;
    struct ListNode** nodes = (struct ListNode**)malloc(n * sizeof(struct ListNode*));
    for (int i = 0; i < n; i++) {
        p->next = (struct ListNode*)malloc(sizeof(struct ListNode));
        p = p->next;
        p->val = vals[i];
        p->next = NULL;
        nodes[i] = p;
    }
    if (pos >= 0) p->next = nodes[pos];  // 尾部接到下标 pos，制造环
    struct ListNode* head = dummy.next;
    free(nodes);
    return head;
}

int main(void) {
    // ACM 输入：第一行 n pos，第二行 n 个整数（pos = -1 表示无环）
    int n, pos;
    if (scanf("%d %d", &n, &pos) != 2) {
        return 0;
    }
    int* vals = (int*)malloc(n * sizeof(int));
    for (int i = 0; i < n; i++) {
        scanf("%d", &vals[i]);
    }
    printf("%s\n", hasCycle(buildList(vals, n, pos)) ? "true" : "false");
    free(vals);
    return 0;
}
```
