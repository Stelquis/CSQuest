# [148. 排序链表](https://leetcode.cn/problems/sort-list/)

- 🟡 **难度**：中等
- 🔗 **分类**：链表
- 🏷️ **标签**：`链表`、`双指针`、`分治`、`排序`、`归并排序`

## 题目描述

给你链表的头结点 `head` ，请将其按 **升序** 排列并返回 **排序后的链表** 。

**示例 1：**

![](https://assets.leetcode.com/uploads/2020/09/14/sort_list_1.jpg)

```
输入：head = [4,2,1,3]
输出：[1,2,3,4]
```

**示例 2：**

![](https://assets.leetcode.com/uploads/2020/09/14/sort_list_2.jpg)

```
输入：head = [-1,5,3,4,0]
输出：[-1,0,3,4,5]
```

**示例 3：**

```
输入：head = []
输出：[]
```

**提示：**
- 链表中节点的数目在范围 `[0, 5 * 10^4]` 内
- `-10^5 <= Node.val <= 10^5`

**进阶：** 你可以在 `O(n log n)` 时间复杂度和常数级空间复杂度下，对链表进行排序吗？

## 题解一：转数组排序（O(n) 额外空间）

思路：把节点值抄进 `ArrayList`，`Collections.sort` 排序后重建链表。时间 `O(n log n)`，空间 `O(n)`（数组 + 重建的 n 个新节点）。

实测性能只 beat 约 25%，三个原因：① `List<Integer>` 装箱拆箱（int → Integer 每个元素包一层对象）；② `Collections.sort` 对对象走 TimSort 比较排序，慢于基本类型排序；③ 重建时二次分配 n 个节点。此外该解法用 O(n) 额外空间绕开了"链表排序"的考点，不满足进阶要求——题解二的归并排序才是本题指定答案。

### ☕ Java

```java
import java.util.ArrayList;
import java.util.Collections;
import java.util.List;
import java.util.Scanner;

public class Solution {
    public ListNode sortList(ListNode head) {
        List<Integer> arr = new ArrayList<>();
        while (head != null) {
            arr.add(head.val);
            head = head.next;
        }
        Collections.sort(arr);
        ListNode dummy = new ListNode();
        ListNode cur = dummy;
        for (int i = 0; i < arr.size(); i++) {
            cur.next = new ListNode(arr.get(i));
            cur = cur.next;
        }
        return dummy.next;
    }

    public static ListNode buildList(int[] vals) {
        ListNode dummy = new ListNode(0), p = dummy;
        for (int v : vals) {
            p.next = new ListNode(v);
            p = p.next;
        }
        return dummy.next;
    }

    public static void main(String[] args) {
        // ACM 输入：第一行 n，第二行 n 个整数
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] vals = new int[n];
        for (int i = 0; i < n; i++) {
            vals[i] = sc.nextInt();
        }
        ListNode res = new Solution().sortList(buildList(vals));
        StringBuilder sb = new StringBuilder();
        for (ListNode p = res; p != null; p = p.next) {
            if (p != res) sb.append(' ');
            sb.append(p.val);
        }
        System.out.println(sb);
        sc.close();
    }
}

// 单文件运行时入口类需在前，故 ListNode 定义置于 Solution 之后
class ListNode {
    int val;
    ListNode next;
    ListNode() {}
    ListNode(int val) { this.val = val; }
    ListNode(int val, ListNode next) { this.val = val; this.next = next; }
}
```

```python
import sys
from typing import Optional


class ListNode:
    def __init__(self, val: int = 0, next: "Optional[ListNode]" = None):
        self.val = val
        self.next = next


class Solution:
    def sortList(self, head: Optional[ListNode]) -> Optional[ListNode]:
        arr = []
        while head is not None:
            arr.append(head.val)
            head = head.next
        arr.sort()
        dummy = ListNode(0)
        cur = dummy
        for v in arr:
            cur.next = ListNode(v)
            cur = cur.next
        return dummy.next


def build_list(vals):
    dummy = ListNode(0)
    p = dummy
    for v in vals:
        p.next = ListNode(v)
        p = p.next
    return dummy.next


if __name__ == "__main__":
    # ACM 输入：第一行 n，第二行 n 个整数
    data = sys.stdin.read().split()
    n = int(data[0])
    vals = [int(x) for x in data[1:1 + n]]
    res = Solution().sortList(build_list(vals))
    out = []
    while res is not None:
        out.append(str(res.val))
        res = res.next
    print(" ".join(out))
```

### ➕ C++

```cpp
#include <vector>
#include <algorithm>
#include <iostream>
using namespace std;

struct ListNode {
    int val;
    ListNode* next;
    ListNode() : val(0), next(nullptr) {}
    ListNode(int x) : val(x), next(nullptr) {}
    ListNode(int x, ListNode* next) : val(x), next(next) {}
};

class Solution {
public:
    ListNode* sortList(ListNode* head) {
        vector<int> arr;
        for (ListNode* p = head; p != nullptr; p = p->next) {
            arr.push_back(p->val);
        }
        sort(arr.begin(), arr.end());
        ListNode dummy(0);
        ListNode* cur = &dummy;
        for (int v : arr) {
            cur->next = new ListNode(v);
            cur = cur->next;
        }
        return dummy.next;
    }
};

ListNode* buildList(const vector<int>& vals) {
    ListNode dummy(0);
    ListNode* p = &dummy;
    for (int v : vals) {
        p->next = new ListNode(v);
        p = p->next;
    }
    return dummy.next;
}

int main() {
    // ACM 输入：第一行 n，第二行 n 个整数
    int n;
    cin >> n;
    vector<int> vals(n);
    for (int i = 0; i < n; i++) {
        cin >> vals[i];
    }
    ListNode* res = Solution().sortList(buildList(vals));
    for (ListNode* p = res; p != nullptr; p = p->next) {
        if (p != res) cout << ' ';
        cout << p->val;
    }
    cout << endl;
    return 0;
}
```

### ⚙️ C

```c
#include <stdio.h>
#include <stdlib.h>

struct ListNode {
    int val;
    struct ListNode* next;
};

static int cmp(const void* a, const void* b) {
    return *(const int*)a - *(const int*)b;
}

struct ListNode* sortList(struct ListNode* head) {
    int n = 0;
    for (struct ListNode* p = head; p != NULL; p = p->next) n++;
    int* arr = (int*)malloc(n * sizeof(int));
    int i = 0;
    for (struct ListNode* p = head; p != NULL; p = p->next) {
        arr[i++] = p->val;
    }
    qsort(arr, n, sizeof(int), cmp);
    struct ListNode dummy;
    struct ListNode* cur = &dummy;
    dummy.next = NULL;
    for (int j = 0; j < n; j++) {
        cur->next = (struct ListNode*)malloc(sizeof(struct ListNode));
        cur = cur->next;
        cur->val = arr[j];
        cur->next = NULL;
    }
    free(arr);
    return dummy.next;
}

static struct ListNode* buildList(const int* vals, int n) {
    struct ListNode dummy;
    struct ListNode* p = &dummy;
    dummy.next = NULL;
    for (int i = 0; i < n; i++) {
        p->next = (struct ListNode*)malloc(sizeof(struct ListNode));
        p = p->next;
        p->val = vals[i];
        p->next = NULL;
    }
    return dummy.next;
}

int main(void) {
    /* ACM 输入：第一行 n，第二行 n 个整数 */
    int n;
    if (scanf("%d", &n) != 1) {
        return 0;
    }
    int* vals = (int*)malloc(n * sizeof(int));
    for (int i = 0; i < n; i++) {
        scanf("%d", &vals[i]);
    }
    struct ListNode* res = sortList(buildList(vals, n));
    for (struct ListNode* p = res; p != NULL; p = p->next) {
        if (p != res) printf(" ");
        printf("%d", p->val);
    }
    printf("\n");
    free(vals);
    return 0;
}
```
