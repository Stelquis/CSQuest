# [23. 合并 K 个升序链表](https://leetcode.cn/problems/merge-k-sorted-lists/)

- 🔴 **难度**：困难
- 🔗 **分类**：链表
- 🏷️ **标签**：`链表`、`分治`、`堆（优先队列）`、`归并排序`、`锦标赛排序`

## 题目描述

给你一个链表数组，每个链表都已经按升序排列。

请你将所有链表合并到一个升序链表中，返回合并后的链表。

**示例 1：**

```
输入：lists = [[1,4,5],[1,3,4],[2,6]]
输出：[1,1,2,3,4,4,5,6]
解释：链表数组如下：
[
  1->4->5,
  1->3->4,
  2->6
]
将它们合并到一个有序链表中得到。
1->1->2->3->4->4->5->6
```

**示例 2：**

```
输入：lists = []
输出：[]
```

**示例 3：**

```
输入：lists = [[]]
输出：[]
```

**提示：**
- `k == lists.length`
- `0 <= k <= 10^4`
- `0 <= lists[i].length <= 500`
- `-10^4 <= lists[i][j] <= 10^4`
- `lists[i]` 按 **升序** 排列
- `lists[i].length` 的总和不超过 `10^4`

## 题解一：顺序两两合并（O(N·k)）

思路：维护已合并结果 `ans`，逐个吞并 `lists[i]`——每次都是一次"合并两个有序链表"（21 题）。时间 `O(N·k)`（N 为节点总数：第 i 轮要扫完整条 `ans`），本题 N 总和 ≤ 10^4，可以过，但 k 很大时接近 10^8 有 TLE 风险；升级解法见题解二（分治 O(N·log k)）。

注意 `lists` 为空数组时需提前返回 null（否则 `lists[0]` 越界）。

### ☕ Java

```java
import java.util.Scanner;

public class Solution {
    public ListNode mergeKLists(ListNode[] lists) {
        if (lists.length == 0 || (lists.length == 1 && lists[0] == null)) return null;
        if (lists.length == 1) return lists[0];
        ListNode ans = mergeTwoLists(lists[0], lists[1]);
        for (int i = 2; i < lists.length; i++) {
            ans = mergeTwoLists(ans, lists[i]);
        }
        return ans;
    }

    public ListNode mergeTwoLists(ListNode list1, ListNode list2) {
        if (list1 == null) return list2;
        if (list2 == null) return list1;
        ListNode head = new ListNode();
        ListNode cur = head;
        // 初始化：先确定第一个节点
        if (list1.val <= list2.val) {
            head.val = list1.val;
            list1 = list1.next;
        } else {
            head.val = list2.val;
            list2 = list2.next;
        }
        while (list1 != null && list2 != null) {
            if (list1.val <= list2.val) {
                cur.next = new ListNode(list1.val);
                list1 = list1.next;
            } else {
                cur.next = new ListNode(list2.val);
                list2 = list2.next;
            }
            cur = cur.next;   // 关键：尾指针前进，否则新节点只会覆盖同一位置
        }
        cur.next = list1 == null ? list2 : list1;   // 接上未耗尽的剩余部分
        return head;
    }

    public static ListNode buildList(int[] vals) {
        ListNode dummy = new ListNode(0), p = dummy;
        for (int v : vals) {
            p.next = new ListNode(v);
            p = p.next;
        }
        return dummy.next;
    }

    public static void main(String[] args) {
        // ACM 输入：第一行 k，随后每个链表先给长度 n 再给 n 个升序整数；
        // 长度为 0 表示空链表
        Scanner sc = new Scanner(System.in);
        int k = sc.nextInt();
        ListNode[] lists = new ListNode[k];
        for (int i = 0; i < k; i++) {
            int n = sc.nextInt();
            int[] vals = new int[n];
            for (int j = 0; j < n; j++) {
                vals[j] = sc.nextInt();
            }
            lists[i] = buildList(vals);
        }
        ListNode res = new Solution().mergeKLists(lists);
        StringBuilder sb = new StringBuilder();
        for (ListNode p = res; p != null; p = p.next) {
            if (p != res) sb.append(' ');
            sb.append(p.val);
        }
        System.out.println(sb);
        sc.close();
    }
}

// 单文件运行时入口类需在前，故 ListNode 定义置于 Solution 之后

    int val;
    ListNode next;
    ListNode() {}
    ListNode(int val) { this.val = val; }
    ListNode(int val, ListNode next) { this.val = val; this.next = next; }
}
```

```python
            return None

            return lists[0]

        for i in range(2, len(lists)):
            ans = merge_two(ans, lists[i])
```

## 题解二：队列版淘汰赛（O(N·log k)）

### ☕ Java

```java
import java.util.ArrayDeque;
import java.util.Deque;
import java.util.Scanner;

public class Solution {
    public ListNode mergeKLists(ListNode[] lists) {
        Deque<ListNode> deque = new ArrayDeque<>();
        for (ListNode list : lists) {
            if (list != null) deque.add(list);   // null 链不入队
        }
        while (deque.size() > 1) {
            ListNode list1 = deque.poll();
            ListNode list2 = deque.poll();
            deque.add(mergeTwoLists(list1, list2));
        }
        return deque.poll();   // 幸存者；空数组时队列本就为空，返回 null
    }

    public ListNode mergeTwoLists(ListNode list1, ListNode list2) {
        if (list1 == null) return list2;
        if (list2 == null) return list1;
        ListNode head = new ListNode();
        ListNode cur = head;
        // 初始化：先确定第一个节点
        if (list1.val <= list2.val) {
            head.val = list1.val;
            list1 = list1.next;
        } else {
            head.val = list2.val;
            list2 = list2.next;
        }
        while (list1 != null && list2 != null) {
            if (list1.val <= list2.val) {
                cur.next = new ListNode(list1.val);
                list1 = list1.next;
            } else {
                cur.next = new ListNode(list2.val);
                list2 = list2.next;
            }
            cur = cur.next;   // 关键：尾指针前进，否则新节点只会覆盖同一位置
        }
        cur.next = list1 == null ? list2 : list1;   // 接上未耗尽的剩余部分
        return head;
    }

    public static void main(String[] args) {
        // ACM 输入：第一行 k，随后每个链表先给长度 n 再给 n 个升序整数
        Scanner sc = new Scanner(System.in);
        int k = sc.nextInt();
        ListNode[] lists = new ListNode[k];
        for (int i = 0; i < k; i++) {
            int n = sc.nextInt();
            int[] vals = new int[n];
            for (int j = 0; j < n; j++) {
                vals[j] = sc.nextInt();
            }
            ListNode dummy = new ListNode(0), p = dummy;
            for (int j = 0; j < n; j++) {
                p.next = new ListNode(vals[j]);
                p = p.next;
            }
            lists[i] = dummy.next;
        }
        ListNode res = new Solution().mergeKLists(lists);
        StringBuilder sb = new StringBuilder();
        for (ListNode p = res; p != null; p = p.next) {
            if (p != res) sb.append(' ');
            sb.append(p.val);
        }
        System.out.println(sb);
        sc.close();
    }
}
```

```python
import sys
from typing import List, Optional
from collections import deque


class ListNode:
    def __init__(self, val: int = 0, next: "Optional[ListNode]" = None):
        self.val = val
        self.next = next


class Solution:
    def mergeKLists(self, lists: List[Optional[ListNode]]) -> Optional[ListNode]:
        dq = deque(lst for lst in lists if lst is not None)   # null 链不入队
        while len(dq) > 1:
            a = dq.popleft()
            b = dq.popleft()
            dq.append(self._merge_two(a, b))
        return dq.popleft() if dq else None

    def _merge_two(a, b):

        cur = dummy

                a = a.next
            else:
                cur.next = ListNode(b.val)
                b = b.next
            cur = cur.next
        cur.next = a if a is not None else b
        return dummy.next
```

```cpp
#include <vector>

#include <iostream>
using namespace std;

struct ListNode {
    int val;
    ListNode* next;
    ListNode() : val(0), next(nullptr) {}
    ListNode(int x) : val(x), next(nullptr) {}
    ListNode(int x, ListNode* next) : val(x), next(next) {}
};

class Solution {
public:
    ListNode* mergeKLists(vector<ListNode*>& lists) {
        deque<ListNode*> dq;
        for (ListNode* l : lists) {
            if (l != nullptr) dq.push_back(l);   // null 链不入队
        }
        while (dq.size() > 1) {
            ListNode* a = dq.front(); dq.pop_front();
            ListNode* b = dq.front(); dq.pop_front();
            dq.push_back(mergeTwoLists(a, b));
        }
        return dq.empty() ? nullptr : dq.front();
    }

                b = b->next;
            }
            cur = cur->next;
        }
        cur->next = a != nullptr ? a : b;
        return dummy.next;
    }
};
```

```c
#include <stdio.h>
#include <stdlib.h>

struct ListNode {
    int val;
    struct ListNode* next;
};

static struct ListNode* mergeTwoLists(struct ListNode* a, struct ListNode* b) {
    if (a == NULL) return b;
    if (b == NULL) return a;
    struct ListNode dummy;
    struct ListNode* cur = &dummy;
    dummy.next = NULL;
    while (a != NULL && b != NULL) {
        if (a->val <= b->val) {
            cur->next = (struct ListNode*)malloc(sizeof(struct ListNode));
            cur = cur->next;
            cur->val = a->val;
            cur->next = NULL;
            a = a->next;
        } else {
            cur->next = (struct ListNode*)malloc(sizeof(struct ListNode));
            cur = cur->next;
            cur->val = b->val;
            cur->next = NULL;
            b = b->next;
        }
    }
    cur->next = a != NULL ? a : b;
    return dummy.next;
}

struct ListNode* mergeKLists(struct ListNode** lists, int k) {
    /* C 版用数组模拟队列：head/tail 下标循环使用 */
    struct ListNode** queue = (struct ListNode**)malloc(k * sizeof(struct ListNode*));
    int count = 0;
    for (int i = 0; i < k; i++) {
        if (lists[i] != NULL) queue[count++] = lists[i];
    }
    int headIdx = 0;
    while (count - headIdx > 1) {
        struct ListNode* a = queue[headIdx++];
        struct ListNode* b = queue[headIdx++];
        queue[count++] = mergeTwoLists(a, b);
    }
    struct ListNode* res = headIdx < count ? queue[headIdx] : NULL;
    free(queue);
    return res;
}

static struct ListNode* buildList(const int* vals, int n) {
    struct ListNode dummy;
    struct ListNode* p = &dummy;
    dummy.next = NULL;
    for (int i = 0; i < n; i++) {
        p->next = (struct ListNode*)malloc(sizeof(struct ListNode));
        p = p->next;
        p->val = vals[i];
        p->next = NULL;
    }
    return dummy.next;
}
```
