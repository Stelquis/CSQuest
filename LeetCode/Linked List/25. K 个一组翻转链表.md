# [25. K 个一组翻转链表](https://leetcode.cn/problems/reverse-nodes-in-k-group/)

- 🔴 **难度**：困难
- 🔗 **分类**：链表
- 🏷️ **标签**：`递归`、`链表`

## 题目描述

给你链表的头节点 `head` ，每 `k` 个节点一组进行翻转，请你返回修改后的链表。

`k` 是一个正整数，它的值小于或等于链表的长度。如果节点总数不是 `k` 的整数倍，那么请将最后剩余的节点保持原有顺序。

你不能只是单纯的改变节点内部的值，而是需要实际进行节点交换。

**示例 1：**

![](https://assets.leetcode.com/uploads/2020/10/03/reverse_ex1.jpg)

```
输入：head = [1,2,3,4,5], k = 2
输出：[2,1,4,3,5]
```

**示例 2：**

![](https://assets.leetcode.com/uploads/2020/10/03/reverse_ex2.jpg)

```
输入：head = [1,2,3,4,5], k = 3
输出：[3,2,1,4,5]
```

**提示：**
- 链表中的节点数目为 `n`
- `1 <= k <= n <= 5000`
- `0 <= Node.val <= 1000`

**进阶：** 你可以设计一个只用 `O(1)` 额外内存空间的算法解决此问题吗？

## 题解：分组探测 + 切断翻转接驳（迭代，O(1) 空间）

思路：dummy 挂头；每轮先用 `end` 从前驱出发探测 k 步（不足 k 或撞 null 则收工）；探测成功后抓拍 `nextGroup`、切断本组、翻转（复用 206 的 `reverseList`），再把**新尾巴**（原组头，翻转前抓拍为 `newTail`）接回 `nextGroup`，`start/end` 滚动到新尾巴。时间 `O(n)`，空间 `O(1)`。

三个易错点：① 翻转会令组内头尾互换——原组头变成新尾巴，必须在翻转**前**抓拍为 `newTail`，否则接驳断链；② 探测起点必须永远是**前驱**（第一轮是 dummy，之后是新尾巴），从组头出发走 k 步会多走一格；③ 探测中的 `end==null` 拦截是"末尾不足 k 个保持原样"规则的实现载体。

### ☕ Java

```java
import java.util.Scanner;

public class Solution {
    public ListNode reverseKGroup(ListNode head, int k) {
        if (head.next == null) return head;
        // 每次找 k 个 Node，将其拆分为独立小链表，翻转后再接回
        ListNode dummy = new ListNode(0);
        dummy.next = head;
        ListNode start = dummy, end = dummy;
        while (end.next != null) {
            // 探测：从 start 走 k 步，end 落在本组最后一个；不足 k 则收工
            for (int i = 0; i < k; i++) {
                end = end.next;
                if (end == null) return dummy.next;
            }
            ListNode nextGroup = end.next;    // 切断前抓拍下一组入口
            end.next = null;                  // 切断本组
            ListNode newTail = start.next;    // 原组头，翻转后成为新尾巴
            start.next = reverseList(start.next);
            newTail.next = nextGroup;         // 新尾巴接回下一组
            start = newTail;                  // 滚动：新尾巴是下一组的前驱
            end = start;
        }
        return dummy.next;
    }

    public static ListNode reverseList(ListNode head) {
        ListNode pre = null, cur = head;
        while (cur != null) {
            ListNode next = cur.next;
            cur.next = pre;
            pre = cur;
            cur = next;
        }
        return pre;
    }

    public static ListNode buildList(int[] vals) {
        ListNode dummy = new ListNode(0), p = dummy;
        for (int v : vals) {
            p.next = new ListNode(v);
            p = p.next;
        }
        return dummy.next;
    }

    public static void main(String[] args) {
        // ACM 输入：第一行 n k，第二行 n 个整数
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        int[] vals = new int[n];
        for (int i = 0; i < n; i++) {
            vals[i] = sc.nextInt();
        }
        ListNode res = new Solution().reverseKGroup(buildList(vals), k);
        StringBuilder sb = new StringBuilder();
        for (ListNode p = res; p != null; p = p.next) {
            if (p != res) sb.append(' ');
            sb.append(p.val);
        }
        System.out.println(sb);
        sc.close();
    }
}

// 单文件运行时入口类需在前，故 ListNode 定义置于 Solution 之后
class ListNode {
    int val;
    ListNode next;

    public ListNode(int val) {
        this.val = val;
    }
}
```

### 🐍 Python

```python
import sys
from typing import Optional


class ListNode:
    def __init__(self, val: int = 0, next: "Optional[ListNode]" = None):
        self.val = val
        self.next = next


class Solution:
    def reverseKGroup(self, head: Optional[ListNode], k: int) -> Optional[ListNode]:
        dummy = ListNode(0, head)
        start = end = dummy
        while end.next is not None:
            for _ in range(k):          # 探测 k 步
                end = end.next
                if end is None:
                    return dummy.next   # 不足 k：收工
            next_group = end.next       # 切断前抓拍
            end.next = None             # 切断本组
            new_tail = start.next       # 原组头 → 新尾巴
            start.next = self._reverse(start.next)
            new_tail.next = next_group
            start = new_tail
            end = start
        return dummy.next

    @staticmethod
    def _reverse(head: Optional[ListNode]) -> Optional[ListNode]:
        prev, cur = None, head
        while cur is not None:
            nxt = cur.next
            cur.next = prev
            prev = cur
            cur = nxt
        return prev


def build_list(vals):
    dummy = ListNode(0)
    p = dummy
    for v in vals:
        p.next = ListNode(v)
        p = p.next
    return dummy.next


if __name__ == "__main__":
    # ACM 输入：第一行 n k，第二行 n 个整数
    data = sys.stdin.read().split()
    n, k = int(data[0]), int(data[1])
    vals = [int(x) for x in data[2:2 + n]]
    res = Solution().reverseKGroup(build_list(vals), k)
    out = []
    while res is not None:
        out.append(str(res.val))
        res = res.next
    print(" ".join(out))
```

```cpp
#include <vector>
#include <iostream>
using namespace std;

struct ListNode {
    int val;
    ListNode* next;
    ListNode(int x) : val(x), next(nullptr) {}
};

class Solution {
public:
    ListNode* reverseKGroup(ListNode* head, int k) {
        ListNode dummy(0);
        dummy.next = head;
        ListNode* start = &dummy;
        ListNode* end = &dummy;
        while (end->next != nullptr) {
            for (int i = 0; i < k; i++) {   // 探测 k 步
                end = end->next;
                if (end == nullptr) return dummy.next;   // 不足 k：收工
            }
            ListNode* nextGroup = end->next;   // 切断前抓拍
            end->next = nullptr;               // 切断本组
            ListNode* newTail = start->next;   // 原组头 → 新尾巴
            start->next = reverseList(start->next);
            newTail->next = nextGroup;
            start = newTail;
            end = start;
        }
        return dummy.next;
    }

private:
    ListNode* reverseList(ListNode* head) {
        ListNode* prev = nullptr;
        ListNode* cur = head;
        while (cur != nullptr) {
            ListNode* next = cur->next;
            cur->next = prev;
            prev = cur;
            cur = next;
        }
        return prev;
    }
};

ListNode* buildList(const vector<int>& vals) {
    ListNode dummy(0);
    ListNode* p = &dummy;
    for (int v : vals) {
        p->next = new ListNode(v);
        p = p->next;
    }
    return dummy.next;
}

    // ACM 输入：第一行 n k，第二行 n 个整数
    int n, k;
    cin >> n >> k;
    vector<int> vals(n);
    for (int i = 0; i < n; i++) {
        cin >> vals[i];

    ListNode* res = Solution().reverseKGroup(buildList(vals), k);
    for (ListNode* p = res; p != nullptr; p = p->next) {
        if (p != res) cout << ' ';
        cout << p->val;
    }
    cout << endl;
    return 0;
}
```

### ⚙️ C

```c
#include <stdio.h>
#include <stdlib.h>

struct ListNode {
    int val;
    struct ListNode* next;
};

static struct ListNode* reverseList(struct ListNode* head) {
    struct ListNode* prev = NULL;
    struct ListNode* cur = head;
    while (cur != NULL) {
        struct ListNode* next = cur->next;
        cur->next = prev;
        prev = cur;
        cur = next;
    }
    return prev;
}

struct ListNode* reverseKGroup(struct ListNode* head, int k) {
    struct ListNode dummy;
    dummy.val = 0;
    dummy.next = head;
    struct ListNode* start = &dummy;
    struct ListNode* end = &dummy;
    while (end->next != NULL) {
        int i;
        for (i = 0; i < k; i++) {           /* 探测 k 步 */
            end = end->next;
            if (end == NULL) return dummy.next;   /* 不足 k：收工 */
        }
        struct ListNode* nextGroup = end->next;   /* 切断前抓拍 */
        end->next = NULL;                         /* 切断本组 */
        struct ListNode* newTail = start->next;   /* 原组头 → 新尾巴 */
        start->next = reverseList(start->next);
        newTail->next = nextGroup;
        start = newTail;
        end = start;
    }
    return dummy.next;
}

static struct ListNode* buildList(const int* vals, int n) {
    struct ListNode dummy;
    struct ListNode* p = &dummy;
    dummy.next = NULL;
    for (int i = 0; i < n; i++) {
        p->next = (struct ListNode*)malloc(sizeof(struct ListNode));
        p = p->next;
        p->val = vals[i];
        p->next = NULL;
    }
    return dummy.next;
}

int main(void) {
    /* ACM 输入：第一行 n k，第二行 n 个整数 */
    int n, k;
    if (scanf("%d %d", &n, &k) != 2) {
        return 0;
    }
    int* vals = (int*)malloc(n * sizeof(int));
    for (int i = 0; i < n; i++) {
        scanf("%d", &vals[i]);
    }
    struct ListNode* res = reverseKGroup(buildList(vals, n), k);
    for (struct ListNode* p = res; p != NULL; p = p->next) {
        if (p != res) printf(" ");
        printf("%d", p->val);
    }
    printf("\n");
    free(vals);
    return 0;
}
```
