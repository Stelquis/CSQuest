# [146. LRU 缓存](https://leetcode.cn/problems/lru-cache/)

- 🟡 **难度**：中等
- 🔗 **分类**：链表
- 🏷️ **标签**：`设计`、`哈希表`、`链表`、`双向链表`

## 题目描述

请你设计并实现一个满足  [LRU (最近最少使用) 缓存](https://baike.baidu.com/item/LRU) 约束的数据结构。

实现 `LRUCache` 类：
- `LRUCache(int capacity)` 以 **正整数** 作为容量 `capacity` 初始化 LRU 缓存
- `int get(int key)` 如果关键字 `key` 存在于缓存中，则返回关键字的值，否则返回 `-1` 。
- `void put(int key, int value)` 如果关键字 `key` 已经存在，则变更其数据值 `value` ；如果不存在，则向缓存中插入该组 `key-value` 。如果插入操作导致关键字数量超过 `capacity` ，则应该 **逐出** 最久未使用的关键字。

函数 `get` 和 `put` 必须以 `O(1)` 的平均时间复杂度运行。

**示例：**

```
输入
["LRUCache", "put", "put", "get", "put", "get", "put", "get", "get", "get"]
[[2], [1, 1], [2, 2], [1], [3, 3], [2], [4, 4], [1], [3], [4]]
输出
[null, null, null, 1, null, -1, null, -1, 3, 4]

解释
LRUCache lRUCache = new LRUCache(2);
lRUCache.put(1, 1); // 缓存是 {1=1}
lRUCache.put(2, 2); // 缓存是 {1=1, 2=2}
lRUCache.get(1);    // 返回 1
lRUCache.put(3, 3); // 该操作会使得关键字 2 作废，缓存是 {1=1, 3=3}
lRUCache.get(2);    // 返回 -1 (未找到)
lRUCache.put(4, 4); // 该操作会使得关键字 1 作废，缓存是 {4=4, 3=3}
lRUCache.get(1);    // 返回 -1 (未找到)
lRUCache.get(3);    // 返回 3
lRUCache.get(4);    // 返回 4
```

**提示：**
- `1 <= capacity <= 3000`
- `0 <= key <= 10000`
- `0 <= value <= 10^5`
- 最多调用 `2 * 10^5` 次 `get` 和 `put`

## 题解：哈希表 + 双向链表（哈希管找得到，链表管排得出）

设计分工：哈希表 `key → 节点引用` 负责 O(1) 定位；双向链表（`head` 哨兵后是最新、`tail` 哨兵前是最旧）负责维护使用顺序。`get/put` 命中后都做"摘下 + 头插"焕新；容量满时淘汰 `tail.prev`（节点自带 key，可回哈希表删条目）。两个哨兵让空缓存也有 `head ⇄ tail` 形态，全程零特判。时间 `O(1)`，空间 `O(capacity)`。

### ☕ Java

```java
import java.util.HashMap;
import java.util.Map;
import java.util.Scanner;

class LRUCache {
    private final Map<Integer, Node> map = new HashMap<>();
    private final int capacity;
    private final Node head, tail;

    private static class Node {
        int key, val;          // key 要存：淘汰时回哈希表删条目用
        Node prev, next;
        Node(int key, int val) {
            this.key = key;
            this.val = val;
        }
    }

    // 摘除：前驱越过 node，后继回指前驱，两根线
    private void remove(Node node) {
        node.prev.next = node.next;
        node.next.prev = node.prev;
        node.prev = null;
        node.next = null;
    }

    // 头插：先装配 node（prev/next），再让链上原节点与 head "承认"它，四根线
    private void addFirst(Node node) {
        head.next.prev = node;
        node.next = head.next;
        head.next = node;
        node.prev = head;
    }

    public LRUCache(int capacity) {
        this.capacity = capacity;
        head = new Node(0, 0);
        tail = new Node(0, 0);
        head.next = tail;
        tail.prev = head;
    }

    public int get(int key) {
        Node node = map.get(key);
        if (node == null) return -1;
        remove(node);
        addFirst(node);        // get 也是一次使用，必须焕新
        return node.val;
    }

    public void put(int key, int value) {
        Node node = map.get(key);
        if (node != null) {
            node.val = value;
            remove(node);
            addFirst(node);
            return;            // 已存在的 key 焕新完就返回，勿落入下方插入逻辑
        }
        if (map.size() == capacity) {   // 满了：淘汰最久未用
            map.remove(tail.prev.key);
            remove(tail.prev);
        }
        Node newNode = new Node(key, value);
        map.put(key, newNode);
        addFirst(newNode);
    }
}

public class Solution {
    public static void main(String[] args) {
        // ACM 输入：第一行 capacity 和操作数 n，接下来 n 行每行为
        //   put key value   或   get key
        Scanner sc = new Scanner(System.in);
        int capacity = sc.nextInt();
        int n = sc.nextInt();
        LRUCache cache = new LRUCache(capacity);
        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < n; i++) {
            String op = sc.next();
            if (op.equals("put")) {
                int k = sc.nextInt(), v = sc.nextInt();
                cache.put(k, v);
            } else {
                int k = sc.nextInt();
                if (sb.length() > 0) sb.append('\n');
                sb.append(cache.get(k));
            }
        }
        System.out.println(sb);
        sc.close();
    }
}
```

### 🐍 Python

```python
import sys


class Node:
    __slots__ = ("key", "val", "prev", "next")

    def __init__(self, key: int = 0, val: int = 0):
        self.key = key
        self.val = val
        self.prev = None
        self.next = None


class LRUCache:
    def __init__(self, capacity: int):
        self.capacity = capacity
        self.map = {}
        self.head = Node()
        self.tail = Node()
        self.head.next = self.tail
        self.tail.prev = self.head

    def _remove(self, node: Node) -> None:
        node.prev.next = node.next
        node.next.prev = node.prev

    def _add_first(self, node: Node) -> None:
        node.next = self.head.next
        node.prev = self.head
        self.head.next.prev = node
        self.head.next = node

    def get(self, key: int) -> int:
        node = self.map.get(key)
        if node is None:
            return -1
        self._remove(node)
        self._add_first(node)
        return node.val

    def put(self, key: int, value: int) -> None:
        node = self.map.get(key)
        if node is not None:
            node.val = value
            self._remove(node)
            self._add_first(node)
            return
        if len(self.map) == self.capacity:
            victim = self.tail.prev
            self._remove(victim)
            del self.map[victim.key]
        fresh = Node(key, value)
        self._add_first(fresh)
        self.map[key] = fresh


def main():
    # ACM 输入：第一行 capacity 和操作数 n，接下来 n 行每行为
    #   put key value   或   get key
    data = sys.stdin.read().split()
    idx = 0
    capacity, n = int(data[idx]), int(data[idx + 1])
    idx += 2
    cache = LRUCache(capacity)
    out = []
    for _ in range(n):
        op = data[idx]
        idx += 1
        if op == "put":
            cache.put(int(data[idx]), int(data[idx + 1]))
            idx += 2
        else:
            out.append(str(cache.get(int(data[idx]))))
            idx += 1
    print("\n".join(out))


if __name__ == "__main__":
    main()
```

### ➕ C++

```cpp
#include <unordered_map>
#include <iostream>
using namespace std;

class LRUCache {
    struct Node {
        int key, val;
        Node *prev, *next;
        Node(int k, int v) : key(k), val(v), prev(nullptr), next(nullptr) {}
    };

    unordered_map<int, Node*> map;
    int capacity;
    Node *head, *tail;

    void remove(Node* node) {
        node->prev->next = node->next;
        node->next->prev = node->prev;
    }

    void addFirst(Node* node) {
        node->next = head->next;
        node->prev = head;
        head->next->prev = node;
        head->next = node;
    }

public:
    LRUCache(int capacity) : capacity(capacity) {
        head = new Node(0, 0);
        tail = new Node(0, 0);
        head->next = tail;
        tail->prev = head;
    }

    int get(int key) {
        auto it = map.find(key);
        if (it == map.end()) return -1;
        Node* node = it->second;
        remove(node);
        addFirst(node);
        return node->val;
    }

    void put(int key, int value) {
        auto it = map.find(key);
        if (it != map.end()) {
            Node* node = it->second;
            node->val = value;
            remove(node);
            addFirst(node);
            return;
        }
        if ((int)map.size() == capacity) {
            Node* victim = tail->prev;
            map.erase(victim->key);
            remove(victim);
            delete victim;
        }
        Node* fresh = new Node(key, value);
        map[key] = fresh;
        addFirst(fresh);
    }
};

int main() {
    // ACM 输入：第一行 capacity 和操作数 n，接下来 n 行每行为
    //   put key value   或   get key
    int capacity, n;
    cin >> capacity >> n;
    LRUCache cache(capacity);
    for (int i = 0; i < n; i++) {
        string op;
        cin >> op;
        if (op == "put") {
            int k, v;
            cin >> k >> v;
            cache.put(k, v);
        } else {
            int k;
            cin >> k;
            cout << cache.get(k) << "\n";
        }
    }
    return 0;
}
```

### ⚙️ C

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

/* 哈希表：手写开放寻址，键为 int key，值为链表节点指针 */
typedef struct Node {
    int key, val;
    struct Node *prev, *next;
} Node;

typedef struct {
    int capacity, cap;
    Node **slot;      /* 哈希桶 */
    Node *head, *tail;
} LRUCache;

static int hashKey(LRUCache *c, int key) {
    unsigned int h = (unsigned int)key * 2654435761u;
    return (int)(h & (c->cap - 1));
}

static void hashMapPut(LRUCache *c, Node *node) {
    int h = hashKey(c, node->key);
    while (c->slot[h] != NULL) h = (h + 1) & (c->cap - 1);
    c->slot[h] = node;
}

static Node *hashMapGet(LRUCache *c, int key) {
    int h = hashKey(c, key);
    while (c->slot[h] != NULL) {
        if (c->slot[h]->key == key) return c->slot[h];
        h = (h + 1) & (c->cap - 1);
    }
    return NULL;
}

static void hashMapRemove(LRUCache *c, int key) {
    int h = hashKey(c, key);
    while (c->slot[h] != NULL) {
        if (c->slot[h]->key == key) { c->slot[h] = NULL; return; }
        h = (h + 1) & (c->cap - 1);
    }
}

static void removeNode(Node *node) {
    node->prev->next = node->next;
    node->next->prev = node->prev;
}

static void addFirst(LRUCache *c, Node *node) {
    node->next = c->head->next;
    node->prev = c->head;
    c->head->next->prev = node;
    c->head->next = node;
}

LRUCache *lRUCacheCreate(int capacity) {
    LRUCache *c = (LRUCache *)malloc(sizeof(LRUCache));
    c->capacity = capacity;
    c->cap = 1024;
    while (c->cap < capacity * 4) c->cap <<= 1;
    c->slot = (Node **)calloc(c->cap, sizeof(Node *));
    c->head = (Node *)malloc(sizeof(Node));
    c->tail = (Node *)malloc(sizeof(Node));
    c->head->next = c->tail;
    c->tail->prev = c->head;
    return c;
}

int lRUCacheGet(LRUCache *c, int key) {
    Node *node = hashMapGet(c, key);
    if (node == NULL) return -1;
    removeNode(node);
    addFirst(c, node);
    return node->val;
}

void lRUCachePut(LRUCache *c, int key, int value) {
    Node *node = hashMapGet(c, key);
    if (node != NULL) {
        node->val = value;
        removeNode(node);
        addFirst(c, node);
        return;
    }
    if (c->capacity == 0) return;
    /* 满员时先淘汰：简单起见统计链表长度 */
    int size = 0;
    for (Node *p = c->head->next; p != c->tail; p = p->next) size++;
    if (size == c->capacity) {
        Node *victim = c->tail->prev;
        hashMapRemove(c, victim->key);
        removeNode(victim);
        free(victim);
    }
    Node *fresh = (Node *)malloc(sizeof(Node));
    fresh->key = key;
    fresh->val = value;
    hashMapPut(c, fresh);
    addFirst(c, fresh);
}

int main(void) {
    /* ACM 输入：第一行 capacity 和操作数 n，接下来 n 行每行为
       put key value   或   get key */
    int capacity, n;
    if (scanf("%d %d", &capacity, &n) != 2) {
        return 0;
    }
    LRUCache *cache = lRUCacheCreate(capacity);
    char op[8];
    for (int i = 0; i < n; i++) {
        scanf("%s", op);
        if (strcmp(op, "put") == 0) {
            int k, v;
            scanf("%d %d", &k, &v);
            lRUCachePut(cache, k, v);
        } else {
            int k;
            scanf("%d", &k);
            printf("%d\n", lRUCacheGet(cache, k));
        }
    }
    return 0;
}
```
