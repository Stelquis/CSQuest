# [142. 环形链表 II](https://leetcode.cn/problems/linked-list-cycle-ii/)

- 🟡 **难度**：中等
- 🔗 **分类**：链表
- 🏷️ **标签**：`哈希表`、`链表`、`双指针`、`Floyd 判圈算法`

## 题目描述

给定一个链表的头节点  `head` ，返回链表开始入环的第一个节点。 *如果链表无环，则返回 `null`。*

如果链表中有某个节点，可以通过连续跟踪 `next` 指针再次到达，则链表中存在环。 为了表示给定链表中的环，评测系统内部使用整数 `pos` 来表示链表尾连接到链表中的位置（**索引从 0 开始**）。如果 `pos` 是 `-1`，则在该链表中没有环。**注意：`pos` 不作为参数进行传递**，仅仅是为了标识链表的实际情况。

**不允许修改** 链表。

**示例 1：**

```
输入：head = [3,2,0,-4], pos = 1
输出：返回索引为 1 的链表节点
解释：链表中有一个环，其尾部连接到第二个节点。
```

**提示：**
- 链表中节点的数目范围在范围 `[0, 10^4]` 内
- `-10^5 <= Node.val <= 10^5`
- `pos` 的值为 `-1` 或者链表中的一个有效索引

**进阶：** 你是否可以使用 `O(1)` 空间解决此题？

## 题解一：哈希表记录访问轨迹（O(n) 空间）

思路：顺序扫描链表，把访问过的节点引用存入 `HashSet`；第一个"已见过"的节点即环入口（无环则扫到 null）。注意必须存**节点引用**而不是 `val`——值会重复，与"同一个节点"是两码事。时间 `O(n)`，空间 `O(n)`。

### ☕ Java

```java
import java.util.HashSet;
import java.util.Scanner;
import java.util.Set;

public class Solution {
    public ListNode detectCycle(ListNode head) {
        Set<ListNode> seen = new HashSet<>();
        ListNode p = head;
        while (p != null) {
            if (!seen.add(p)) return p;   // add 返回 false = 已存在，即环入口
            p = p.next;
        }
        return null;
    }

    public static ListNode buildList(int[] vals, int pos) {
        if (vals.length == 0) return null;
        ListNode dummy = new ListNode(0), p = dummy;
        ListNode[] nodes = new ListNode[vals.length];
        for (int i = 0; i < vals.length; i++) {
            p.next = new ListNode(vals[i]);
            p = p.next;
            nodes[i] = p;
        }
        if (pos >= 0) p.next = nodes[pos];   // 尾部接到下标 pos，制造环
        return dummy.next;
    }

    public static void main(String[] args) {
        // ACM 输入：第一行 n pos，第二行 n 个整数（pos = -1 表示无环）
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int pos = sc.nextInt();
        int[] vals = new int[n];
        for (int i = 0; i < n; i++) {
            vals[i] = sc.nextInt();
        }
        ListNode res = new Solution().detectCycle(buildList(vals, pos));
        System.out.println(res == null ? "no cycle" : "entry val = " + res.val);
        sc.close();
    }
}

// 单文件运行时入口类需在前，故 ListNode 定义置于 Solution 之后
class ListNode {
    int val;
    ListNode next;
    ListNode() {}
    ListNode(int val) { this.val = val; }
    ListNode(int val, ListNode next) { this.val = val; this.next = next; }
}
```

```python
import sys
from typing import Optional


class ListNode:
    def __init__(self, x: int):
        self.val = x
        self.next = None


class Solution:
    def detectCycle(self, head: Optional[ListNode]) -> Optional[ListNode]:
        seen = set()
        p = head
        while p is not None:
            if p in seen:
                return p
            seen.add(p)
            p = p.next
        return None


def build_list(vals, pos):
    if not vals:
        return None
    dummy = ListNode(0)
    p, nodes = dummy, []
    for v in vals:
        p.next = ListNode(v)
        p = p.next
        nodes.append(p)
    if pos >= 0:
        p.next = nodes[pos]
    return dummy.next
```

```cpp
#include <vector>
#include <iostream>
#include <unordered_set>
using namespace std;

struct ListNode {
    int val;
    ListNode* next;
    ListNode(int x) : val(x), next(nullptr) {}
};

class Solution {
public:
    ListNode* detectCycle(ListNode* head) {
        unordered_set<ListNode*> seen;
        ListNode* p = head;
        while (p != nullptr) {
            if (seen.count(p)) return p;
            seen.insert(p);
            p = p->next;
        }
        return nullptr;
    }
};

ListNode* buildList(const vector<int>& vals, int pos) {
    if (vals.empty()) return nullptr;
    ListNode dummy(0);
    ListNode* p = &dummy;
    vector<ListNode*> nodes;

        p->next = new ListNode(v);
        p = p->next;
        nodes.push_back(p);
    }
    if (pos >= 0) p->next = nodes[pos];
    return dummy.next;
}
```

### ⚙️ C

```c
#include <stdio.h>
#include <stdlib.h>
#include <stdint.h>

struct ListNode {
    int val;
    struct ListNode* next;
};

/* C 版哈希用开放寻址手写实现：以节点指针为键 */
struct ListNode* detectCycle(struct ListNode* head) {
    int cap = 1024;
    while (cap < 16384) cap <<= 1;
    struct ListNode** slot = (struct ListNode**)calloc(cap, sizeof(struct ListNode*));
    struct ListNode* p = head;
    while (p != NULL) {
        int h = (int)(((uintptr_t)p / sizeof(struct ListNode)) & (cap - 1));
        while (slot[h] != NULL) {
            if (slot[h] == p) { free(slot); return p; }   /* 已访问过 = 环入口 */
            h = (h + 1) & (cap - 1);
        }
        slot[h] = p;
        p = p->next;
    }
    free(slot);
    return NULL;
}

static struct ListNode* buildList(const int* vals, int n, int pos) {
    if (n == 0) return NULL;
    struct ListNode dummy;
    struct ListNode* p = &dummy;
    dummy.next = NULL;
    struct ListNode** nodes = (struct ListNode**)malloc(n * sizeof(struct ListNode*));
    for (int i = 0; i < n; i++) {
        p->next = (struct ListNode*)malloc(sizeof(struct ListNode));
        p = p->next;
        p->val = vals[i];
        p->next = NULL;
        nodes[i] = p;
    }
    if (pos >= 0) p->next = nodes[pos];
    struct ListNode* head = dummy.next;
    free(nodes);
    return head;
}

int main(void) {
    /* ACM 输入：第一行 n pos，第二行 n 个整数（pos = -1 表示无环） */
    int n, pos;
    if (scanf("%d %d", &n, &pos) != 2) {
        return 0;
    }
    int* vals = (int*)malloc(n * sizeof(int));
    for (int i = 0; i < n; i++) {
        scanf("%d", &vals[i]);
    }
    struct ListNode* res = detectCycle(buildList(vals, n, pos));
    if (res == NULL) {
        printf("no cycle\n");
    } else {
        printf("entry val = %d\n", res->val);
    }
    free(vals);
    return 0;
}
```

## 题解二：Floyd 快慢指针两阶段（O(1) 空间）

思路：阶段一快（2 步）慢（1 步）指针判圈，相遇则有环；阶段二把一个指针放回 `head`，两指针同速前进，再次相遇处即环入口。

数学依据：设头到入口距离 `a`，入口到相遇点 `b`，环剩余 `c`。相遇时慢指针走了 `a+b`（进环后不满一圈必被追上），快指针走了 `2(a+b)`，多走的全在环内：`a+b = k·(b+c)`。变形得 `a = c + (k-1)·(b+c)`——从 `head` 走 `a` 步与从相遇点走 `c` 步（可多绕整圈）在入口会师，故两同速指针必相遇于入口。

### ☕ Java

```java
import java.util.Scanner;

public class Solution {
    public ListNode detectCycle(ListNode head) {
        if (head == null || head.next == null) {
            return null;
        }
        ListNode slow = head;
        ListNode fast = head;
        // 阶段一：判圈
        while (fast != null && fast.next != null) {
            slow = slow.next;
            fast = fast.next.next;
            if (slow == fast) {
                // 阶段二：一个指针回头部，同速前进，再次相遇即入口
                fast = head;
                while (slow != fast) {
                    slow = slow.next;
                    fast = fast.next;
                }
                return slow;
            }
        }
        return null;
    }

    public static ListNode buildList(int[] vals, int pos) {
        if (vals.length == 0) return null;
        ListNode dummy = new ListNode(0), p = dummy;
        ListNode[] nodes = new ListNode[vals.length];
        for (int i = 0; i < vals.length; i++) {
            p.next = new ListNode(vals[i]);
            p = p.next;
            nodes[i] = p;
        }
        if (pos >= 0) p.next = nodes[pos];
        return dummy.next;
    }

    public static void main(String[] args) {
        // ACM 输入：第一行 n pos，第二行 n 个整数（pos = -1 表示无环）
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int pos = sc.nextInt();
        int[] vals = new int[n];
        for (int i = 0; i < n; i++) {
            vals[i] = sc.nextInt();
        }
        ListNode res = new Solution().detectCycle(buildList(vals, pos));
        System.out.println(res == null ? "no cycle" : "entry val = " + res.val);
        sc.close();
    }
}

// 单文件运行时入口类需在前，故 ListNode 定义置于 Solution 之后

    ListNode next;
    ListNode() {}
    ListNode(int val) { this.val = val; }
```
