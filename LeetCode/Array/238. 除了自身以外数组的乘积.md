# [238. 除了自身以外数组的乘积](https://leetcode.cn/problems/product-of-array-except-self/)

- 🟡 **难度**：中等
- 🔢 **分类**：普通数组
- 🏷️ **标签**：`数组`、`前缀和`

## 题目描述

给你一个整数数组 `nums`，返回 数组 `answer` ，其中 `answer[i]` 等于 `nums` 中除了 `nums[i]` 之外其余各元素的乘积 。

题目数据 **保证** 数组 `nums`之中任意元素的全部前缀元素和后缀的乘积都在  **32 位** 整数范围内。

请 **不要使用除法，** 且在 `O(n)` 时间复杂度内完成此题。

**示例 1:**

```
输入: nums = [1,2,3,4]
输出: [24,12,8,6]
```

**示例 2:**

```
输入: nums = [-1,1,0,-3,3]
输出: [0,0,9,0,0]
```

**提示：**
- `2 <= nums.length <= 10^5`
- `-30 <= nums[i] <= 30`
- 输入 **保证** 数组 `answer[i]` 在  **32 位** 整数范围内

**进阶：** 你可以在 `O(1)` 的额外空间复杂度内完成这个题目吗？（ 出于对空间复杂度分析的目的，输出数组 **不被视为** 额外空间。）

## 题解：左右前缀乘积

### ☕ Java

```java
import java.util.Scanner;

class Solution {
    public int[] productExceptSelf(int[] nums) {
        int[] left = new int[nums.length];
        left[0] = nums[0];
        int[] right = new int[nums.length];
        right[nums.length - 1] = nums[nums.length - 1];
        for (int i = 1; i < nums.length; i++) {
            left[i] = left[i - 1] * nums[i];
        }
        for (int i = nums.length - 2; i >= 0; i--) {
            right[i] = right[i + 1] * nums[i];
        }
        int[] ans = new int[nums.length];
        ans[0] = right[1];
        ans[nums.length - 1] = left[nums.length - 2];
        for (int i = 1; i < nums.length - 1; i++) {
            ans[i] = left[i - 1] * right[i + 1];
        }
        return ans;
    }

    public static void main(String[] args) {
        // ACM 输入：第一行 n，第二行 n 个整数
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] nums = new int[n];
        for (int i = 0; i < n; i++) {
            nums[i] = sc.nextInt();
        }
        int[] res = new Solution().productExceptSelf(nums);
        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < res.length; i++) {
            if (i > 0) sb.append(' ');
            sb.append(res[i]);
        }
        System.out.println(sb);
        sc.close();
    }
}
```

### 🐍 Python

```python
import sys
from typing import List


class Solution:
    def productExceptSelf(self, nums: List[int]) -> List[int]:
        n = len(nums)
        left = [0] * n
        right = [0] * n
        left[0] = nums[0]
        right[n - 1] = nums[n - 1]
        # left[i] = nums[0..i] 的乘积
        for i in range(1, n):
            left[i] = left[i - 1] * nums[i]
        # right[i] = nums[i..n-1] 的乘积
        for i in range(n - 2, -1, -1):
            right[i] = right[i + 1] * nums[i]

        ans = [0] * n
        ans[0] = right[1]
        ans[n - 1] = left[n - 2]
        for i in range(1, n - 1):
            ans[i] = left[i - 1] * right[i + 1]
        return ans


if __name__ == "__main__":
    # ACM 输入：第一行 n，第二行 n 个整数
    data = sys.stdin.read().split()
    n = int(data[0])
    nums = [int(x) for x in data[1:1 + n]]
    res = Solution().productExceptSelf(nums)
    print(" ".join(map(str, res)))
```

### ➕ C++

```cpp
#include <vector>
#include <iostream>
using namespace std;

class Solution {
public:
    vector<int> productExceptSelf(vector<int>& nums) {
        int n = nums.size();
        vector<int> left(n), right(n);
        left[0] = nums[0];
        right[n - 1] = nums[n - 1];
        // left[i] = nums[0..i] 的乘积
        for (int i = 1; i < n; i++) {
            left[i] = left[i - 1] * nums[i];
        }
        // right[i] = nums[i..n-1] 的乘积
        for (int i = n - 2; i >= 0; i--) {
            right[i] = right[i + 1] * nums[i];
        }

        vector<int> ans(n);
        ans[0] = right[1];
        ans[n - 1] = left[n - 2];
        for (int i = 1; i < n - 1; i++) {
            ans[i] = left[i - 1] * right[i + 1];
        }
        return ans;
    }
};

int main() {
    // ACM 输入：第一行 n，第二行 n 个整数
    int n;
    cin >> n;
    vector<int> nums(n);
    for (int i = 0; i < n; i++) {
        cin >> nums[i];
    }
    vector<int> res = Solution().productExceptSelf(nums);
    for (int i = 0; i < n; i++) {
        if (i) cout << ' ';
        cout << res[i];
    }
    cout << endl;
    return 0;
}
```

### ⚙️ C

```c
#include <stdio.h>
#include <stdlib.h>

/**
 * Note: The returned array must be malloced, assume caller calls free().
 */
int* productExceptSelf(int* nums, int numsSize, int* returnSize) {
    int n = numsSize;
    int* left = (int*)malloc(n * sizeof(int));
    int* right = (int*)malloc(n * sizeof(int));
    int* ans = (int*)malloc(n * sizeof(int));

    left[0] = nums[0];
    right[n - 1] = nums[n - 1];
    // left[i] = nums[0..i] 的乘积
    for (int i = 1; i < n; i++) {
        left[i] = left[i - 1] * nums[i];
    }
    // right[i] = nums[i..n-1] 的乘积
    for (int i = n - 2; i >= 0; i--) {
        right[i] = right[i + 1] * nums[i];
    }

    ans[0] = right[1];
    ans[n - 1] = left[n - 2];
    for (int i = 1; i < n - 1; i++) {
        ans[i] = left[i - 1] * right[i + 1];
    }

    free(left);
    free(right);
    *returnSize = n;
    return ans;
}

int main(void) {
    // ACM 输入：第一行 n，第二行 n 个整数
    int n;
    if (scanf("%d", &n) != 1) {
        return 0;
    }
    int* nums = (int*)malloc(n * sizeof(int));
    for (int i = 0; i < n; i++) {
        scanf("%d", &nums[i]);
    }

    int returnSize = 0;
    int* res = productExceptSelf(nums, n, &returnSize);
    for (int i = 0; i < returnSize; i++) {
        if (i) printf(" ");
        printf("%d", res[i]);
    }
    printf("\n");

    free(res);
    free(nums);
    return 0;
}
```
