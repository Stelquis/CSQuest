# [189. 轮转数组](https://leetcode.cn/problems/rotate-array/)

- 🟡 **难度**：中等
- 🔢 **分类**：普通数组
- 🏷️ **标签**：`数组`、`数学`、`双指针`

## 题目描述

给定一个整数数组 `nums`，将数组中的元素向右轮转 `k` 个位置，其中 `k` 是非负数。

**示例 1:**

```
输入: nums = [1,2,3,4,5,6,7], k = 3
输出: [5,6,7,1,2,3,4]
解释:
向右轮转 1 步: [7,1,2,3,4,5,6]
向右轮转 2 步: [6,7,1,2,3,4,5]
向右轮转 3 步: [5,6,7,1,2,3,4]
```

**示例 2:**

```
输入：nums = [-1,-100,3,99], k = 2
输出：[3,99,-1,-100]
解释: 
向右轮转 1 步: [99,-1,-100,3]
向右轮转 2 步: [3,99,-1,-100]
```

**提示：**
- `1 <= nums.length <= 10^5`
- `-2^31 <= nums[i] <= 2^31 - 1`
- `0 <= k <= 10^5`

**进阶：**
- 尽可能想出更多的解决方案，至少有 **三种** 不同的方法可以解决这个问题。
- 你可以使用空间复杂度为 `O(1)` 的 **原地** 算法解决这个问题吗？

## 题解：三次翻转

### ☕ Java

```java
import java.util.Scanner;

class Solution {
    public void rotate(int[] nums, int k) {
        k = k % nums.length;
        reverse(nums, 0, nums.length - 1);
        reverse(nums, 0, k - 1);
        reverse(nums, k, nums.length - 1);
    }

    public static void reverse(int[] nums, int s, int e) {
        while (s < e) {
            int temp = nums[s];
            nums[s] = nums[e];
            nums[e] = temp;
            s++;
            e--;
        }
    }

    public static void main(String[] args) {
        // ACM 输入：第一行 n k，第二行 n 个整数
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        int[] nums = new int[n];
        for (int i = 0; i < n; i++) {
            nums[i] = sc.nextInt();
        }
        new Solution().rotate(nums, k);
        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < n; i++) {
            if (i > 0) sb.append(' ');
            sb.append(nums[i]);
        }
        System.out.println(sb);
        sc.close();
    }
}
```

### 🐍 Python

```python
import sys
from typing import List


class Solution:
    def rotate(self, nums: List[int], k: int) -> None:
        n = len(nums)
        k %= n
        self.reverse(nums, 0, n - 1)
        self.reverse(nums, 0, k - 1)
        self.reverse(nums, k, n - 1)

    def reverse(self, nums: List[int], s: int, e: int) -> None:
        while s < e:
            nums[s], nums[e] = nums[e], nums[s]
            s += 1
            e -= 1


if __name__ == "__main__":
    # ACM 输入：第一行 n k，第二行 n 个整数
    data = sys.stdin.read().split()
    n, k = int(data[0]), int(data[1])
    nums = [int(x) for x in data[2:2 + n]]
    Solution().rotate(nums, k)
    print(" ".join(map(str, nums)))
```

### ➕ C++

```cpp
#include <vector>
#include <iostream>
using namespace std;

class Solution {
public:
    void rotate(vector<int>& nums, int k) {
        int n = nums.size();
        k %= n;
        reverse(nums, 0, n - 1);
        reverse(nums, 0, k - 1);
        reverse(nums, k, n - 1);
    }

    static void reverse(vector<int>& nums, int s, int e) {
        while (s < e) {
            int temp = nums[s];
            nums[s] = nums[e];
            nums[e] = temp;
            s++;
            e--;
        }
    }
};

int main() {
    // ACM 输入：第一行 n k，第二行 n 个整数
    int n, k;
    cin >> n >> k;
    vector<int> nums(n);
    for (int i = 0; i < n; i++) {
        cin >> nums[i];
    }
    Solution().rotate(nums, k);
    for (int i = 0; i < n; i++) {
        if (i) cout << ' ';
        cout << nums[i];
    }
    cout << endl;
    return 0;
}
```

### ⚙️ C

```c
#include <stdio.h>
#include <stdlib.h>

static void reverse(int* nums, int s, int e) {
    while (s < e) {
        int temp = nums[s];
        nums[s] = nums[e];
        nums[e] = temp;
        s++;
        e--;
    }
}

void rotate(int* nums, int numsSize, int k) {
    k %= numsSize;
    reverse(nums, 0, numsSize - 1);
    reverse(nums, 0, k - 1);
    reverse(nums, k, numsSize - 1);
}

int main(void) {
    // ACM 输入：第一行 n k，第二行 n 个整数
    int n, k;
    if (scanf("%d %d", &n, &k) != 2) {
        return 0;
    }
    int* nums = (int*)malloc(n * sizeof(int));
    for (int i = 0; i < n; i++) {
        scanf("%d", &nums[i]);
    }
    rotate(nums, n, k);
    for (int i = 0; i < n; i++) {
        if (i) printf(" ");
        printf("%d", nums[i]);
    }
    printf("\n");
    free(nums);
    return 0;
}
```
