# [283. 移动零](https://leetcode.cn/problems/move-zeroes/)

- 🟢 **难度**：简单
- ↔️ **分类**：双指针
- 🏷️ **标签**：`数组`、`双指针`

## 题目描述

给定一个数组 `nums`，编写一个函数将所有 `0` 移动到数组的末尾，同时保持非零元素的相对顺序。

**请注意** ，必须在不复制数组的情况下原地对数组进行操作。

**示例 1:**

```
输入: nums = [0,1,0,3,12]
输出: [1,3,12,0,0]
```

**示例 2:**

```
输入: nums = [0]
输出: [0]
```

**提示**:
- `1 <= nums.length <= 10^4`
- `-2^31 <= nums[i] <= 2^31 - 1`

**进阶：** 你能尽量减少完成的操作次数吗？

## 题解一：查找交换

### ☕ Java

```java
import java.util.Scanner;

class Solution {
    public void moveZeroes(int[] nums) {
        int l = 0;
        while (true) {
            l = findZero(nums, l);
            if (l == -1) break;
            int r = findNum(nums, l + 1);
            if (r == -1) break;
            nums[l] = nums[r];
            nums[r] = 0;
            l++;
        }
    }

    public int findZero(int[] nums, int begin) {
        // 从指定位置开始，往后找到首个0的位置
        while (begin < nums.length && nums[begin] != 0) {
            begin++;
        }
        return begin == nums.length ? -1 : begin;
    }

    public int findNum(int[] nums, int begin) {
        // 从指定位置开始，往后找到首个非0的位置
        while (begin < nums.length && nums[begin] == 0) {
            begin++;
        }
        return begin == nums.length ? -1 : begin;
    }

    public static void main(String[] args) {
        // ACM 输入：第一行 n，第二行 n 个整数
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] nums = new int[n];
        for (int i = 0; i < n; i++) {
            nums[i] = sc.nextInt();
        }
        new Solution().moveZeroes(nums);
        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < n; i++) {
            if (i > 0) sb.append(' ');
            sb.append(nums[i]);
        }
        System.out.println(sb);
        sc.close();
    }
}
```

```python
import sys
from typing import List


class Solution:
    def moveZeroes(self, nums: List[int]) -> None:
        l = 0
        while True:
            l = self.findZero(nums, l)
            if l == -1:
                break
            r = self.findNum(nums, l + 1)
            if r == -1:
                break
            nums[l] = nums[r]
            nums[r] = 0
            l += 1

    def findZero(self, nums: List[int], begin: int) -> int:
        # 从指定位置开始，往后找到首个0的位置
        while begin < len(nums) and nums[begin] != 0:
            begin += 1
        return -1 if begin == len(nums) else begin

    def findNum(self, nums: List[int], begin: int) -> int:
        # 从指定位置开始，往后找到首个非0的位置
        while begin < len(nums) and nums[begin] == 0:
            begin += 1
        return -1 if begin == len(nums) else begin
```

```c
#include <stdio.h>
#include <stdlib.h>

static int findZero(int* nums, int numsSize, int begin) {
    // 从指定位置开始，往后找到首个0的位置
    while (begin < numsSize && nums[begin] != 0) {
        begin++;
    }
    return begin == numsSize ? -1 : begin;
}

static int findNum(int* nums, int numsSize, int begin) {
    // 从指定位置开始，往后找到首个非0的位置
    while (begin < numsSize && nums[begin] == 0) {
        begin++;
    }
    return begin == numsSize ? -1 : begin;
}

void moveZeroes(int* nums, int numsSize) {
    int l = 0;
    while (1) {
        l = findZero(nums, numsSize, l);
        if (l == -1) break;
        int r = findNum(nums, numsSize, l + 1);
        if (r == -1) break;
        nums[l] = nums[r];
        nums[r] = 0;
        l++;
    }
}

int main(void) {
    // ACM 输入：第一行 n，第二行 n 个整数

    if (scanf("%d", &n) != 1) {
        return 0;
    }

        scanf("%d", &nums[i]);

        printf("%d", nums[i]);
    }
```

## 题解二：快慢指针

### ☕ Java

```java
import java.util.Scanner;

class Solution {
    public void moveZeroes(int[] nums) {
        int slow = 0, fast = 0;
        while (fast < nums.length) {
            if (nums[fast] != 0) {
                if (slow != fast) {
                    nums[slow] = nums[fast];
                    nums[fast] = 0;
                }
                slow++;
            }
            fast++;
        }
    }

    public static void main(String[] args) {
        // ACM 输入：第一行 n，第二行 n 个整数
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] nums = new int[n];
        for (int i = 0; i < n; i++) {
            nums[i] = sc.nextInt();
        }
        new Solution().moveZeroes(nums);
        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < n; i++) {
            if (i > 0) sb.append(' ');
            sb.append(nums[i]);
        }
        System.out.println(sb);
        sc.close();
    }
}
```
