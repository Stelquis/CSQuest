# [295. 数据流的中位数](https://leetcode.cn/problems/find-median-from-data-stream/)

- 🔴 **难度**：困难
- ⛰️ **分类**：堆
- 🏷️ **标签**：`设计`、`双指针`、`数据流`、`排序`、`堆（优先队列）`

## 题目描述

**中位数**是有序整数列表中的中间值。如果列表的大小是偶数，则没有中间值，中位数是两个中间值的平均值。
- 例如 `arr = [2,3,4]` 的中位数是 `3` 。
- 例如 `arr = [2,3]` 的中位数是 `(2 + 3) / 2 = 2.5` 。

实现 MedianFinder 类:

-

`MedianFinder()` 初始化 `MedianFinder` 对象。

-

`void addNum(int num)` 将数据流中的整数 `num` 添加到数据结构中。

-

`double findMedian()` 返回到目前为止所有元素的中位数。与实际答案相差 `10^-5` 以内的答案将被接受。

**示例 1：**

```
输入
["MedianFinder", "addNum", "addNum", "findMedian", "addNum", "findMedian"]
[[], [1], [2], [], [3], []]
输出
[null, null, null, 1.5, null, 2.0]

解释
MedianFinder medianFinder = new MedianFinder();
medianFinder.addNum(1);    // arr = [1]
medianFinder.addNum(2);    // arr = [1, 2]
medianFinder.findMedian(); // 返回 1.5 ((1 + 2) / 2)
medianFinder.addNum(3);    // arr[1, 2, 3]
medianFinder.findMedian(); // return 2.0
```

**提示:**
- `-10^5 <= num <= 10^5`
- 在调用 `findMedian` 之前，数据结构中至少有一个元素
- 最多 `5 * 10^4` 次调用 `addNum` 和 `findMedian`

## 题解：对顶堆（大顶堆 + 小顶堆）

数据流随时新增、随时要中位数，重排序是 `O(n log n)` 每次，不可行。对顶堆的思路是**用两个堆把数据流拦腰分成两半**：

- `maxHeap`（大顶堆）存**较小的一半**：堆顶 = 小半部分的最大值，即中位数候选之一；
- `minHeap`（小顶堆）存**较大的一半**：堆顶 = 大半部分的最小值，即另一个候选。

两个堆顶正好顶在"中间"两侧，这就是"对顶"的含义。维护两条规则：

1. **入堆选边**：`num <= maxHeap.peek()`（或堆为空）进 `maxHeap`，否则进 `minHeap`——保证较小的一半里任何数都不大于较大的一半里任何数；
2. **实时平衡**：两堆大小差达到 2 时，把大堆的堆顶搬到小堆，使两堆大小差始终 `<= 1`。

查询时：

- 总数为**奇数**：中位数就是**较大的那个堆**的堆顶；
- 总数为**偶数**：两堆顶的平均值。

模拟 `addNum(1), addNum(2), findMedian(), addNum(3), findMedian()`（左=maxHeap，右=minHeap）：

```
add 1 → [1 | ]              平衡
add 2 → [1 | 2]             平衡
查询  → (1 + 2) / 2 = 1.5
add 3 → [1 | 2,3]           差为 2 → 2 从右搬到左 → [1,2 | 3]
查询  → 左堆更大 → 堆顶 3.0
```

- `addNum` 时间复杂度：`O(log n)`（至多两次进出堆）
- `findMedian` 时间复杂度：`O(1)`
- 空间复杂度：`O(n)`

**易错点**：`maxHeap` / `minHeap` **必须是实例字段，不能加 `static`**——static 字段属于类、被所有实例共享，`new` 第二个 `MedianFinder` 时旧实例的堆会被重置。力扣只造一个实例时加 `static` 也能过，但那是错的（同 155 题的坑）。

### ☕ Java

```java
import java.util.PriorityQueue;
import java.util.Scanner;

class MedianFinder {
    // 大顶堆存较小的一半，小顶堆存较大的一半
    // 二者实时保持平衡，堆顶正好顶在中间两侧
    public PriorityQueue<Integer> maxHeap;
    public PriorityQueue<Integer> minHeap;

    public MedianFinder() {
        maxHeap = new PriorityQueue<>((a, b) -> b - a);
        minHeap = new PriorityQueue<>((a, b) -> a - b);
    }

    public void addNum(int num) {
        if (maxHeap.isEmpty() || maxHeap.peek() >= num) {
            maxHeap.add(num);
        } else {
            minHeap.add(num);
        }
        balance();
    }

    public double findMedian() {
        if (maxHeap.size() == minHeap.size()) {
            return (double) (minHeap.peek() + maxHeap.peek()) / 2;
        } else {
            return maxHeap.size() > minHeap.size() ? maxHeap.peek() : minHeap.peek();
        }
    }

    public void balance() {
        if (Math.abs(maxHeap.size() - minHeap.size()) == 2) {
            if (maxHeap.size() > minHeap.size()) {
                minHeap.add(maxHeap.poll());
            } else {
                maxHeap.add(minHeap.poll());
            }
        }
    }
}

public class Solution {
    public static void main(String[] args) {
        // ACM 输入：第一行操作数 n，接下来 n 行每行为 addNum x / findMedian
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        MedianFinder mf = new MedianFinder();
        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < n; i++) {
            String op = sc.next();
            if (op.equals("addNum")) {
                mf.addNum(sc.nextInt());
            } else {                 // findMedian
                sb.append(mf.findMedian()).append('\n');
            }
        }
        System.out.print(sb);
        sc.close();
    }
}
```

### 🐍 Python

```python
import heapq
import sys


class MedianFinder:
    def __init__(self):
        self.max_heap = []   # 较小的一半（负号实现大顶堆）
        self.min_heap = []   # 较大的一半

    def addNum(self, num: int) -> None:
        if not self.max_heap or num <= -self.max_heap[0]:
            heapq.heappush(self.max_heap, -num)
        else:
            heapq.heappush(self.min_heap, num)
        self._balance()

    def _balance(self) -> None:
        if abs(len(self.max_heap) - len(self.min_heap)) == 2:
            if len(self.max_heap) > len(self.min_heap):
                heapq.heappush(self.min_heap, -heapq.heappop(self.max_heap))
            else:
                heapq.heappush(self.max_heap, -heapq.heappop(self.min_heap))

    def findMedian(self) -> float:
        if len(self.max_heap) == len(self.min_heap):
            return (self.min_heap[0] - self.max_heap[0]) / 2
        top = -self.max_heap[0] if len(self.max_heap) > len(self.min_heap) else self.min_heap[0]
        return float(top)


def main():
    # ACM 输入：第一行操作数 n，接下来 n 行每行为 addNum x / findMedian
    data = sys.stdin.read().split()
    n = int(data[0])
    idx = 1
    mf = MedianFinder()
    out = []
    for _ in range(n):
        op = data[idx]
        idx += 1
        if op == "addNum":
            mf.addNum(int(data[idx]))
            idx += 1
        else:                   # findMedian
            out.append(str(mf.findMedian()))
    print("\n".join(out))


if __name__ == "__main__":
    main()
```

### ➕ C++

```cpp
#include <queue>
#include <string>
#include <iostream>
using namespace std;

class MedianFinder {
    priority_queue<int> maxHeap;                             // 较小的一半（大顶堆）
    priority_queue<int, vector<int>, greater<int>> minHeap;  // 较大的一半（小顶堆）

public:
    MedianFinder() {
    }

    void addNum(int num) {
        if (maxHeap.empty() || maxHeap.top() >= num) {
            maxHeap.push(num);
        } else {
            minHeap.push(num);
        }
        balance();
    }

    double findMedian() {
        if (maxHeap.size() == minHeap.size()) {
            return (minHeap.top() + maxHeap.top()) / 2.0;
        }
        return maxHeap.size() > minHeap.size() ? maxHeap.top() : minHeap.top();
    }

    void balance() {
        if ((int)maxHeap.size() - (int)minHeap.size() == 2) {
            minHeap.push(maxHeap.top());
            maxHeap.pop();
        } else if ((int)minHeap.size() - (int)maxHeap.size() == 2) {
            maxHeap.push(minHeap.top());
            minHeap.pop();
        }
    }
};

int main() {
    // ACM 输入：第一行操作数 n，接下来 n 行每行为 addNum x / findMedian
    int n;
    cin >> n;
    MedianFinder mf;
    for (int i = 0; i < n; i++) {
        string op;
        cin >> op;
        if (op == "addNum") {
            int x;
            cin >> x;
            mf.addNum(x);
        } else {                 // findMedian
            printf("%.1f\n", mf.findMedian());
        }
    }
    return 0;
}
```

### ⚙️ C

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

/* ---------- 手写堆：通过 isMax 开关复用同一套上浮/下沉 ---------- */
typedef struct {
    int *arr;
    int size;
    int isMax;      /* 1: 大顶堆（存较小的一半），0: 小顶堆（存较大的一半） */
} Heap;

static int prior(Heap *h, int a, int b) {   /* a 是否比 b 更优先（更靠近堆顶） */
    return h->isMax ? a > b : a < b;
}

static void swapInt(int *a, int *b) {
    int t = *a;
    *a = *b;
    *b = t;
}

static void siftUp(Heap *h, int i) {
    while (i > 0 && prior(h, h->arr[i], h->arr[(i - 1) / 2])) {
        swapInt(&h->arr[i], &h->arr[(i - 1) / 2]);
        i = (i - 1) / 2;
    }
}

static void siftDown(Heap *h, int i) {
    while (2 * i + 1 < h->size) {
        int child = 2 * i + 1;              /* 左孩子 */
        if (child + 1 < h->size && prior(h, h->arr[child + 1], h->arr[child])) {
            child++;                        /* 选更优先的孩子 */
        }
        if (!prior(h, h->arr[i], h->arr[child])) {
            break;
        }
        swapInt(&h->arr[i], &h->arr[child]);
        i = child;
    }
}

static void heapPush(Heap *h, int val) {
    h->arr[h->size] = val;
    siftUp(h, h->size);
    h->size++;
}

static int heapPop(Heap *h) {
    int top = h->arr[0];
    h->arr[0] = h->arr[--h->size];
    siftDown(h, 0);
    return top;
}

typedef struct {
    Heap maxHeap;   /* 较小的一半 */
    Heap minHeap;   /* 较大的一半 */
} MedianFinder;

MedianFinder *medianFinderCreate(void) {
    MedianFinder *mf = (MedianFinder *)malloc(sizeof(MedianFinder));
    /* 最多 5*10^4 次 addNum，容量取 50001 即可 */
    mf->maxHeap.arr = (int *)malloc(50001 * sizeof(int));
    mf->maxHeap.size = 0;
    mf->maxHeap.isMax = 1;
    mf->minHeap.arr = (int *)malloc(50001 * sizeof(int));
    mf->minHeap.size = 0;
    mf->minHeap.isMax = 0;
    return mf;
}

void medianFinderAddNum(MedianFinder *mf, int num) {
    if (mf->maxHeap.size == 0 || mf->maxHeap.arr[0] >= num) {
        heapPush(&mf->maxHeap, num);
    } else {
        heapPush(&mf->minHeap, num);
    }
    /* 实时平衡：两堆大小差达到 2 时搬运堆顶 */
    if (mf->maxHeap.size - mf->minHeap.size == 2) {
        heapPush(&mf->minHeap, heapPop(&mf->maxHeap));
    } else if (mf->minHeap.size - mf->maxHeap.size == 2) {
        heapPush(&mf->maxHeap, heapPop(&mf->minHeap));
    }
}

double medianFinderFindMedian(MedianFinder *mf) {
    if (mf->maxHeap.size == mf->minHeap.size) {
        return (mf->maxHeap.arr[0] + mf->minHeap.arr[0]) / 2.0;
    }
    return mf->maxHeap.size > mf->minHeap.size
               ? mf->maxHeap.arr[0]
               : mf->minHeap.arr[0];
}

void medianFinderFree(MedianFinder *mf) {
    free(mf->maxHeap.arr);
    free(mf->minHeap.arr);
    free(mf);
}

int main(void) {
    /* ACM 输入：第一行操作数 n，接下来 n 行每行为 addNum x / findMedian */
    int n;
    if (scanf("%d", &n) != 1) {
        return 0;
    }
    MedianFinder *mf = medianFinderCreate();
    char op[16];
    for (int i = 0; i < n; i++) {
        scanf("%s", op);
        if (strcmp(op, "addNum") == 0) {
            int x;
            scanf("%d", &x);
            medianFinderAddNum(mf, x);
        } else {                 /* findMedian */
            printf("%.1f\n", medianFinderFindMedian(mf));
        }
    }
    medianFinderFree(mf);
    return 0;
}
```
