# [347. 前 K 个高频元素](https://leetcode.cn/problems/top-k-frequent-elements/)

- 🟡 **难度**：中等
- ⛰️ **分类**：堆
- 🏷️ **标签**：`数组`、`哈希表`、`分治`、`桶排序`、`计数`、`快速选择`、`排序`、`堆（优先队列）`

## 题目描述

给你一个整数数组 `nums` 和一个整数 `k` ，请你返回其中出现频率前 `k` 高的元素。你可以按 **任意顺序** 返回答案。

**示例 1：**

**输入：** nums = [1,1,1,2,2,3], k = 2

**输出：**[1,2]

**示例 2：**

**输入：** nums = [1], k = 1

**输出：**[1]

**示例 3：**

**输入：** nums = [1,2,1,2,1,2,3,1,3,2], k = 2

**输出：**[1,2]

**提示：**
- `1 <= nums.length <= 10^5`
- `-10^4 <= nums[i] <= 10^4`
- `k` 的取值范围是 `[1, 数组中不相同的元素的个数]`
- 题目数据保证答案唯一，换句话说，数组中前 `k` 个高频元素的集合是唯一的

**进阶：** 你所设计算法的时间复杂度 **必须** 优于 `O(n log n)` ，其中 `n` 是数组大小。

## 题解：哈希计数 + 大顶堆

比 215 多了一步：先要算出"每个数出现几次"。整体两阶段：

1. **哈希计数**：遍历数组，用 `HashMap` 统计每个元素的出现频率，得到 `[元素, 频率]` 对；
2. **大顶堆按频率取前 k**：把所有 `[元素, 频率]` 放进以**频率为大顶堆比较依据**的优先队列（元素值本身不参与比较），随后弹出 `k` 次堆顶，依次取出的就是频率前 `k` 高的元素。

和 215 正好互补：215 是"找第 K 大 → 小顶堆当守门员"；这里是"要频率**前 k 高**的全部元素，且按频率从高到低依次输出" → 直接大顶堆弹 `k` 次更自然。堆的比较器写的是 `(a, b) -> b[1] - a[1]`，用**频率**（下标 1）比大小，取出时再取**元素**（下标 0）。

- 设不同元素个数为 `m`：时间复杂度 `O(n + m log m)`，空间复杂度 `O(m)`

**注**：题目进阶要求优于 `O(n log n)`，可沿两个方向改进——① 反过来用**大小为 `k` 的小顶堆**守门，降为 `O(n log k)`（同 215 思路）；② **桶排序**：频率范围不超过 `n`，按频率建桶倒序收集，可达 `O(n)`。

### ☕ Java

```java
import java.util.HashMap;
import java.util.Map;
import java.util.PriorityQueue;
import java.util.Scanner;

class Solution {
    public int[] topKFrequent(int[] nums, int k) {
        // 1. 统计每个元素的出现频率
        Map<Integer, Integer> map = new HashMap<>();
        for (int num : nums) {
            map.put(num, map.getOrDefault(num, 0) + 1);
        }
        // 2. 大顶堆：按频率（下标 1）从高到低排
        PriorityQueue<int[]> heap = new PriorityQueue<>((a, b) -> b[1] - a[1]);
        for (Map.Entry<Integer, Integer> entry : map.entrySet()) {
            heap.offer(new int[]{entry.getKey(), entry.getValue()});
        }
        // 3. 弹出 k 次，依次取出频率前 k 高的元素
        int[] ans = new int[k];
        for (int i = 0; i < k; i++) {
            ans[i] = heap.poll()[0];
        }
        return ans;
    }

    public static void main(String[] args) {
        // ACM 输入：第一行 n 和 k，第二行 n 个整数
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        int[] nums = new int[n];
        for (int i = 0; i < n; i++) {
            nums[i] = sc.nextInt();
        }
        StringBuilder sb = new StringBuilder();
        for (int x : new Solution().topKFrequent(nums, k)) {
            sb.append(x).append(' ');
        }
        System.out.println(sb.toString().trim());
        sc.close();
    }
}
```

### 🐍 Python

```python
import heapq
import sys
from collections import Counter
from typing import List


class Solution:
    def topKFrequent(self, nums: List[int], k: int) -> List[int]:
        # 1. 统计每个元素的出现频率
        cnt = Counter(nums)
        # 2. 大顶堆：Python 没有大顶堆，用 (-频率, 元素) 变通
        heap = [(-c, x) for x, c in cnt.items()]
        heapq.heapify(heap)
        # 3. 弹出 k 次，依次取出频率前 k 高的元素
        return [heapq.heappop(heap)[1] for _ in range(k)]


if __name__ == "__main__":
    # ACM 输入：第一行 n 和 k，第二行 n 个整数
    data = sys.stdin.read().split()
    n, k = int(data[0]), int(data[1])
    nums = [int(x) for x in data[2:2 + n]]
    print(" ".join(map(str, Solution().topKFrequent(nums, k))))
```

### ➕ C++

```cpp
#include <unordered_map>
#include <queue>
#include <vector>
#include <iostream>
using namespace std;

class Solution {
public:
    vector<int> topKFrequent(vector<int>& nums, int k) {
        // 1. 统计每个元素的出现频率
        unordered_map<int, int> cnt;
        for (int num : nums) {
            cnt[num]++;
        }
        // 2. 大顶堆：默认按 pair 第一位（频率）从高到低
        //    注意：频率相同时的弹出顺序因实现而异，题目允许任意顺序
        priority_queue<pair<int, int>> heap;
        for (auto& [x, c] : cnt) {
            heap.push({c, x});
        }
        // 3. 弹出 k 次，依次取出频率前 k 高的元素
        vector<int> ans;
        for (int i = 0; i < k; i++) {
            ans.push_back(heap.top().second);
            heap.pop();
        }
        return ans;
    }
};

int main() {
    // ACM 输入：第一行 n 和 k，第二行 n 个整数
    int n, k;
    cin >> n >> k;
    vector<int> nums(n);
    for (int i = 0; i < n; i++) {
        cin >> nums[i];
    }
    vector<int> ans = Solution().topKFrequent(nums, k);
    for (int i = 0; i < (int)ans.size(); i++) {
        if (i > 0) cout << ' ';
        cout << ans[i];
    }
    cout << endl;
    return 0;
}
```

### ⚙️ C

```c
#include <stdio.h>
#include <stdlib.h>

/* ---------- 手写哈希表（开放寻址）统计频率 ---------- */
#define HASH_CAP 200003

typedef struct {
    int key;
    int freq;
    int used;
} Entry;

static Entry table[HASH_CAP];

static int hashSlot(int key) {
    int i = key % HASH_CAP;
    if (i < 0) {
        i += HASH_CAP;
    }
    return i;
}

static Entry *hashFind(int key) {
    int i = hashSlot(key);
    while (table[i].used) {
        if (table[i].key == key) {
            return &table[i];
        }
        i = (i + 1) % HASH_CAP;
    }
    return NULL;
}

static Entry *hashInsert(int key) {
    int i = hashSlot(key);
    while (table[i].used) {
        i = (i + 1) % HASH_CAP;
    }
    table[i].used = 1;
    table[i].key = key;
    table[i].freq = 0;
    return &table[i];
}

/* ---------- 手写大顶堆（按频率比较） ---------- */
typedef struct {
    int val;
    int freq;
} Node;

static Node *heapArr;
static int heapSize;

static void swapNode(int i, int j) {
    Node t = heapArr[i];
    heapArr[i] = heapArr[j];
    heapArr[j] = t;
}

static void siftUp(int i) {
    while (i > 0 && heapArr[i].freq > heapArr[(i - 1) / 2].freq) {
        swapNode(i, (i - 1) / 2);
        i = (i - 1) / 2;
    }
}

static void siftDown(int i) {
    while (2 * i + 1 < heapSize) {
        int child = 2 * i + 1;              /* 左孩子 */
        if (child + 1 < heapSize && heapArr[child + 1].freq > heapArr[child].freq) {
            child++;                        /* 选频率较高的孩子 */
        }
        if (heapArr[i].freq >= heapArr[child].freq) {
            break;
        }
        swapNode(i, child);
        i = child;
    }
}

static void heapPush(Node node) {
    heapArr[heapSize] = node;
    siftUp(heapSize);
    heapSize++;
}

static Node heapPop(void) {
    Node top = heapArr[0];
    heapArr[0] = heapArr[--heapSize];
    siftDown(0);
    return top;
}

int* topKFrequent(int* nums, int numsSize, int k, int* returnSize) {
    heapArr = (Node*)malloc(numsSize * sizeof(Node));
    heapSize = 0;
    /* 1. 统计每个元素的出现频率 */
    for (int i = 0; i < numsSize; i++) {
        Entry *e = hashFind(nums[i]);
        if (e == NULL) {
            e = hashInsert(nums[i]);
        }
        e->freq++;
    }
    /* 2. 全部入大顶堆 */
    for (int i = 0; i < HASH_CAP; i++) {
        if (table[i].used) {
            Node node = {table[i].key, table[i].freq};
            heapPush(node);
        }
    }
    /* 3. 弹出 k 次，依次取出频率前 k 高的元素 */
    int *ans = (int*)malloc(k * sizeof(int));
    *returnSize = k;
    for (int i = 0; i < k; i++) {
        ans[i] = heapPop().val;
    }
    free(heapArr);
    return ans;
}

int main(void) {
    /* ACM 输入：第一行 n 和 k，第二行 n 个整数 */
    int n, k;
    if (scanf("%d %d", &n, &k) != 2) {
        return 0;
    }
    int* nums = (int*)malloc(n * sizeof(int));
    for (int i = 0; i < n; i++) {
        scanf("%d", &nums[i]);
    }
    int returnSize;
    int* ans = topKFrequent(nums, n, k, &returnSize);
    for (int i = 0; i < returnSize; i++) {
        if (i > 0) {
            printf(" ");
        }
        printf("%d", ans[i]);
    }
    printf("\n");
    free(ans);
    free(nums);
    return 0;
}
```
