# [234. 回文链表](https://leetcode.cn/problems/palindrome-linked-list/)

- 🟢 **难度**：简单
- 🔗 **分类**：链表
- 🏷️ **标签**：`栈`、`递归`、`链表`、`双指针`

## 题目描述

给你一个单链表的头节点 `head` ，请你判断该链表是否为回文链表。如果是，返回 `true` ；否则，返回 `false` 。

**示例 1：**

![](https://assets.leetcode.com/uploads/2021/03/03/pal1linked-list.jpg)

```
输入：head = [1,2,2,1]
输出：true
```

**示例 2：**

![](https://assets.leetcode.com/uploads/2021/03/03/pal2linked-list.jpg)

```
输入：head = [1,2]
输出：false
```

**提示：**

- 链表中节点数目在范围`[1, 10^5]` 内
- `0 <= Node.val <= 9`

**进阶：** 你能否用 `O(n)` 时间复杂度和 `O(1)` 空间复杂度解决此题？

## 题解一：复制反转比较（O(n) 空间）

思路：复制一份原链 → 原地反转副本 → 副本与原链逐位比较。副本被反转销毁无妨，原链完好，满足了"保持原状"的要求。时间 `O(n)`，空间 `O(n)`（副本），过题足够，进阶的 `O(1)` 空间解法见题解二。

### ☕ Java

```java
import java.util.Scanner;

public class Solution {
    public boolean isPalindrome(ListNode head) {
        if (head == null || head.next == null) return true;
        ListNode copy = copyList(head);
        ListNode res = reverseList(copy);
        while (res != null) {
            if (res.val != head.val) return false;
            res = res.next;
            head = head.next;
        }
        return true;
    }

    public static ListNode copyList(ListNode head) {
        ListNode dummy = new ListNode(0);   // 假节点占位，统一挂接逻辑
        ListNode p = dummy;                 // p：新链的尾巴
        while (head != null) {
            p.next = new ListNode(head.val); // 造一个值相同的新节点挂上
            p = p.next;                      // 新链尾巴前进
            head = head.next;                // 原链读指针前进
        }
        return dummy.next;
    }

    public static ListNode reverseList(ListNode head) {
        ListNode prev = null;
        while (head != null) {
            ListNode next = head.next;   // ① 每轮现场抓住下一个节点
            head.next = prev;            // ② 当前节点箭头掉头
            prev = head;                 // ③ 已反转段向前扩张
            head = next;                 // ④ 前进到下一个待处理节点
        }
        return prev;
    }

    public static ListNode buildList(int[] vals) {
        ListNode dummy = new ListNode(0), p = dummy;
        for (int v : vals) {
            p.next = new ListNode(v);
            p = p.next;
        }
        return dummy.next;
    }

    public static void main(String[] args) {
        // ACM 输入：第一行 n，第二行 n 个整数
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] vals = new int[n];
        for (int i = 0; i < n; i++) {
            vals[i] = sc.nextInt();
        }
        System.out.println(new Solution().isPalindrome(buildList(vals)));
        sc.close();
    }
}

// 单文件运行时入口类需在前，故 ListNode 定义置于 Solution 之后
class ListNode {
    int val;
    ListNode next;
    ListNode() {}
    ListNode(int val) { this.val = val; }
    ListNode(int val, ListNode next) { this.val = val; this.next = next; }
}
```

### 🐍 Python

```python
import sys
from typing import Optional


class ListNode:
    def __init__(self, val: int = 0, next: "Optional[ListNode]" = None):
        self.val = val
        self.next = next


class Solution:
    def isPalindrome(self, head: Optional[ListNode]) -> bool:
        if head is None or head.next is None:
            return True
        # 1. 复制原链（新对象，与原链零共享）
        copy_dummy = ListNode(0)
        tail, p = copy_dummy, head
        while p is not None:
            tail.next = ListNode(p.val)
            tail = tail.next
            p = p.next
        # 2. 原地反转副本
        prev = None
        cur = copy_dummy.next
        while cur is not None:
            nxt = cur.next
            cur.next = prev
            prev = cur
            cur = nxt
        # 3. 副本（已反转）与原链逐位比较
        res, p = prev, head
        while res is not None:
            if res.val != p.val:
                return False
            res = res.next
            p = p.next
        return True


def build_list(vals):
    dummy = ListNode(0)
    p = dummy
    for v in vals:
        p.next = ListNode(v)
        p = p.next
    return dummy.next


if __name__ == "__main__":
    # ACM 输入：第一行 n，第二行 n 个整数
    data = sys.stdin.read().split()
    n = int(data[0])
    vals = [int(x) for x in data[1:1 + n]]
    print(Solution().isPalindrome(build_list(vals)))
```

### ➕ C++

```cpp
#include <vector>
#include <iostream>
using namespace std;

struct ListNode {
    int val;
    ListNode* next;
    ListNode() : val(0), next(nullptr) {}
    ListNode(int x) : val(x), next(nullptr) {}
    ListNode(int x, ListNode* next) : val(x), next(next) {}
};

class Solution {
public:
    bool isPalindrome(ListNode* head) {
        if (head == nullptr || head->next == nullptr) return true;
        // 1. 复制原链
        ListNode copyDummy(0);
        ListNode* tail = &copyDummy;
        for (ListNode* p = head; p != nullptr; p = p->next) {
            tail->next = new ListNode(p->val);
            tail = tail->next;
        }
        // 2. 原地反转副本
        ListNode* prev = nullptr;
        ListNode* cur = copyDummy.next;
        while (cur != nullptr) {
            ListNode* next = cur->next;
            cur->next = prev;
            prev = cur;
            cur = next;
        }
        // 3. 副本（已反转）与原链逐位比较
        for (ListNode* p = head; prev != nullptr; prev = prev->next, p = p->next) {
            if (p->val != prev->val) return false;
        }
        return true;
    }
};

ListNode* buildList(const vector<int>& vals) {
    ListNode dummy(0);
    ListNode* p = &dummy;
    for (int v : vals) {
        p->next = new ListNode(v);
        p = p->next;
    }
    return dummy.next;
}

int main() {
    // ACM 输入：第一行 n，第二行 n 个整数
    int n;
    cin >> n;
    vector<int> vals(n);
    for (int i = 0; i < n; i++) {
        cin >> vals[i];
    }
    cout << (Solution().isPalindrome(buildList(vals)) ? "true" : "false") << endl;
    return 0;
}
```

### ⚙️ C

```c
#include <stdio.h>
#include <stdlib.h>
#include <stdbool.h>

struct ListNode {
    int val;
    struct ListNode* next;
};

static struct ListNode* copyList(struct ListNode* head) {
    struct ListNode dummy;
    struct ListNode* p = &dummy;
    dummy.next = NULL;
    for (; head != NULL; head = head->next) {
        p->next = (struct ListNode*)malloc(sizeof(struct ListNode));
        p = p->next;
        p->val = head->val;
        p->next = NULL;
    }
    return dummy.next;
}

static struct ListNode* reverseList(struct ListNode* head) {
    struct ListNode* prev = NULL;
    while (head != NULL) {
        struct ListNode* next = head->next;
        head->next = prev;
        prev = head;
        head = next;
    }
    return prev;
}

bool isPalindrome(struct ListNode* head) {
    if (head == NULL || head->next == NULL) return true;
    struct ListNode* copy = copyList(head);
    struct ListNode* res = reverseList(copy);
    for (; res != NULL; res = res->next, head = head->next) {
        if (res->val != head->val) return false;
    }
    return true;
}

static struct ListNode* buildList(const int* vals, int n) {
    struct ListNode dummy;
    struct ListNode* p = &dummy;
    dummy.next = NULL;
    for (int i = 0; i < n; i++) {
        p->next = (struct ListNode*)malloc(sizeof(struct ListNode));
        p = p->next;
        p->val = vals[i];
        p->next = NULL;
    }
    return dummy.next;
}

int main(void) {
    // ACM 输入：第一行 n，第二行 n 个整数
    int n;
    if (scanf("%d", &n) != 1) {
        return 0;
    }
    int* vals = (int*)malloc(n * sizeof(int));
    for (int i = 0; i < n; i++) {
        scanf("%d", &vals[i]);
    }
    printf("%s\n", isPalindrome(buildList(vals, n)) ? "true" : "false");
    free(vals);
    return 0;
}
```

## 题解二：反转后半段（O(1) 空间）

思路：p 走到后半段头（`len/2` 步，奇数再跳过中点）→ 原地反转后半段 → 反转链（从尾往头读）与前半段（从头往尾读）同步比较。无需复制、无显式切断，空间 `O(1)`。注意：函数返回后原链的后半段处于反转状态（本题判题不要求恢复原链）。

### ☕ Java

```java
import java.util.Scanner;

public class Solution {
    public boolean isPalindrome(ListNode head) {
        int len = getLength(head);
        ListNode p = head;
        for (int i = 0; i < len / 2; i++) {
            p = p.next;
        }
        if (len % 2 == 1) p = p.next;   // 奇数长度跳过正中间的节点
        ListNode q = reverseList(p);    // 反转后半段，q 为反转后的头
        while (q != null) {             // 后半段不比前半段长，用它做终止标尺
            if (q.val != head.val) return false;
            q = q.next;
            head = head.next;
        }
        return true;
    }

    public static int getLength(ListNode head) {
        int len = 0;
        ListNode p = head;
        while (p != null) {
            len++;
            p = p.next;
        }
        return len;
    }

    public static ListNode reverseList(ListNode head) {
        ListNode prev = null;
        while (head != null) {
            ListNode next = head.next;   // ① 每轮现场抓住下一个节点
            head.next = prev;            // ② 当前节点箭头掉头
            prev = head;                 // ③ 已反转段向前扩张
            head = next;                 // ④ 前进到下一个待处理节点
        }
        return prev;
    }

    public static ListNode buildList(int[] vals) {
        ListNode dummy = new ListNode(0), p = dummy;
        for (int v : vals) {
            p.next = new ListNode(v);
            p = p.next;
        }
        return dummy.next;
    }

    public static void main(String[] args) {
        // ACM 输入：第一行 n，第二行 n 个整数
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] vals = new int[n];
        for (int i = 0; i < n; i++) {
            vals[i] = sc.nextInt();
        }
        System.out.println(new Solution().isPalindrome(buildList(vals)));
        sc.close();
    }
}

// 单文件运行时入口类需在前，故 ListNode 定义置于 Solution 之后

    ListNode next;
    ListNode() {}
    ListNode(int val) { this.val = val; }
    ListNode(int val, ListNode next) { this.val = val; this.next = next; }
}
```

```python
class ListNode:
    def __init__(self, val: int = 0, next: "Optional[ListNode]" = None):

        self.next = next


class Solution:

        while p is not None:
            n += 1
            p = p.next
        p = head
        for _ in range(n // 2):
            p = p.next
        if n % 2 == 1:
            p = p.next  # 奇数长度跳过正中间的节点

            p.next = prev
            prev = p
            p = nxt
        q = prev  # 反转后的头
        while q is not None:
            if q.val != head.val:
                return False
            q = q.next
            head = head.next
        return True
```
