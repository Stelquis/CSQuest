# [138. 随机链表的复制](https://leetcode.cn/problems/copy-list-with-random-pointer/)

- 🟡 **难度**：中等
- 🔗 **分类**：链表
- 🏷️ **标签**：`哈希表`、`链表`

## 题目描述

给你一个长度为 `n` 的链表，每个节点包含一个额外增加的随机指针 `random` ，该指针可以指向链表中的任何节点或空节点。

构造这个链表的 **[深拷贝](https://baike.baidu.com/item/深拷贝/22785317?fr=aladdin)**。 深拷贝应该正好由 `n` 个 **全新** 节点组成，其中每个新节点的值都设为其对应的原节点的值。新节点的 `next` 指针和 `random` 指针也都应指向复制链表中的新节点，并使原链表和复制链表中的这些指针能够表示相同的链表状态。**复制链表中的指针都不应指向原链表中的节点** 。

例如，如果原链表中有 `X` 和 `Y` 两个节点，其中 `X.random --> Y` 。那么在复制链表中对应的两个节点 `x` 和 `y` ，同样有 `x.random --> y` 。

返回复制链表的头节点。

用一个由 `n` 个节点组成的链表来表示输入/输出中的链表。每个节点用一个 `[val, random_index]` 表示：
- `val`：一个表示 `Node.val` 的整数。
- `random_index`：随机指针指向的节点索引（范围从 `0` 到 `n-1`）；如果不指向任何节点，则为  `null` 。

你的代码 **只** 接受原链表的头节点 `head` 作为传入参数。

**示例 1：**

![](https://assets.leetcode.cn/aliyun-lc-upload/uploads/2020/01/09/e1.png)

```
输入：head = [[7,null],[13,0],[11,4],[10,2],[1,0]]
输出：[[7,null],[13,0],[11,4],[10,2],[1,0]]
```

**示例 2：**

![](https://assets.leetcode.cn/aliyun-lc-upload/uploads/2020/01/09/e2.png)

```
输入：head = [[1,1],[2,1]]
输出：[[1,1],[2,1]]
```

**示例 3：**

![](https://assets.leetcode.cn/aliyun-lc-upload/uploads/2020/01/09/e3.png)

```
输入：head = [[3,null],[3,0],[3,null]]
输出：[[3,null],[3,0],[3,null]]
```

**提示：**
- `0 <= n <= 1000`
- `-10^4 <= Node.val <= 10^4`
- `Node.random` 为 `null` 或指向链表中的节点。

## 题解：哈希表两遍扫描（原节点 → 副本 的"翻译字典"）

思路：`Map<Node,Node>` 建立"原节点 → 副本"的户口本。第一遍复制 `next` 并登记户口；第二遍用原链做钥匙串，`map.get(q).random = map.get(q.random)` 把 random 关系整体"翻译"到副本上——`q.random` 为 null 时查表得 null，零特判。副本只在第一遍出生（`new` 仅一处），保证"指向同一个原节点"的 random 在副本里也指向同一个节点。时间 `O(n)`，空间 `O(n)`。

易错点：map 的键值方向必须是**原节点 → 副本**（存成"原节点 → 原节点的 random"等于没建字典；存成"副本 → 原版"会让 random 指进原链）；第二遍必须用**原链**驱动，副本链不能当钥匙串。

### ☕ Java

```java
import java.util.HashMap;
import java.util.Map;
import java.util.Scanner;

public class Solution {
    public Node copyRandomList(Node head) {
        if (head == null) return null;
        Map<Node, Node> map = new HashMap<>();
        Node dummy = new Node(0);
        Node cur = dummy;
        Node p = head;               // 保存原头：第二遍的钥匙串
        while (p != null) {          // 第一遍：复制 next + 登记户口
            Node copy = new Node(p.val);
            cur.next = copy;
            map.put(p, copy);
            cur = cur.next;
            p = p.next;
        }
        Node q = head;
        while (q != null) {          // 第二遍：原链驱动，翻译 random
            map.get(q).random = map.get(q.random);
            q = q.next;
        }
        return dummy.next;
    }

    public static void main(String[] args) {
        // ACM 输入：第一行 n，接下来 n 行每行 val 和 random_index（-1 表示 null）
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        Node[] nodes = new Node[n];
        int[] rnd = new int[n];
        for (int i = 0; i < n; i++) {
            nodes[i] = new Node(sc.nextInt());
            rnd[i] = sc.nextInt();
        }
        for (int i = 0; i < n; i++) {
            if (i + 1 < n) nodes[i].next = nodes[i + 1];
            nodes[i].random = rnd[i] < 0 ? null : nodes[rnd[i]];
        }
        Node res = new Solution().copyRandomList(nodes[0]);
        StringBuilder sb = new StringBuilder();
        int i = 0;
        for (Node p = res; p != null; p = p.next, i++) {
            int r = -1, j = 0;
            for (Node q = res; q != null; q = q.next, j++) {
                if (q == p.random) { r = j; break; }
            }
            if (i > 0) sb.append(' ');
            sb.append('[').append(p.val).append(',').append(r).append(']');
        }
        System.out.println(sb);
        sc.close();
    }
}

// 单文件运行时入口类需在前，故 Node 定义置于 Solution 之后
class Node {
    int val;
    Node next;
    Node random;

    public Node(int val) {
        this.val = val;
        this.next = null;
        this.random = null;
    }
}
```

### 🐍 Python

```python
import sys
from typing import Optional


class Node:
    def __init__(self, x: int, next: "Optional[Node]" = None, random: "Optional[Node]" = None):
        self.val = int(x)
        self.next = next
        self.random = random


class Solution:
    def copyRandomList(self, head: "Optional[Node]") -> "Optional[Node]":
        if head is None:
            return None
        mapping = {}                     # 原节点 → 副本
        dummy = Node(0)
        cur, p = dummy, head
        while p is not None:             # 第一遍：复制 next + 登记
            cur.next = Node(p.val)
            mapping[p] = cur.next
            cur = cur.next
            p = p.next
        q = head
        while q is not None:             # 第二遍：原链驱动，翻译 random
            mapping[q].random = mapping.get(q.random)
            q = q.next
        return dummy.next


def main():
    # ACM 输入：第一行 n，接下来 n 行每行 val 和 random_index（-1 表示 null）
    data = sys.stdin.read().split()
    n = int(data[0])
    nodes = []
    spec = []
    idx = 1
    for i in range(n):
        val, r = int(data[idx]), int(data[idx + 1])
        idx += 2
        nodes.append(Node(val))
        spec.append(r)
    for i in range(n):
        if i + 1 < n:
            nodes[i].next = nodes[i + 1]
        nodes[i].random = None if spec[i] < 0 else nodes[spec[i]]
    res = Solution().copyRandomList(nodes[0])
    arr = []
    p = res
    while p is not None:
        arr.append(p)
        p = p.next
    lines = []
    for i, p in enumerate(arr):
        r = -1
        if p.random is not None:
            r = arr.index(p.random)      # 同一对象才有下标，副本内互指才查得到
        lines.append(f"[{p.val},{r}]")
    print(" ".join(lines))


if __name__ == "__main__":
    main()
```

```cpp
#include <unordered_map>
#include <vector>
#include <iostream>
using namespace std;

struct Node {
    int val;
    Node* next;
    Node* random;
    Node(int val) : val(val), next(nullptr), random(nullptr) {}
};

class Solution {
public:
    Node* copyRandomList(Node* head) {
        if (head == nullptr) return nullptr;
        unordered_map<Node*, Node*> map;   // 原节点 → 副本
        Node dummy(0);
        Node* cur = &dummy;
        Node* p = head;                    // 保存原头：第二遍的钥匙串
        while (p != nullptr) {             // 第一遍：复制 next + 登记
            cur->next = new Node(p->val);
            map[p] = cur->next;
            cur = cur->next;
            p = p->next;
        }
        Node* q = head;
        while (q != nullptr) {             // 第二遍：原链驱动，翻译 random
            map[q]->random = q->random == nullptr ? nullptr : map[q->random];
            q = q->next;
        }
        return dummy.next;
    }
};

int main() {
    // ACM 输入：第一行 n，接下来 n 行每行 val 和 random_index（-1 表示 null）
    int n;
    cin >> n;
    vector<Node*> nodes(n);
    vector<int> rnd(n);
    for (int i = 0; i < n; i++) {
        int v, r;
        cin >> v >> r;
        nodes[i] = new Node(v);
        rnd[i] = r;
    }
    for (int i = 0; i < n; i++) {
        if (i + 1 < n) nodes[i]->next = nodes[i + 1];
        nodes[i]->random = rnd[i] < 0 ? nullptr : nodes[rnd[i]];

    Node* res = Solution().copyRandomList(nodes[0]);
    for (Node* p = res; p != nullptr; p = p->next) {

        if (p != res) cout << ' ';
        cout << '[' << p->val << ',' << r << ']';
    }
    cout << endl;
    return 0;
}
```

### ⚙️ C

```c
#include <stdio.h>
#include <stdlib.h>
#include <stdint.h>

struct Node {
    int val;
    struct Node* next;
    struct Node* random;
};

/* 哈希表：手写开放寻址，键为原节点指针，值为副本节点指针 */
struct Node* copyRandomList(struct Node* head) {
    if (head == NULL) return NULL;
    int cap = 1024;
    while (cap < 4096) cap <<= 1;
    struct Node** keys = (struct Node**)calloc(cap, sizeof(struct Node*));
    struct Node** vals = (struct Node**)calloc(cap, sizeof(struct Node*));

    struct Node dummy;
    struct Node* cur = &dummy;
    dummy.next = NULL;
    struct Node* p = head;                 /* 保存原头：第二遍的钥匙串 */
    while (p != NULL) {                    /* 第一遍：复制 next + 登记 */
        struct Node* copy = (struct Node*)malloc(sizeof(struct Node));
        copy->val = p->val;
        copy->next = NULL;
        copy->random = NULL;
        cur->next = copy;
        cur = cur->next;
        int h = (int)(((uintptr_t)p / sizeof(struct Node)) & (cap - 1));
        while (keys[h] != NULL) h = (h + 1) & (cap - 1);
        keys[h] = p;
        vals[h] = copy;
        p = p->next;
    }
    struct Node* q = head;
    while (q != NULL) {                    /* 第二遍：原链驱动，翻译 random */
        struct Node* copy = NULL;
        int h = (int)(((uintptr_t)q / sizeof(struct Node)) & (cap - 1));
        while (keys[h] != NULL) {
            if (keys[h] == q) { copy = vals[h]; break; }
            h = (h + 1) & (cap - 1);
        }
        if (q->random == NULL) {
            copy->random = NULL;
        } else {
            h = (int)(((uintptr_t)q->random / sizeof(struct Node)) & (cap - 1));
            while (keys[h] != q->random) h = (h + 1) & (cap - 1);
            copy->random = vals[h];
        }
        q = q->next;
    }
    free(keys);
    free(vals);
    return dummy.next;
}

int main(void) {
    /* ACM 输入：第一行 n，接下来 n 行每行 val 和 random_index（-1 表示 null） */
    int n;
    if (scanf("%d", &n) != 1) {
        return 0;
    }
    struct Node** nodes = (struct Node**)malloc(n * sizeof(struct Node*));
    int* rnd = (int*)malloc(n * sizeof(int));
    for (int i = 0; i < n; i++) {
        int v, r;
        scanf("%d %d", &v, &r);
        nodes[i] = (struct Node*)malloc(sizeof(struct Node));
        nodes[i]->val = v;
        nodes[i]->next = NULL;
        nodes[i]->random = NULL;
        rnd[i] = r;
    }
    for (int i = 0; i < n; i++) {
        if (i + 1 < n) nodes[i]->next = nodes[i + 1];
        nodes[i]->random = rnd[i] < 0 ? NULL : nodes[rnd[i]];
    }
    struct Node* res = copyRandomList(nodes[0]);
    for (struct Node* p = res; p != NULL; p = p->next) {
        int r = -1, j = 0;
        for (struct Node* q = res; q != NULL; q = q->next, j++) {
            if (q == p->random) { r = j; break; }
        }
        if (p != res) printf(" ");
        printf("[%d,%d]", p->val, r);
    }
    printf("\n");
    free(nodes);
    free(rnd);
    return 0;
}
```
