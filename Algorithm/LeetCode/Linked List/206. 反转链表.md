# [206. 反转链表](https://leetcode.cn/problems/reverse-linked-list/)

- 🟢 **难度**：简单
- 🔗 **分类**：链表
- 🏷️ **标签**：`递归`、`链表`

## 题目描述

给你单链表的头节点 `head` ，请你反转链表，并返回反转后的链表。

**示例 1：**

![](https://assets.leetcode.com/uploads/2021/02/19/rev1ex1.jpg)

```
输入：head = [1,2,3,4,5]
输出：[5,4,3,2,1]
```

**示例 2：**

![](https://assets.leetcode.com/uploads/2021/02/19/rev1ex2.jpg)

```
输入：head = [1,2]
输出：[2,1]
```

**示例 3：**

```
输入：head = []
输出：[]
```

**提示：**

- 链表中节点的数目范围是 `[0, 5000]`
- `-5000 <= Node.val <= 5000`

**进阶：** 链表可以选用迭代或递归方式完成反转。你能否用两种方法解决这道题？

## 题解：迭代（三指针掉头）

### ☕ Java

```java
import java.util.Scanner;

public class Solution {
    public ListNode reverseList(ListNode head) {
        ListNode prev=null;
        while(head!=null){
            ListNode next=head.next;   // ① 每轮现场抓住下一个节点
            head.next=prev;            // ② 当前节点箭头掉头
            prev=head;                 // ③ 已反转段向前扩张
            head=next;                 // ④ 前进到下一个待处理节点
        }
        return prev;                   // 空链表时循环不进，返回 null，天然正确
    }

    public static ListNode buildList(int[] vals) {
        ListNode dummy = new ListNode(0), p = dummy;
        for (int v : vals) {
            p.next = new ListNode(v);
            p = p.next;
        }
        return dummy.next;
    }

    public static void main(String[] args) {
        // ACM 输入：第一行 n，第二行 n 个整数
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] vals = new int[n];
        for (int i = 0; i < n; i++) {
            vals[i] = sc.nextInt();
        }
        ListNode res = new Solution().reverseList(buildList(vals));
        StringBuilder sb = new StringBuilder();
        for (ListNode p = res; p != null; p = p.next) {
            if (p != res) sb.append(' ');
            sb.append(p.val);
        }
        System.out.println(sb);
        sc.close();
    }
}

// 单文件运行时入口类需在前，故 ListNode 定义置于 Solution 之后
class ListNode {
    int val;
    ListNode next;
    ListNode() {}
    ListNode(int val) { this.val = val; }
    ListNode(int val, ListNode next) { this.val = val; this.next = next; }
}
```

```python
import sys
from typing import Optional


class ListNode:
    def __init__(self, val: int = 0, next: "Optional[ListNode]" = None):
        self.val = val
        self.next = next


class Solution:
    def reverseList(self, head: Optional[ListNode]) -> Optional[ListNode]:
        prev = None
        while head is not None:
            nxt = head.next     # ① 现场抓住下一个节点
            head.next = prev    # ② 箭头掉头
            prev = head         # ③ 已反转段向前扩张
            head = nxt          # ④ 前进
        return prev


def build_list(vals):
    dummy = ListNode(0)
    p = dummy
    for v in vals:
        p.next = ListNode(v)
        p = p.next
    return dummy.next


if __name__ == "__main__":
    # ACM 输入：第一行 n，第二行 n 个整数
    data = sys.stdin.read().split()
    n = int(data[0])
    vals = [int(x) for x in data[1:1 + n]]
    res = Solution().reverseList(build_list(vals))
    out = []
    while res is not None:
        out.append(str(res.val))
        res = res.next
    print(" ".join(out))
```

```cpp
#include <vector>
#include <iostream>
using namespace std;

struct ListNode {
    int val;
    ListNode* next;
    ListNode() : val(0), next(nullptr) {}
    ListNode(int x) : val(x), next(nullptr) {}
    ListNode(int x, ListNode* next) : val(x), next(next) {}
};

class Solution {
public:
    ListNode* reverseList(ListNode* head) {
        ListNode* prev = nullptr;
        while (head != nullptr) {
            ListNode* next = head->next;  // ① 现场抓住下一个节点
            head->next = prev;            // ② 箭头掉头
            prev = head;                  // ③ 已反转段向前扩张
            head = next;                  // ④ 前进
        }
        return prev;
    }
};

ListNode* buildList(const vector<int>& vals) {
    ListNode dummy;
    ListNode* p = &dummy;
    for (int v : vals) {
        p->next = new ListNode(v);
        p = p->next;
    }
    return dummy.next;
}

int main() {
    // ACM 输入：第一行 n，第二行 n 个整数
    int n;
    cin >> n;
    vector<int> vals(n);
    for (int i = 0; i < n; i++) {
        cin >> vals[i];
    }
    ListNode* res = Solution().reverseList(buildList(vals));
    for (ListNode* p = res; p != nullptr; p = p->next) {
        if (p != res) cout << ' ';
        cout << p->val;
    }
    cout << endl;
    return 0;
}
```

### ⚙️ C

```c
#include <stdio.h>
#include <stdlib.h>

struct ListNode {
    int val;
    struct ListNode* next;
};

struct ListNode* reverseList(struct ListNode* head) {
    struct ListNode* prev = NULL;
    while (head != NULL) {
        struct ListNode* next = head->next;  // ① 现场抓住下一个节点
        head->next = prev;                   // ② 箭头掉头
        prev = head;                         // ③ 已反转段向前扩张
        head = next;                         // ④ 前进
    }
    return prev;
}

static struct ListNode* buildList(const int* vals, int n) {
    struct ListNode dummy;
    struct ListNode* p = &dummy;
    dummy.next = NULL;
    for (int i = 0; i < n; i++) {
        p->next = (struct ListNode*)malloc(sizeof(struct ListNode));
        p = p->next;
        p->val = vals[i];
        p->next = NULL;
    }
    return dummy.next;
}

int main(void) {
    // ACM 输入：第一行 n，第二行 n 个整数
    int n;
    if (scanf("%d", &n) != 1) {
        return 0;
    }
    int* vals = (int*)malloc(n * sizeof(int));
    for (int i = 0; i < n; i++) {
        scanf("%d", &vals[i]);
    }

    struct ListNode* res = reverseList(buildList(vals, n));
    for (struct ListNode* p = res; p != NULL; p = p->next) {
        if (p != res) printf(" ");
        printf("%d", p->val);
    }
    printf("\n");

    while (res != NULL) {
        struct ListNode* t = res;
        res = res->next;
        free(t);
    }
    free(vals);
    return 0;
}
```
