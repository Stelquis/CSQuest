# [2. 两数相加](https://leetcode.cn/problems/add-two-numbers/)

- 🟡 **难度**：中等
- 🔗 **分类**：链表
- 🏷️ **标签**：`递归`、`链表`、`数学`

## 题目描述

给你两个 **非空** 的链表，表示两个非负的整数。它们每位数字都是按照 **逆序** 的方式存储的，并且每个节点只能存储 **一位** 数字。

请你将两个数相加，并以相同形式返回一个表示和的链表。

你可以假设除了数字 0 之外，这两个数都不会以 0 开头。

**示例 1：**

![](https://assets.leetcode.cn/aliyun-lc-upload/uploads/2021/01/02/addtwonumber1.jpg)

```
输入：l1 = [2,4,3], l2 = [5,6,4]
输出：[7,0,8]
解释：342 + 465 = 807.
```

**示例 2：**

```
输入：l1 = [0], l2 = [0]
输出：[0]
```

**示例 3：**

```
输入：l1 = [9,9,9,9,9,9,9], l2 = [9,9,9,9]
输出：[8,9,9,9,0,0,0,1]
```

**提示：**
- 每个链表中的节点数在范围 `[1, 100]` 内
- `0 <= Node.val <= 9`
- 题目数据保证列表表示的数字不含前导零

## 题解：dummy + 逐位相加（进位后传）

思路：链表本身就是最低位在前——恰好是加法最友好的顺序，**无需反转**，直接逐位相加。dummy 占位免去头部特判；短链缺位按 0 补；`carry != 0` 放进循环条件，末位进位自动多走一轮。时间 `O(max(m,n))`，空间 `O(1)`（结果链不计）。

### ☕ Java

```java
import java.util.Scanner;

public class Solution {
    public ListNode addTwoNumbers(ListNode l1, ListNode l2) {
        ListNode dummy = new ListNode(0);
        ListNode cur = dummy;
        int carry = 0;
        // 三条件缺一不可：任一链没走完，或还有进位，都得继续
        while (l1 != null || l2 != null || carry != 0) {
            int a = l1 != null ? l1.val : 0;   // 短链缺位补 0，避免 NPE
            int b = l2 != null ? l2.val : 0;
            int sum = a + b + carry;
            carry = sum / 10;
            cur.next = new ListNode(sum % 10);
            cur = cur.next;
            if (l1 != null) l1 = l1.next;
            if (l2 != null) l2 = l2.next;
        }
        return dummy.next;
    }

    public static ListNode buildList(int[] vals) {
        ListNode dummy = new ListNode(0), p = dummy;
        for (int v : vals) {
            p.next = new ListNode(v);
            p = p.next;
        }
        return dummy.next;
    }

    public static void main(String[] args) {
        // ACM 输入：第一行 m n，第二行 m 个整数，第三行 n 个整数
        Scanner sc = new Scanner(System.in);
        int m = sc.nextInt();
        int n = sc.nextInt();
        int[] a = new int[m];
        int[] b = new int[n];
        for (int i = 0; i < m; i++) {
            a[i] = sc.nextInt();
        }
        for (int i = 0; i < n; i++) {
            b[i] = sc.nextInt();
        }
        ListNode res = new Solution().addTwoNumbers(buildList(a), buildList(b));
        StringBuilder sb = new StringBuilder();
        for (ListNode p = res; p != null; p = p.next) {
            if (p != res) sb.append(' ');
            sb.append(p.val);
        }
        System.out.println(sb);
        sc.close();
    }
}

// 单文件运行时入口类需在前，故 ListNode 定义置于 Solution 之后
class ListNode {
    int val;
    ListNode next;
    ListNode() {}
    ListNode(int val) { this.val = val; }
    ListNode(int val, ListNode next) { this.val = val; this.next = next; }
}
```

### 🐍 Python

```python
import sys
from typing import Optional


class ListNode:
    def __init__(self, val: int = 0, next: "Optional[ListNode]" = None):
        self.val = val
        self.next = next


class Solution:
    def addTwoNumbers(self, l1: Optional[ListNode], l2: Optional[ListNode]) -> Optional[ListNode]:
        dummy = ListNode(0)
        cur = dummy
        carry = 0
        while l1 is not None or l2 is not None or carry != 0:
            a = l1.val if l1 is not None else 0
            b = l2.val if l2 is not None else 0
            s = a + b + carry
            carry = s // 10
            cur.next = ListNode(s % 10)
            cur = cur.next
            if l1 is not None:
                l1 = l1.next
            if l2 is not None:
                l2 = l2.next
        return dummy.next


def build_list(vals):
    dummy = ListNode(0)
    p = dummy
    for v in vals:
        p.next = ListNode(v)
        p = p.next
    return dummy.next


if __name__ == "__main__":
    # ACM 输入：第一行 m n，第二行 m 个整数，第三行 n 个整数
    data = sys.stdin.read().split()
    m, n = int(data[0]), int(data[1])
    a = [int(x) for x in data[2:2 + m]]
    b = [int(x) for x in data[2 + m:2 + m + n]]
    res = Solution().addTwoNumbers(build_list(a), build_list(b))
    out = []
    while res is not None:
        out.append(str(res.val))
        res = res.next
    print(" ".join(out))
```

```cpp
    ListNode* addTwoNumbers(ListNode* l1, ListNode* l2) {
        ListNode dummy(0);
        ListNode* cur = &dummy;
        int carry = 0;
        while (l1 != nullptr || l2 != nullptr || carry != 0) {

            int sum = a + b + carry;

            cur->next = new ListNode(sum % 10);
            cur = cur->next;
            if (l1 != nullptr) l1 = l1->next;
            if (l2 != nullptr) l2 = l2->next;
        }
        return dummy.next;
    }
};

    ListNode dummy(0);
    ListNode* p = &dummy;

        p->next = new ListNode(v);
        p = p->next;
    }
    return dummy.next;
```

### ⚙️ C

```c
#include <stdio.h>
#include <stdlib.h>

struct ListNode {
    int val;
    struct ListNode* next;
};

struct ListNode* addTwoNumbers(struct ListNode* l1, struct ListNode* l2) {
    struct ListNode dummy;
    struct ListNode* cur = &dummy;
    dummy.next = NULL;
    int carry = 0;
    while (l1 != NULL || l2 != NULL || carry != 0) {
        int a = l1 != NULL ? l1->val : 0;
        int b = l2 != NULL ? l2->val : 0;
        int sum = a + b + carry;
        carry = sum / 10;
        cur->next = (struct ListNode*)malloc(sizeof(struct ListNode));
        cur = cur->next;
        cur->val = sum % 10;
        cur->next = NULL;
        if (l1 != NULL) l1 = l1->next;
        if (l2 != NULL) l2 = l2->next;
    }
    return dummy.next;
}

static struct ListNode* buildList(const int* vals, int n) {
    struct ListNode dummy;
    struct ListNode* p = &dummy;
    dummy.next = NULL;
    for (int i = 0; i < n; i++) {
        p->next = (struct ListNode*)malloc(sizeof(struct ListNode));
        p = p->next;
        p->val = vals[i];
        p->next = NULL;
    }
    return dummy.next;
}

int main(void) {
    // ACM 输入：第一行 m n，第二行 m 个整数，第三行 n 个整数
    int m, n;
    if (scanf("%d %d", &m, &n) != 2) {
        return 0;
    }
    int* a = (int*)malloc(m * sizeof(int));
    int* b = (int*)malloc(n * sizeof(int));
    for (int i = 0; i < m; i++) {
        scanf("%d", &a[i]);
    }
    for (int i = 0; i < n; i++) {
        scanf("%d", &b[i]);
    }
    struct ListNode* res = addTwoNumbers(buildList(a, m), buildList(b, n));
    for (struct ListNode* p = res; p != NULL; p = p->next) {
        if (p != res) printf(" ");
        printf("%d", p->val);
    }
    printf("\n");
    free(a);
    free(b);
    return 0;
}
```
