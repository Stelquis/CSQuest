# [118. 杨辉三角](https://leetcode.cn/problems/pascals-triangle/)

- 🟢 **难度**：简单
- 🧩 **分类**：动态规划
- 🏷️ **标签**：`数组`、`动态规划`

## 题目描述

给定一个非负整数 *`numRows`，* 生成「杨辉三角」的前 *`numRows`* 行。

在 **「杨辉三角」** 中，每个数是它左上方和右上方的数的和。

![](https://pic.leetcode.cn/1626927345-DZmfxB-PascalTriangleAnimated2.gif)

**示例 1:**

```
输入: numRows = 5
输出: [[1],[1,1],[1,2,1],[1,3,3,1],[1,4,6,4,1]]
```

**示例 2:**

```
输入: numRows = 1
输出: [[1]]
```

**提示:**
- `1 <= numRows <= 30`

## 题解一：迭代逐行构建

杨辉三角的递推关系：第 `i` 行（1-based）共 `i` 个元素，**首尾恒为 1**，中间第 `j` 个元素 = 上一行（`i-1` 行）的第 `j-1` 与第 `j` 个元素之和。逐行构建，每行只依赖上一行。

实现要点（本轮调试踩过的坑）：
- **上一行的下标是 `i-2`**：`ans` 是 0-based 列表，"第 i-1 行"对应下标 `i-2`；写 `i-1` 会恰好取到 `ans.size()`，必越界
- **中间元素个数为 `i-2`**：`numRows ≤ 2` 时内层循环一次都不执行，行由首尾两个 1 拼成——跳过是正常的
- 各行统一用可变 `ArrayList`，避免混入不可变视图（`Arrays.asList`）导致后续修改抛异常

- 时间复杂度：`O(n²)`，总共输出 `n(n+1)/2` 个元素
- 空间复杂度：`O(1)`（不计返回值本身）

### ☕ Java

```java
import java.util.ArrayList;
import java.util.List;
import java.util.Scanner;

public class Solution {
    public List<List<Integer>> generate(int numRows) {
        List<List<Integer>> ans = new ArrayList<>();
        for (int i = 1; i <= numRows; i++) {        // i：当前第几行（1-based）
            List<Integer> row = new ArrayList<>();
            row.add(1);                             // 行首
            for (int j = 1; j < i - 1; j++) {       // 中间共 i-2 个元素
                List<Integer> prev = ans.get(i - 2);   // 上一行（0-based 下标 i-2）
                row.add(prev.get(j - 1) + prev.get(j));
            }
            if (i > 1) {
                row.add(1);                         // 行尾（第 1 行只有一个元素）
            }
            ans.add(row);
        }
        return ans;
    }

    public static void main(String[] args) {
        // ACM 输入：一个整数 numRows
        Scanner sc = new Scanner(System.in);
        int numRows = sc.nextInt();
        System.out.println(new Solution().generate(numRows));
        sc.close();
    }
}
```

### 🐍 Python

```python
import sys
from typing import List


class Solution:
    def generate(self, numRows: int) -> List[List[int]]:
        ans: List[List[int]] = []
        for i in range(1, numRows + 1):          # i：当前第几行（1-based）
            row = [1]                             # 行首
            for _ in range(i - 2):                # 中间共 i-2 个元素
                prev = ans[-1]                    # 上一行
                row.append(prev[-2] + prev[-1])   # 上一行末尾相邻两项
            if i > 1:
                row.append(1)                     # 行尾
            ans.append(row)
        return ans


if __name__ == "__main__":
    # ACM 输入：一个整数 numRows
    numRows = int(sys.stdin.read().split()[0])
    print(Solution().generate(numRows))
```

### ⚙️ C

```c
#include <stdio.h>
#include <stdlib.h>

// 返回行数与每行长度（通过出参），行数据存放在连续的二维数组中
int** generate(int numRows, int** returnColumnSizes, int* returnSize) {
    int** ans = (int**)malloc(numRows * sizeof(int*));
    *returnColumnSizes = (int*)malloc(numRows * sizeof(int));
    *returnSize = numRows;
    for (int i = 1; i <= numRows; i++) {          // i：当前第几行（1-based）
        int* row = (int*)malloc(i * sizeof(int));
        row[0] = 1;                               // 行首
        for (int j = 1; j < i - 1; j++) {         // 中间共 i-2 个元素
            row[j] = ans[i - 2][j - 1] + ans[i - 2][j];   // 上一行相邻两项
        }
        if (i > 1) {
            row[i - 1] = 1;                       // 行尾
        }
        (*returnColumnSizes)[i - 1] = i;
        ans[i - 1] = row;
    }
    return ans;
}

int main(void) {
    // ACM 输入：一个整数 numRows
    int numRows;
    if (scanf("%d", &numRows) != 1) {
        printf("[]\n");
        return 0;
    }
    int returnSize = 0;
    int* colSizes = NULL;
    int** ans = generate(numRows, &colSizes, &returnSize);
    printf("[");
    for (int i = 0; i < returnSize; i++) {
        printf(i ? ", [" : "[");
        for (int j = 0; j < colSizes[i]; j++) {
            printf("%d%s", ans[i][j], j + 1 < colSizes[i] ? ", " : "");
        }
        printf("]");
    }
    printf("]\n");
    return 0;
}
```

## 题解二：递归（自顶向下）

与题解一同一递推式，改为递归表述：`row(i)` 依赖 `row(i-1)`——先递归造出上一行，再据其生成当前行。递归出口是第 1 行。

- 时间复杂度：`O(n²)`
- 空间复杂度：`O(n²)`（输出）+ `O(n)` 递归栈

### ☕ Java

```java
import java.util.ArrayList;
import java.util.List;
import java.util.Scanner;

public class Solution {
    public List<List<Integer>> generate(int numRows) {
        List<List<Integer>> ans = new ArrayList<>();
        if (numRows == 0) {
            return ans;                          // 递归出口之外的防护
        }
        build(numRows, ans);                     // 自顶向下：先造第 numRows 行的上游
        return ans;
    }

    // 递归生成第 1..n 行（按顺序加入 ans）
    public static void build(int n, List<List<Integer>> ans) {
        if (n == 1) {
            ans.add(new ArrayList<>(List.of(1)));    // 递归出口：第 1 行
            return;
        }
        build(n - 1, ans);                       // 先保证上一行已存在
        List<Integer> prev = ans.get(n - 2);     // 上一行（0-based 下标 n-2）
        List<Integer> row = new ArrayList<>();
        row.add(1);
        for (int j = 1; j < n - 1; j++) {
            row.add(prev.get(j - 1) + prev.get(j));
        }
        row.add(1);
        ans.add(row);
    }

    public static void main(String[] args) {
        // ACM 输入：一个整数 numRows
        Scanner sc = new Scanner(System.in);
        int numRows = sc.nextInt();
        System.out.println(new Solution().generate(numRows));
        sc.close();
    }
}
```

### 🐍 Python

```python
import sys
from typing import List


class Solution:
    def generate(self, numRows: int) -> List[List[int]]:
        ans: List[List[int]] = []
        self.build(numRows, ans)
        return ans

    def build(self, n: int, ans: List[List[int]]) -> None:
        if n == 1:
            ans.append([1])                      # 递归出口：第 1 行
            return
        self.build(n - 1, ans)                   # 先保证上一行已存在
        prev = ans[n - 2]                        # 上一行（0-based 下标 n-2）
        row = [1]
        for j in range(1, n - 1):
            row.append(prev[j - 1] + prev[j])
        row.append(1)
        ans.append(row)


if __name__ == "__main__":
    # ACM 输入：一个整数 numRows
    numRows = int(sys.stdin.read().split()[0])
    print(Solution().generate(numRows))
```

```cpp
class Solution {

    vector<vector<int>> generate(int numRows) {

        return ans;
```

### ⚙️ C

```c
#include <stdio.h>
#include <stdlib.h>

static int** ans;
static int* colSizes;

// 递归生成第 1..n 行
void build(int n) {
    if (n == 1) {
        ans[0] = (int*)malloc(sizeof(int));
        ans[0][0] = 1;                           // 递归出口：第 1 行
        colSizes[0] = 1;
        return;
    }
    build(n - 1);                                // 先保证上一行已存在
    int* prev = ans[n - 2];                      // 上一行（0-based 下标 n-2）
    int* row = (int*)malloc(n * sizeof(int));
    row[0] = 1;
    for (int j = 1; j < n - 1; j++) {
        row[j] = prev[j - 1] + prev[j];
    }
    row[n - 1] = 1;
    colSizes[n - 1] = n;
    ans[n - 1] = row;
}

int** generate(int numRows, int** returnColumnSizes, int* returnSize) {
    ans = (int**)malloc(numRows * sizeof(int*));
    colSizes = (int*)malloc(numRows * sizeof(int));
    *returnSize = numRows;
    if (numRows == 0) {
        return ans;
    }
    build(numRows);
    *returnColumnSizes = colSizes;
    return ans;
}

int main(void) {
    // ACM 输入：一个整数 numRows
    int numRows;
    if (scanf("%d", &numRows) != 1) {
        printf("[]\n");
        return 0;
    }
    int returnSize = 0;
    int* colSizes = NULL;
    int** ans = generate(numRows, &colSizes, &returnSize);
    printf("[");
    for (int i = 0; i < returnSize; i++) {
        printf(i ? ", [" : "[");
        for (int j = 0; j < colSizes[i]; j++) {
            printf("%d%s", ans[i][j], j + 1 < colSizes[i] ? ", " : "");
        }
        printf("]");
    }
    printf("]\n");
    return 0;
}
```
