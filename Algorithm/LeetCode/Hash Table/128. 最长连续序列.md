# [128. 最长连续序列](https://leetcode.cn/problems/longest-consecutive-sequence/)

- 🟡 **难度**：中等
- 🔑 **分类**：哈希
- 🏷️ **标签**：`并查集`、`数组`、`哈希表`

## 题目描述

给定一个未排序的整数数组 `nums` ，找出数字连续的最长序列（不要求序列元素在原数组中连续）的长度。

请你设计并实现时间复杂度为 `O(n)` 的算法解决此问题。

**示例 1：**

```
输入：nums = [100,4,200,1,3,2]
输出：4
解释：最长数字连续序列是 [1, 2, 3, 4]。它的长度为 4。
```

**示例 2：**

```
输入：nums = [0,3,7,2,5,8,4,6,0,1]
输出：9
```

**示例 3：**

```
输入：nums = [1,0,1,2]
输出：3
```

**提示：**

- `0 <= nums.length <= 10^5`
- `-10^9 <= nums[i] <= 10^9`

## 题解：哈希集合

### ☕ Java

```java
import java.util.HashSet;
import java.util.Set;

import java.util.Scanner;

class Solution {
    public int longestConsecutive(int[] nums) {
        Set<Integer> s = new HashSet<>();
        for (int num : nums) {
            s.add(num);
        }
        int max = 0;
        for (int num : s) {
            // 只从起点开始
            if (!s.contains(num - 1)) {
                int cur = num;
                int len = 1;

                // 只向右数
                while (s.contains(cur + 1)) {
                    cur++;
                    len++;
                }
                max = Math.max(max, len);
            }
        }
        return max;
    }

    public static void main(String[] args) {
        // ACM 输入：第一行 n，第二行 n 个整数
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] nums = new int[n];
        for (int i = 0; i < n; i++) {
            nums[i] = sc.nextInt();
        }
        System.out.println(new Solution().longestConsecutive(nums));
        sc.close();
    }
}
```

```python
import sys
from typing import List

class Solution:
    def longestConsecutive(self, nums: List[int]) -> int:
        s = set()
        for num in nums:
            s.add(num)

        ans = 0
        for num in s:
            # 只从起点开始
            if num - 1 not in s:
                cur = num
                length = 1

                # 只向右数
                while cur + 1 in s:
                    cur += 1
                    length += 1
                ans = max(ans, length)
        return ans


if __name__ == "__main__":

    data = sys.stdin.read().split()
    n = int(data[0])
    nums = [int(x) for x in data[1:1 + n]]
    print(Solution().longestConsecutive(nums))
```

### ➕ C++

```cpp
#include <algorithm>
#include <unordered_set>
#include <vector>
#include <iostream>
using namespace std;

class Solution {
public:
    int longestConsecutive(vector<int>& nums) {
        unordered_set<int> s;
        for (int num : nums) {
            s.insert(num);
        }

        int ans = 0;
        for (int num : s) {
            // 只从起点开始
            if (!s.count(num - 1)) {
                int cur = num;
                int len = 1;

                // 只向右数
                while (s.count(cur + 1)) {
                    cur++;
                    len++;
                }
                ans = max(ans, len);
            }
        }
        return ans;
    }
};

int main() {
    // ACM 输入：第一行 n，第二行 n 个整数
    int n;
    cin >> n;
    vector<int> nums(n);
    for (int i = 0; i < n; i++) {

}
```

### ⚙️ C

```c
#include <stdio.h>
#include <stdlib.h>

static unsigned int hashOf(int v, int cap) {
    return ((unsigned int)v * 2654435761u) & (cap - 1);
}

static int contains(int* table, char* used, int cap, int v) {
    unsigned int h = hashOf(v, cap);
    while (used[h]) {
        if (table[h] == v) {
            return 1;
        }
        h = (h + 1) & (cap - 1);
    }
    return 0;
}

int longestConsecutive(int* nums, int numsSize) {
    if (numsSize == 0) {
        return 0;
    }

    int cap = 1;
    while (cap < numsSize * 2) {
        cap <<= 1;
    }
    int* table = (int*)malloc(cap * sizeof(int));
    char* used = (char*)calloc(cap, sizeof(char));

    // 去重后放进哈希集合
    for (int i = 0; i < numsSize; i++) {
        if (contains(table, used, cap, nums[i])) {
            continue;
        }
        unsigned int h = hashOf(nums[i], cap);
        while (used[h]) {
            h = (h + 1) & (cap - 1);
        }
        used[h] = 1;
        table[h] = nums[i];
    }

    int ans = 0;
    for (int h = 0; h < cap; h++) {
        if (!used[h]) {
            continue;
        }
        int num = table[h];
        // 只从起点开始
        if (contains(table, used, cap, num - 1)) {
            continue;
        }
        int cur = num;
        int len = 1;
        // 只向右数
        while (contains(table, used, cap, cur + 1)) {
            cur++;
            len++;
        }
        if (len > ans) {
            ans = len;
        }
    }

    free(table);
    free(used);
    return ans;
}

int main(void) {
    // ACM 输入：第一行 n，第二行 n 个整数
    int n;
    if (scanf("%d", &n) != 1) {
        return 0;
    }
    int* nums = (int*)malloc(n * sizeof(int));
    for (int i = 0; i < n; i++) {
        scanf("%d", &nums[i]);
    }
    printf("%d\n", longestConsecutive(nums, n));
    free(nums);
    return 0;
}
```
