# [739. 每日温度](https://leetcode.cn/problems/daily-temperatures/)

- 🟡 **难度**：中等
- 📚 **分类**：栈
- 🏷️ **标签**：`栈`、`数组`、`单调栈`

## 题目描述

给定一个整数数组 `temperatures` ，表示每天的温度，返回一个数组 `answer` ，其中 `answer[i]` 是指对于第 `i` 天，下一个更高温度出现在几天后。如果气温在这之后都不会升高，请在该位置用 `0` 来代替。

**示例 1:**

```
输入: temperatures = [73,74,75,71,69,72,76,73]
输出: [1,1,4,2,1,1,0,0]
```

**示例 2:**

```
输入: temperatures = [30,40,50,60]
输出: [1,1,1,0]
```

**示例 3:**

```
输入: temperatures = [30,60,90]
输出: [1,1,0]
```

**提示：**
- `1 <= temperatures.length <= 10^5`
- `30 <= temperatures[i] <= 100`

## 题解：单调栈（存下标，从左到右）

本质是**"右边第一个更大"的距离版**：对每一天 `i`，找右边第一个满足 `temperatures[j] > temperatures[i]` 的 `j`，答案就是 `j - i`；找不到填 `0`。

用一个栈维护**"还没等到更高温"的那些天**，栈里**只存下标**（要算距离，不能只存值）：

- 遍历每一天 `i`，只要「栈非空 且 `temperatures[栈顶] < temperatures[i]`」：
  - 弹出栈顶 `k`（第 `k` 天终于等到了第 `i` 天），记 `ans[k] = i - k`；
  - 用 `while` 连续做——`i` 可能一次性"暖到"好几天。
- 然后 `push(i)`，它也要等自己的更高温。

为什么弹出时 `i` 就是答案？栈里存的都是"尚未等到更高温"的日子，它们**下标递增、温度递减**。对栈顶的 `k` 来说，从 `k` 到 `i` 之间若有更热的天，`k` 早就被弹掉了；所以 `i` 正是 `k` 右边第一个更热的天。

遍历结束后仍在栈里的天，右边没有更高温，保持初始值 `0` 即可。

- 时间复杂度：`O(n)`，每天进栈一次、出栈最多一次
- 空间复杂度：`O(n)`，栈

### ☕ Java

```java
import java.util.ArrayDeque;
import java.util.Deque;
import java.util.Scanner;

public class Solution {
    public int[] dailyTemperatures(int[] temperatures) {
        int n = temperatures.length;
        int[] ans = new int[n];                       // 默认 0：右边没有更高温的日子
        Deque<Integer> stack = new ArrayDeque<>();    // 存"还没等到更高温"的下标
        for (int i = 0; i < n; i++) {
            while (!stack.isEmpty() && temperatures[stack.peek()] < temperatures[i]) {
                int k = stack.pop();                  // 第 k 天等到了第 i 天
                ans[k] = i - k;
            }
            stack.push(i);                            // 自己入栈，继续等
        }
        return ans;
    }

    public static void main(String[] args) {
        // ACM 输入：第一行 n，第二行 n 个整数
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] temperatures = new int[n];
        for (int i = 0; i < n; i++) {
            temperatures[i] = sc.nextInt();
        }
        int[] ans = new Solution().dailyTemperatures(temperatures);
        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < ans.length; i++) {
            if (i > 0) {
                sb.append(' ');
            }
            sb.append(ans[i]);
        }
        System.out.println(sb);
        sc.close();
    }
}
```

### 🐍 Python

```python
import sys
from typing import List


class Solution:
    def dailyTemperatures(self, temperatures: List[int]) -> List[int]:
        n = len(temperatures)
        ans = [0] * n                       # 默认 0：右边没有更高温的日子
        stack = []                          # 存"还没等到更高温"的下标
        for i in range(n):
            while stack and temperatures[stack[-1]] < temperatures[i]:
                k = stack.pop()             # 第 k 天等到了第 i 天
                ans[k] = i - k
            stack.append(i)                 # 自己入栈，继续等
        return ans


if __name__ == "__main__":
    # ACM 输入：第一行 n，第二行 n 个整数
    data = sys.stdin.read().split()
    n = int(data[0])
    temperatures = [int(x) for x in data[1:1 + n]]
    print(" ".join(map(str, Solution().dailyTemperatures(temperatures))))
```

### ➕ C++

```cpp
#include <vector>
#include <iostream>
using namespace std;

class Solution {
public:
    vector<int> dailyTemperatures(vector<int>& temperatures) {
        int n = (int)temperatures.size();
        vector<int> ans(n, 0);               // 默认 0：右边没有更高温的日子
        vector<int> stack;                   // 存"还没等到更高温"的下标
        for (int i = 0; i < n; i++) {
            while (!stack.empty() && temperatures[stack.back()] < temperatures[i]) {
                int k = stack.back();
                stack.pop_back();
                ans[k] = i - k;              // 第 k 天等到了第 i 天
            }
            stack.push_back(i);              // 自己入栈，继续等
        }
        return ans;
    }
};

int main() {
    // ACM 输入：第一行 n，第二行 n 个整数
    int n;
    cin >> n;
    vector<int> temperatures(n);
    for (int i = 0; i < n; i++) {
        cin >> temperatures[i];
    }
    vector<int> ans = Solution().dailyTemperatures(temperatures);
    for (int i = 0; i < (int)ans.size(); i++) {
        if (i) cout << ' ';
        cout << ans[i];
    }
    cout << endl;
    return 0;
}
```

### ⚙️ C

```c
#include <stdio.h>
#include <stdlib.h>

int* dailyTemperatures(int* temperatures, int temperaturesSize, int* returnSize) {
    int n = temperaturesSize;
    int* ans = (int*)calloc(n, sizeof(int));     /* 默认 0：右边没有更高温的日子 */
    int* stack = (int*)malloc(n * sizeof(int));  /* 存"还没等到更高温"的下标 */
    int top = 0;                                 /* top == 0 表示空栈 */
    for (int i = 0; i < n; i++) {
        while (top > 0 && temperatures[stack[top - 1]] < temperatures[i]) {
            int k = stack[--top];                /* 第 k 天等到了第 i 天 */
            ans[k] = i - k;
        }
        stack[top++] = i;                        /* 自己入栈，继续等 */
    }
    free(stack);
    *returnSize = n;
    return ans;
}

int main(void) {
    /* ACM 输入：第一行 n，第二行 n 个整数 */
    int n;
    if (scanf("%d", &n) != 1) {
        return 0;
    }
    int* temperatures = (int*)malloc(n * sizeof(int));
    for (int i = 0; i < n; i++) {
        scanf("%d", &temperatures[i]);
    }
    int size = 0;
    int* ans = dailyTemperatures(temperatures, n, &size);
    for (int i = 0; i < size; i++) {
        if (i) printf(" ");
        printf("%d", ans[i]);
    }
    printf("\n");
    free(ans);
    free(temperatures);
    return 0;
}
```
