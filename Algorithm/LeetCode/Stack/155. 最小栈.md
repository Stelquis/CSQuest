# [155. 最小栈](https://leetcode.cn/problems/min-stack/)

- 🟡 **难度**：中等
- 📚 **分类**：栈
- 🏷️ **标签**：`栈`、`设计`

## 题目描述

设计一个支持 `push` ，`pop` ，`top` 操作，并能在常数时间内检索到最小元素的栈。

实现 `MinStack` 类:
- `MinStack()` 初始化堆栈对象。
- `void push(int value)` 将元素 `value` 推入堆栈。
- `void pop()` 删除堆栈顶部的元素。
- `int top()` 获取堆栈顶部的元素。
- `int getMin()` 获取堆栈中的最小元素。

**示例 1:**

```
输入：
["MinStack","push","push","push","getMin","pop","top","getMin"]
[[],[-2],[0],[-3],[],[],[],[]]

输出：
[null,null,null,null,-3,null,0,-2]

解释：
MinStack minStack = new MinStack();
minStack.push(-2);
minStack.push(0);
minStack.push(-3);
minStack.getMin();   --> 返回 -3.
minStack.pop();
minStack.top();      --> 返回 0.
minStack.getMin();   --> 返回 -2.
```

**提示：**
- `-2^31 <= val <= 2^31 - 1`
- `pop`、`top` 和 `getMin` 操作总是在 **非空栈** 上调用
- `push`, `pop`, `top`, and `getMin`最多被调用 `3 * 10^4` 次

## 题解：双栈同步（数据栈 + 最小栈）

要求所有操作 `O(1)`，难点在 `getMin`：不能每次去扫一遍栈。做法是**再开一个"最小栈"**，让它和数据栈**同高、同步**：

- `data`：正常压入/弹出元素；
- `mins`：每次 `push` 时压入"**到目前为止的最小值**"（即 `min(mins 栈顶, 新元素)`，栈为空时就是新元素本身）。

这样 `mins` 的栈顶**永远是当前整个栈的最小值**，`getMin` 直接取栈顶即可，无需遍历。

为什么两者同步弹出就够？因为 `mins` 里每个位置记录的是"数据栈压到该高度时的历史最小值"；弹出一个元素，只要 `mins` 也弹一个，栈顶自然回退到"上一高度的最小值"，完全一致。

- 各操作时间复杂度：`O(1)`
- 空间复杂度：`O(n)`，额外一个等高的最小值栈

**易错点**：`data` / `mins` / `size` 这些**必须是实例字段，不能加 `static`**——`static` 属于类、被所有实例共享，一旦 `new` 出第二个 `MinStack`，构造函数会把共享数组和计数重置，前一个实例的数据就被抹掉（典型现象：`getMin` 读到下标 `-1` 抛异常）。力扣只造一个实例时加 `static` 也能过，但那是错的。

### ☕ Java

```java
import java.util.ArrayDeque;
import java.util.Deque;
import java.util.Scanner;

class MinStack {
    private final Deque<Integer> data = new ArrayDeque<>();   // 数据栈
    private final Deque<Integer> mins = new ArrayDeque<>();   // 与数据栈同步的最小值栈

    public MinStack() {
    }

    public void push(int val) {
        data.push(val);
        // 当前最小值 = min(旧最小值, 新元素)；栈空时即新元素
        mins.push(mins.isEmpty() ? val : Math.min(mins.peek(), val));
    }

    public void pop() {
        data.pop();
        mins.pop();                      // 两栈同步弹出
    }

    public int top() {
        return data.peek();
    }

    public int getMin() {
        return mins.peek();              // 最小值栈顶即全局最小值
    }
}

public class Solution {
    public static void main(String[] args) {
        // ACM 输入：第一行操作数 n，接下来 n 行每行为
        //   push x / pop / top / getMin
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        MinStack st = new MinStack();
        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < n; i++) {
            String op = sc.next();
            if (op.equals("push")) {
                st.push(sc.nextInt());
            } else if (op.equals("pop")) {
                st.pop();
            } else if (op.equals("top")) {
                sb.append(st.top()).append('\n');
            } else {                     // getMin
                sb.append(st.getMin()).append('\n');
            }
        }
        System.out.print(sb);
        sc.close();
    }
}
```

### 🐍 Python

```python
import sys


class MinStack:
    def __init__(self):
        self.data = []          # 数据栈
        self.mins = []          # 与数据栈同步的最小值栈

    def push(self, val: int) -> None:
        self.data.append(val)
        # 当前最小值 = min(旧最小值, 新元素)；栈空时即新元素
        self.mins.append(val if not self.mins else min(self.mins[-1], val))

    def pop(self) -> None:
        self.data.pop()
        self.mins.pop()         # 两栈同步弹出

    def top(self) -> int:
        return self.data[-1]

    def getMin(self) -> int:
        return self.mins[-1]    # 最小值栈顶即全局最小值


def main():
    # ACM 输入：第一行操作数 n，接下来 n 行每行为
    #   push x / pop / top / getMin
    data = sys.stdin.read().split()
    n = int(data[0])
    idx = 1
    st = MinStack()
    out = []
    for _ in range(n):
        op = data[idx]
        idx += 1
        if op == "push":
            st.push(int(data[idx]))
            idx += 1
        elif op == "pop":
            st.pop()
        elif op == "top":
            out.append(str(st.top()))
        else:                   # getMin
            out.append(str(st.getMin()))
    print("\n".join(out))


if __name__ == "__main__":
    main()
```

### ➕ C++

```cpp
#include <algorithm>
#include <stack>
#include <string>
#include <iostream>
using namespace std;

class MinStack {
    stack<int> data;                     // 数据栈
    stack<int> mins;                     // 与数据栈同步的最小值栈
public:
    MinStack() {
    }

    void push(int val) {
        data.push(val);
        // 当前最小值 = min(旧最小值, 新元素)；栈空时即新元素
        mins.push(mins.empty() ? val : min(mins.top(), val));
    }

    void pop() {
        data.pop();
        mins.pop();                      // 两栈同步弹出
    }

    int top() {
        return data.top();
    }

    int getMin() {
        return mins.top();               // 最小值栈顶即全局最小值
    }
};

int main() {
    // ACM 输入：第一行操作数 n，接下来 n 行每行为
    //   push x / pop / top / getMin
    int n;
    cin >> n;
    MinStack st;
    for (int i = 0; i < n; i++) {
        string op;
        cin >> op;
        if (op == "push") {
            int x;
            cin >> x;
            st.push(x);
        } else if (op == "pop") {
            st.pop();
        } else if (op == "top") {
            cout << st.top() << "\n";
        } else {
            cout << st.getMin() << "\n";
        }
    }
    return 0;
}
```

### ⚙️ C

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct {
    int *data;      /* 数据栈 */
    int *mins;      /* 与数据栈同步的最小值栈 */
    int size;       /* 当前元素个数 */
    int cap;        /* 容量 */
} MinStack;

static void ensureCapacity(MinStack *s) {
    if (s->size < s->cap) {
        return;
    }
    s->cap *= 2;
    s->data = (int *)realloc(s->data, s->cap * sizeof(int));
    s->mins = (int *)realloc(s->mins, s->cap * sizeof(int));
}

MinStack *minStackCreate(void) {
    MinStack *s = (MinStack *)malloc(sizeof(MinStack));
    s->cap = 16;
    s->size = 0;
    s->data = (int *)malloc(s->cap * sizeof(int));
    s->mins = (int *)malloc(s->cap * sizeof(int));
    return s;
}

void minStackPush(MinStack *s, int val) {
    ensureCapacity(s);
    s->data[s->size] = val;
    /* 当前最小值 = min(旧最小值, 新元素)；栈空时即新元素 */
    s->mins[s->size] = (s->size == 0 || val < s->mins[s->size - 1])
                           ? val
                           : s->mins[s->size - 1];
    s->size++;
}

void minStackPop(MinStack *s) {
    if (s->size > 0) {
        s->size--;                       /* 两栈同步弹出 */
    }
}

int minStackTop(MinStack *s) {
    return s->data[s->size - 1];
}

int minStackGetMin(MinStack *s) {
    return s->mins[s->size - 1];         /* 最小值栈顶即全局最小值 */
}

void minStackFree(MinStack *s) {
    free(s->data);
    free(s->mins);
    free(s);
}

int main(void) {
    /* ACM 输入：第一行操作数 n，接下来 n 行每行为
       push x / pop / top / getMin */
    int n;
    if (scanf("%d", &n) != 1) {
        return 0;
    }
    MinStack *st = minStackCreate();
    char op[16];
    for (int i = 0; i < n; i++) {
        scanf("%s", op);
        if (strcmp(op, "push") == 0) {
            int x;
            scanf("%d", &x);
            minStackPush(st, x);
        } else if (strcmp(op, "pop") == 0) {
            minStackPop(st);
        } else if (strcmp(op, "top") == 0) {
            printf("%d\n", minStackTop(st));
        } else {
            printf("%d\n", minStackGetMin(st));
        }
    }
    minStackFree(st);
    return 0;
}
```
