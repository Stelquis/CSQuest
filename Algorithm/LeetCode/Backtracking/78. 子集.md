# [78. 子集](https://leetcode.cn/problems/subsets/)

- 🟡 **难度**：中等
- ↩️ **分类**：回溯
- 🏷️ **标签**：`位运算`、`数组`、`回溯`

## 题目描述

给你一个整数数组 `nums` ，数组中的元素 **互不相同** 。返回该数组所有可能的子集（幂集）。

解集 **不能** 包含重复的子集。你可以按 **任意顺序** 返回解集。

**示例 1：**

```
输入：nums = [1,2,3]
输出：[[],[1],[2],[1,2],[3],[1,3],[2,3],[1,2,3]]
```

**示例 2：**

```
输入：nums = [0]
输出：[[],[0]]
```

**提示：**
- `1 <= nums.length <= 10`
- `-10 <= nums[i] <= 10`
- `nums` 中的所有元素 **互不相同**

## 题解：回溯（每个元素"选 / 不选"）

子集的决策树比全排列更简单：**按下标逐个走，每个元素只有"不选 / 选"两种分支**，走完 `n` 个元素到达叶子，路径就是一个子集。

- `2^n` 个叶子 = `2^n` 个子集，**每个恰好出现一次**，天然不重复——因为决策按"下标"进行（第 `count` 个元素选不选），与元素值无关，同一个子集只有唯一一条决策路径；
- 与 46 全排列的区别：全排列只在 `path` 装满时收集，而这里**每个叶子都是答案**，空集 `[]` 也算（什么都没选的叶子）；
- "不选"分支**不需要撤销**——它根本没改 `arr`；只有"选"分支需要 `remove` 撤销。

决策树（`nums = [1,2,3]`，左 = 不选，右 = 选）：

```
                count=0
           不选 /      \ 选1
          count=1      count=1
         不选/ \选2   不选/ \选2
          ...      ...  ...
叶子:  [] [3] [2] [2,3] [1] [1,3] [1,2] [1,2,3]   ← 2^3 = 8 个
```

- 时间复杂度：`O(n · 2^n)`，共 `2^n` 个子集，每个收集时拷贝 `O(n)`
- 空间复杂度：`O(n)`，递归深度与 `arr`（不计返回答案）

**备注**：子集还有两种等价写法——① **位运算枚举**：`mask ∈ [0, 2^n)`，第 `i` 位为 1 表示选 `nums[i]`，最直白；② **start 起点式回溯**：每进入一层就把当前 `path` 收集一次，循环从 `start` 开始向后选，就是 46 模板改成"每层收集"。三种写法结果集相同。

**易错点**：`ans` 等状态字段**必须是实例字段，不能加 `static`**（同 46 / 155 的坑），本题靠"每次 `subsets` 重建 `ans`"掩盖了现象，写法上仍应正确。

### ☕ Java

```java
import java.util.ArrayList;
import java.util.List;
import java.util.Scanner;

class Solution {
    public List<List<Integer>> ans;

    public List<List<Integer>> subsets(int[] nums) {
        ans = new ArrayList<>();
        List<Integer> arr = new ArrayList<>();
        backTrack(ans, arr, nums, 0);
        return ans;
    }

    public static void backTrack(List<List<Integer>> ans, List<Integer> arr,
                                 int[] nums, int count) {
        if (count == nums.length) {
            ans.add(new ArrayList<>(arr));      // 每个叶子都是一个子集，拍快照
            return;
        }
        backTrack(ans, arr, nums, count + 1);   // 不选 nums[count]（无需撤销）
        arr.add(nums[count]);                   // 选 nums[count]
        backTrack(ans, arr, nums, count + 1);
        arr.remove(arr.size() - 1);             // 撤销选择
    }

    public static void main(String[] args) {
        // ACM 输入：第一行 n，第二行 n 个整数
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] nums = new int[n];
        for (int i = 0; i < n; i++) {
            nums[i] = sc.nextInt();
        }
        List<List<Integer>> ans = new Solution().subsets(nums);
        StringBuilder sb = new StringBuilder();
        for (List<Integer> subset : ans) {
            sb.append(subset).append('\n');
        }
        System.out.print(sb);
        sc.close();
    }
}
```

### 🐍 Python

```python
import sys
from typing import List


class Solution:
    def subsets(self, nums: List[int]) -> List[List[int]]:
        ans = []
        arr = []

        def back_track(count: int):
            if count == len(nums):
                ans.append(arr[:])          # 每个叶子都是一个子集，拍快照
                return
            back_track(count + 1)           # 不选 nums[count]（无需撤销）
            arr.append(nums[count])         # 选 nums[count]
            back_track(count + 1)
            arr.pop()                       # 撤销选择

        back_track(0)
        return ans


if __name__ == "__main__":
    # ACM 输入：第一行 n，第二行 n 个整数
    data = sys.stdin.read().split()
    n = int(data[0])
    nums = [int(x) for x in data[1:1 + n]]
    for subset in Solution().subsets(nums):
        print(subset)
```

### ➕ C++

```cpp
#include <vector>
#include <iostream>
using namespace std;

class Solution {
public:
    vector<vector<int>> ans;

    vector<vector<int>> subsets(vector<int>& nums) {
        ans.clear();
        vector<int> arr;
        backTrack(ans, arr, nums, 0);
        return ans;
    }

    static void backTrack(vector<vector<int>>& ans, vector<int>& arr,
                          vector<int>& nums, int count) {
        if (count == (int)nums.size()) {
            ans.push_back(arr);                 // 每个叶子都是一个子集，push_back 自带快照
            return;
        }
        backTrack(ans, arr, nums, count + 1);   // 不选 nums[count]（无需撤销）
        arr.push_back(nums[count]);             // 选 nums[count]
        backTrack(ans, arr, nums, count + 1);
        arr.pop_back();                         // 撤销选择
    }
};

int main() {
    // ACM 输入：第一行 n，第二行 n 个整数
    int n;
    cin >> n;
    vector<int> nums(n);
    for (int i = 0; i < n; i++) {
        cin >> nums[i];
    }
    vector<vector<int>> ans = Solution().subsets(nums);
    for (auto& subset : ans) {
        cout << "[";
        for (int i = 0; i < (int)subset.size(); i++) {
            if (i > 0) cout << ", ";
            cout << subset[i];
        }
        cout << "]\n";
    }
    return 0;
}
```

### ⚙️ C

```c
#include <stdio.h>
#include <stdlib.h>

static int **ans;
static int ansSize;
static int *ansColSizes;

static void backTrack(int *nums, int n, int *arr, int depth, int count) {
    if (count == n) {
        ans[ansSize] = (int *)malloc(depth * sizeof(int));
        for (int i = 0; i < depth; i++) {
            ans[ansSize][i] = arr[i];           /* 拍快照 */
        }
        ansColSizes[ansSize] = depth;
        ansSize++;
        return;
    }
    backTrack(nums, n, arr, depth, count + 1);          /* 不选 nums[count] */
    arr[depth] = nums[count];                           /* 选 nums[count] */
    backTrack(nums, n, arr, depth + 1, count + 1);
    /* 撤销选择：arr[depth] 会被覆盖，无需清 */
}

int **subsets(int *nums, int numsSize, int *returnSize, int **returnColumnSizes) {
    int total = 1 << numsSize;                          /* 2^n 个子集，按上界预分配 */
    ans = (int **)malloc(total * sizeof(int *));
    ansColSizes = (int *)malloc(total * sizeof(int));
    ansSize = 0;
    int *arr = (int *)malloc(numsSize * sizeof(int));
    backTrack(nums, numsSize, arr, 0, 0);
    free(arr);
    *returnSize = ansSize;
    *returnColumnSizes = ansColSizes;
    return ans;
}

int main(void) {
    /* ACM 输入：第一行 n，第二行 n 个整数 */
    int n;
    if (scanf("%d", &n) != 1) {
        return 0;
    }
    int *nums = (int *)malloc(n * sizeof(int));
    for (int i = 0; i < n; i++) {
        scanf("%d", &nums[i]);
    }
    int returnSize;
    int *colSizes;
    int **res = subsets(nums, n, &returnSize, &colSizes);
    for (int i = 0; i < returnSize; i++) {
        printf("[");
        for (int j = 0; j < colSizes[i]; j++) {
            if (j > 0) {
                printf(", ");
            }
            printf("%d", res[i][j]);
        }
        printf("]\n");
        free(res[i]);
    }
    free(res);
    free(colSizes);
    free(nums);
    return 0;
}
```
