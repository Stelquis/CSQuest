# [17. 电话号码的字母组合](https://leetcode.cn/problems/letter-combinations-of-a-phone-number/)

- 🟡 **难度**：中等
- ↩️ **分类**：回溯
- 🏷️ **标签**：`哈希表`、`字符串`、`回溯`

## 题目描述

给定一个仅包含数字 `2-9` 的字符串，返回所有它能表示的字母组合。答案可以按 **任意顺序** 返回。

给出数字到字母的映射如下（与电话按键相同）。注意 1 不对应任何字母。

![](https://pic.leetcode.cn/1752723054-mfIHZs-image.png)

**示例 1：**

```
输入：digits = "23"
输出：["ad","ae","af","bd","be","bf","cd","ce","cf"]
```

**示例 2：**

```
输入：digits = "2"
输出：["a","b","c"]
```

**提示：**
- `1 <= digits.length <= 4`
- `digits[i]` 是范围 `['2', '9']` 的一个数字。

## 题解：回溯（逐位展开 + 映射表）

和 46 全排列、78 子集是同一套框架，变的只是"选择列表"：**第 `pos` 层的选择不是数组元素，而是数字 `digits[pos]` 对应的 3~4 个字母**。树高等于 `digits` 长度，每个叶子是一个完整组合。

- **映射表**：`word[digits.charAt(pos) - '2']` 把 `'2'-'9'` 直接下标映射到字母组（7 和 9 对应 4 个字母，其余 3 个）；
- **每层循环**：遍历该字母组 → `append` 到路径 → 递归下一层 → `deleteCharAt` 撤销；
- **收集快照**：`path.toString()`——`String` 不可变，天然就是快照，不像 `List` 那样需要手动拷贝；
- **空串特判**：`digits` 为空时树里没有任何叶子，若不特判会递归到"0 位即终止"返回 `[""]`，而期望答案是 `[]`。

决策树（`digits = "23"`）：

```
        ""                 ← pos=0，选 2 的字母
    /   |   \
   a    b    c             ← pos=1，选 3 的字母
 / | \ ...
ad ae af bd be bf cd ce cf   ← 2 × 3 = 6 个叶子
```

- 时间复杂度：`O(n · 3^n · 4^m)`，其中 `n` 是对应 3 字母的数字个数、`m` 对应 4 字母的（7/9）个数；每个组合收集时拷贝 `O(n)`
- 空间复杂度：`O(n)`，递归深度与路径（不计返回答案）

**对比 46 / 78**：三题框架完全一致——"选择列表 → 做选择 → 递归 → 撤销选择"。差别只在：46 每层选"没用过的数"、78 每层做"选/不选"决策、17 每层选"映射字母"。

### ☕ Java

```java
import java.util.ArrayList;
import java.util.List;
import java.util.Scanner;

class Solution {
    public char[][] word = new char[][]{
        {'a', 'b', 'c'},          // 2
        {'d', 'e', 'f'},          // 3
        {'g', 'h', 'i'},          // 4
        {'j', 'k', 'l'},          // 5
        {'m', 'n', 'o'},          // 6
        {'p', 'q', 'r', 's'},     // 7
        {'t', 'u', 'v'},          // 8
        {'w', 'x', 'y', 'z'}      // 9
    };

    public List<String> ans = new ArrayList<>();

    public List<String> letterCombinations(String digits) {
        ans = new ArrayList<>();
        if (digits.isEmpty()) {
            return ans;             // 空串没有组合（否则会返回 [""]）
        }
        backTrack(word, ans, digits, new StringBuilder(), 0);
        return ans;
    }

    public static void backTrack(char[][] word, List<String> ans, String digits,
                                 StringBuilder path, int pos) {
        if (pos == digits.length()) {
            ans.add(path.toString());   // String 不可变，天然快照
            return;
        }
        char[] letters = word[digits.charAt(pos) - '2'];
        for (char c : letters) {
            path.append(c);             // 做选择
            backTrack(word, ans, digits, path, pos + 1);
            path.deleteCharAt(path.length() - 1);   // 撤销选择
        }
    }

    public static void main(String[] args) {
        // ACM 输入：一行数字串 digits
        Scanner sc = new Scanner(System.in);
        String digits = sc.hasNext() ? sc.next() : "";
        List<String> ans = new Solution().letterCombinations(digits);
        StringBuilder sb = new StringBuilder();
        for (String s : ans) {
            sb.append(s).append('\n');
        }
        System.out.print(sb);
        sc.close();
    }
}
```

### 🐍 Python

```python
import sys
from typing import List

WORD = {
    "2": "abc", "3": "def", "4": "ghi", "5": "jkl",
    "6": "mno", "7": "pqrs", "8": "tuv", "9": "wxyz",
}


class Solution:
    def letterCombinations(self, digits: str) -> List[str]:
        ans: List[str] = []
        if not digits:
            return ans                 # 空串没有组合

        def back_track(path: str, pos: int):
            if pos == len(digits):
                ans.append(path)       # str 不可变，天然快照
                return
            for c in WORD[digits[pos]]:
                back_track(path + c, pos + 1)   # str 拼接生成新串，无需显式撤销

        back_track("", 0)
        return ans


if __name__ == "__main__":
    # ACM 输入：一行数字串 digits
    digits = sys.stdin.readline().strip()
    for s in Solution().letterCombinations(digits):
        print(s)
```

### ➕ C++

```cpp
#include <string>
#include <vector>
#include <iostream>
using namespace std;

class Solution {
public:
    vector<string> letterCombinations(string digits) {
        vector<string> ans;
        if (digits.empty()) {
            return ans;                 // 空串没有组合
        }
        static const vector<string> word = {
            "abc", "def", "ghi", "jkl", "mno", "pqrs", "tuv", "wxyz"
        };
        string path;
        backTrack(word, ans, digits, path, 0);
        return ans;
    }

    static void backTrack(const vector<string>& word, vector<string>& ans,
                          const string& digits, string& path, int pos) {
        if (pos == (int)digits.size()) {
            ans.push_back(path);        // push_back 自带快照
            return;
        }
        const string& letters = word[digits[pos] - '2'];
        for (char c : letters) {
            path.push_back(c);          // 做选择
            backTrack(word, ans, digits, path, pos + 1);
            path.pop_back();            // 撤销选择
        }
    }
};

int main() {
    // ACM 输入：一行数字串 digits
    string digits;
    cin >> digits;
    vector<string> ans = Solution().letterCombinations(digits);
    for (const string& s : ans) {
        cout << s << "\n";
    }
    return 0;
}
```

### ⚙️ C

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

static const char *WORD[8] = {
    "abc", "def", "ghi", "jkl", "mno", "pqrs", "tuv", "wxyz"
};

static void backTrack(const char *digits, int pos, char *path, int depth,
                      char **ans, int *ansSize) {
    if (digits[pos] == '\0') {
        path[depth] = '\0';
        ans[*ansSize] = (char *)malloc(depth + 1);
        strcpy(ans[*ansSize], path);            /* 拍快照 */
        (*ansSize)++;
        return;
    }
    const char *letters = WORD[digits[pos] - '2'];
    for (int i = 0; letters[i] != '\0'; i++) {
        path[depth] = letters[i];               /* 做选择 */
        backTrack(digits, pos + 1, path, depth + 1, ans, ansSize);
        /* 撤销选择：path[depth] 会被覆盖，无需清 */
    }
}

char **letterCombinations(char *digits, int *returnSize) {
    int len = (int)strlen(digits);
    *returnSize = 0;
    if (len == 0) {
        return NULL;
    }
    int total = 1;                              /* 按组合数上界预分配 */
    for (int i = 0; i < len; i++) {
        total *= (digits[i] == '7' || digits[i] == '9') ? 4 : 3;
    }
    char **ans = (char **)malloc(total * sizeof(char *));
    char *path = (char *)malloc(len + 1);
    backTrack(digits, 0, path, 0, ans, returnSize);
    free(path);
    return ans;
}

int main(void) {
    /* ACM 输入：一行数字串 digits */
    char digits[8];
    if (scanf("%7s", digits) != 1) {
        return 0;
    }
    int returnSize;
    char **res = letterCombinations(digits, &returnSize);
    for (int i = 0; i < returnSize; i++) {
        printf("%s\n", res[i]);
        free(res[i]);
    }
    if (res != NULL) {
        free(res);
    }
    return 0;
}
```
