# [39. 组合总和](https://leetcode.cn/problems/combination-sum/)

- 🟡 **难度**：中等
- ↩️ **分类**：回溯
- 🏷️ **标签**：`数组`、`回溯`

## 题目描述

给你一个 **无重复元素** 的整数数组 `candidates` 和一个目标整数 `target` ，找出 `candidates` 中可以使数字和为目标数 `target` 的 所有 **不同组合** ，并以列表形式返回。你可以按 **任意顺序** 返回这些组合。

`candidates` 中的 **同一个** 数字可以 **无限制重复被选取** 。如果至少一个数字的被选数量不同，则两种组合是不同的。

对于给定的输入，保证和为 `target` 的不同组合数少于 `150` 个。

**示例 1：**

```
输入：candidates = [2,3,6,7], target = 7
输出：[[2,2,3],[7]]
解释：
2 和 3 可以形成一组候选，2 + 2 + 3 = 7 。注意 2 可以使用多次。
7 也是一个候选， 7 = 7 。
仅有这两种组合。
```

**示例 2：**

```
输入: candidates = [2,3,5], target = 8
输出: [[2,2,2,2],[2,3,3],[3,5]]
```

**示例 3：**

```
输入: candidates = [2], target = 1
输出: []
```

**提示：**
- `1 <= candidates.length <= 30`
- `2 <= candidates[i] <= 40`
- `candidates` 的所有元素 **互不相同**
- `1 <= target <= 40`

## 题解：回溯（start 起点允许重复选）

和 46 全排列同框架，但有两个关键不同：

1. **允许重复选**：递归时传 `i` 而不是 `i + 1`，下一层还能再选同一个数（"2 可以使用多次"）；
2. **组合去重靠起点 `x`**：每层循环只从 `x` 向后选，保证组合内的下标严格不降——`[2,3,2]`、`[3,2,2]` 这类"同一组数字换顺序"的重复路径从一开始就走不出来，所以**不需要 used 数组、也不需要结果去重**。

**对比记忆**：46 传 `i + 1`（排列，元素只用一次，顺序算不同答案）；39 传 `i`（组合，可重复用，顺序不算）。这一字之差就是"排列"和"组合"的分界。

**剪枝**：所有 `candidates[i] >= 2 > 0`，`sum` 只会越加越大——`sum > t` 时整棵子树都不可能有答案，直接返回；`sum == t` 收集快照后返回。

搜索过程（`candidates = [2,3,6,7]`、`target = 7`）：

```
2 → 2,2 → 2,2,2 → 2,2,2,2 (8>7 剪) ✗
                ↘ 2,2,3 = 7 ✓ 收集 [2,2,3]
    ↘ 2,3 → 2,3,3 (8>7 剪) ✗
    ↘ 2,6 (8>7 剪) ✗  2,7 (9>7 剪) ✗
3 → 3,3 → 3,3,3 (9>7 剪) ✗
  ↘ 3,6 (9>7 剪) ✗
6 → 6,6 (12>7 剪) ✗
7 = 7 ✓ 收集 [7]
```

- 时间复杂度：指数级（递归树规模与答案数成正比，题目保证答案少于 150 个）
- 空间复杂度：`O(target / min(candidates))`，递归深度上限

**易错点**：`ans` 等状态字段**必须是实例字段，不能加 `static`**（同 46 / 78 的坑）。

### ☕ Java

```java
import java.util.ArrayList;
import java.util.List;
import java.util.Scanner;

class Solution {
    public List<List<Integer>> ans;

    public List<List<Integer>> combinationSum(int[] candidates, int target) {
        ans = new ArrayList<>();
        List<Integer> arr = new ArrayList<>();
        backTrack(ans, arr, candidates, target, 0, 0);
        return ans;
    }

    public static void backTrack(List<List<Integer>> ans, List<Integer> arr,
                                 int[] nums, int t, int sum, int x) {
        if (sum == t) {
            ans.add(new ArrayList<>(arr));      // 拍快照
            return;
        } else if (sum > t) {
            return;                             // 剪枝：和只会越加越大
        }
        for (int i = x; i < nums.length; i++) {
            arr.add(nums[i]);                   // 做选择
            backTrack(ans, arr, nums, t, sum + nums[i], i);   // 传 i：可重复选
            arr.remove(arr.size() - 1);         // 撤销选择
        }
    }

    public static void main(String[] args) {
        // ACM 输入：第一行 n 和 target，第二行 n 个整数
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int target = sc.nextInt();
        int[] nums = new int[n];
        for (int i = 0; i < n; i++) {
            nums[i] = sc.nextInt();
        }
        List<List<Integer>> ans = new Solution().combinationSum(nums, target);
        StringBuilder sb = new StringBuilder();
        for (List<Integer> combo : ans) {
            sb.append(combo).append('\n');
        }
        System.out.print(sb);
        sc.close();
    }
}
```

### 🐍 Python

```python
import sys
from typing import List


class Solution:
    def combinationSum(self, candidates: List[int], target: int) -> List[List[int]]:
        ans = []
        arr = []

        def back_track(total: int, start: int):
            if total == target:
                ans.append(arr[:])              # 拍快照
                return
            if total > target:
                return                          # 剪枝
            for i in range(start, len(candidates)):
                arr.append(candidates[i])       # 做选择
                back_track(total + candidates[i], i)   # 传 i：可重复选
                arr.pop()                       # 撤销选择

        back_track(0, 0)
        return ans


if __name__ == "__main__":
    # ACM 输入：第一行 n 和 target，第二行 n 个整数
    data = sys.stdin.read().split()
    n, target = int(data[0]), int(data[1])
    nums = [int(x) for x in data[2:2 + n]]
    for combo in Solution().combinationSum(nums, target):
        print(combo)
```

### ➕ C++

```cpp
#include <vector>
#include <iostream>
using namespace std;

class Solution {
public:
    vector<vector<int>> ans;

    vector<vector<int>> combinationSum(vector<int>& candidates, int target) {
        ans.clear();
        vector<int> arr;
        backTrack(ans, arr, candidates, target, 0, 0);
        return ans;
    }

    static void backTrack(vector<vector<int>>& ans, vector<int>& arr,
                          vector<int>& nums, int t, int sum, int x) {
        if (sum == t) {
            ans.push_back(arr);                 // push_back 自带快照
            return;
        }
        if (sum > t) {
            return;                             // 剪枝
        }
        for (int i = x; i < (int)nums.size(); i++) {
            arr.push_back(nums[i]);             // 做选择
            backTrack(ans, arr, nums, t, sum + nums[i], i);   // 传 i：可重复选
            arr.pop_back();                     // 撤销选择
        }
    }
};

int main() {
    // ACM 输入：第一行 n 和 target，第二行 n 个整数
    int n, target;
    cin >> n >> target;
    vector<int> nums(n);
    for (int i = 0; i < n; i++) {
        cin >> nums[i];
    }
    vector<vector<int>> ans = Solution().combinationSum(nums, target);
    for (auto& combo : ans) {
        cout << "[";
        for (int i = 0; i < (int)combo.size(); i++) {
            if (i > 0) cout << ", ";
            cout << combo[i];
        }
        cout << "]\n";
    }
    return 0;
}
```

### ⚙️ C

```c
#include <stdio.h>
#include <stdlib.h>

static int **ans;
static int ansSize;
static int ansCap;
static int *ansColSizes;

static void addCombo(int *arr, int depth) {
    if (ansSize == ansCap) {                    /* 组合数未知，动态扩容 */
        ansCap = ansCap ? ansCap * 2 : 16;
        ans = (int **)realloc(ans, ansCap * sizeof(int *));
        ansColSizes = (int *)realloc(ansColSizes, ansCap * sizeof(int));
    }
    ans[ansSize] = (int *)malloc(depth * sizeof(int));
    for (int i = 0; i < depth; i++) {
        ans[ansSize][i] = arr[i];               /* 拍快照 */
    }
    ansColSizes[ansSize] = depth;
    ansSize++;
}

static void backTrack(int *nums, int n, int t, int sum, int x, int *arr, int depth) {
    if (sum == t) {
        addCombo(arr, depth);
        return;
    }
    if (sum > t) {
        return;                                 /* 剪枝 */
    }
    for (int i = x; i < n; i++) {
        arr[depth] = nums[i];                   /* 做选择 */
        backTrack(nums, n, t, sum + nums[i], i, arr, depth + 1);   /* 传 i：可重复选 */
        /* 撤销选择：arr[depth] 会被覆盖 */
    }
}

int **combinationSum(int *candidates, int candidatesSize, int target,
                     int *returnSize, int **returnColumnSizes) {
    ans = NULL;
    ansColSizes = NULL;
    ansSize = 0;
    ansCap = 0;
    /* candidates[i] >= 2，递归深度不超过 target / 2 < target */
    int *arr = (int *)malloc(target * sizeof(int));
    backTrack(candidates, candidatesSize, target, 0, 0, arr, 0);
    free(arr);
    *returnSize = ansSize;
    *returnColumnSizes = ansColSizes;
    return ans;
}

int main(void) {
    /* ACM 输入：第一行 n 和 target，第二行 n 个整数 */
    int n, target;
    if (scanf("%d %d", &n, &target) != 2) {
        return 0;
    }
    int *nums = (int *)malloc(n * sizeof(int));
    for (int i = 0; i < n; i++) {
        scanf("%d", &nums[i]);
    }
    int returnSize;
    int *colSizes;
    int **res = combinationSum(nums, n, target, &returnSize, &colSizes);
    for (int i = 0; i < returnSize; i++) {
        printf("[");
        for (int j = 0; j < colSizes[i]; j++) {
            if (j > 0) {
                printf(", ");
            }
            printf("%d", res[i][j]);
        }
        printf("]\n");
        free(res[i]);
    }
    if (res != NULL) {
        free(res);
        free(colSizes);
    }
    free(nums);
    return 0;
}
```
