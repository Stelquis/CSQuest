# [226. 翻转二叉树](https://leetcode.cn/problems/invert-binary-tree/)

- 🟢 **难度**：简单
- 🌳 **分类**：二叉树
- 🏷️ **标签**：`树`、`深度优先搜索`、`广度优先搜索`、`二叉树`

## 题目描述

给你一棵二叉树的根节点 `root` ，翻转这棵二叉树，并返回其根节点。

**示例 1：**

![](https://assets.leetcode.com/uploads/2021/03/14/invert1-tree.jpg)

```
输入：root = [4,2,7,1,3,6,9]
输出：[4,7,2,9,6,3,1]
```

**示例 2：**

![](https://assets.leetcode.com/uploads/2021/03/14/invert2-tree.jpg)

```
输入：root = [2,1,3]
输出：[2,3,1]
```

**示例 3：**

```
输入：root = []
输出：[]
```

**提示：**
- 树中节点数目范围在 `[0, 100]` 内
- `-100 <= Node.val <= 100`

## 题解一：深度优先搜索（前序递归）

翻转的单位是「每个节点交换自己的左右子树指针」，该动作不依赖子树状态，因此自顶向下即可：先交换当前节点的左右孩子，再把同样的任务交给新的左右子树递归执行。空树是递归出口。

- 时间复杂度：`O(n)`，每个节点恰好访问一次
- 空间复杂度：`O(h)`，递归栈深度为树高，最坏（链状树）为 `O(n)`

### ☕ Java

```java
import java.util.LinkedList;
import java.util.Queue;
import java.util.Scanner;

public class Solution {
    public TreeNode invertTree(TreeNode root) {
        if (root == null) {
            return root;
        }
        // 先交换自己的左右子树
        TreeNode l = root.left;
        root.left = root.right;
        root.right = l;
        // 再把同样的任务交给新的左右子树
        invertTree(root.left);
        invertTree(root.right);
        return root;
    }

    public static void main(String[] args) {
        // ACM 输入：一行层序序列，null 表示空节点
        Scanner sc = new Scanner(System.in);
        TreeNode root = buildTree(sc);
        new Solution().invertTree(root);
        System.out.println(levelOrder(root));
        sc.close();
    }

    // 按层序遍历读入建树，null 表示空节点
    public static TreeNode buildTree(Scanner sc) {
        if (!sc.hasNext()) {
            return null;
        }
        String token = sc.next();
        if ("null".equals(token)) {
            return null;
        }
        TreeNode root = new TreeNode(Integer.parseInt(token));
        Queue<TreeNode> queue = new LinkedList<>();
        queue.offer(root);
        while (!queue.isEmpty()) {
            TreeNode cur = queue.poll();
            if (!sc.hasNext()) {
                break;
            }
            String left = sc.next();
            if (!"null".equals(left)) {
                cur.left = new TreeNode(Integer.parseInt(left));
                queue.offer(cur.left);
            }
            if (!sc.hasNext()) {
                break;
            }
            String right = sc.next();
            if (!"null".equals(right)) {
                cur.right = new TreeNode(Integer.parseInt(right));
                queue.offer(cur.right);
            }
        }
        return root;
    }

    // 层序遍历输出节点值，如 [4, 7, 2]
    public static String levelOrder(TreeNode root) {
        if (root == null) {
            return "[]";
        }
        Queue<TreeNode> queue = new LinkedList<>();
        queue.offer(root);
        StringBuilder sb = new StringBuilder("[");
        while (!queue.isEmpty()) {
            TreeNode cur = queue.poll();
            sb.append(cur.val);
            if (cur.left != null) {
                queue.offer(cur.left);
            }
            if (cur.right != null) {
                queue.offer(cur.right);
            }
            if (!queue.isEmpty()) {
                sb.append(", ");
            }
        }
        return sb.append("]").toString();
    }
}

class TreeNode {
    int val;
    TreeNode left;
    TreeNode right;

    TreeNode() {}

    TreeNode(int val) { this.val = val; }

    TreeNode(int val, TreeNode left, TreeNode right) {
        this.val = val;
        this.left = left;
        this.right = right;
    }
}
```

```python
import sys
from collections import deque
from typing import Optional


class TreeNode:
    def __init__(self, val=0, left=None, right=None):
        self.val = val
        self.left = left
        self.right = right


class Solution:
    def invertTree(self, root: Optional[TreeNode]) -> Optional[TreeNode]:
        if root is None:
            return root
        # 先交换自己的左右子树
        root.left, root.right = root.right, root.left
        # 再把同样的任务交给新的左右子树
        self.invertTree(root.left)
        self.invertTree(root.right)
        return root


def build_tree(tokens, pos):
    """按层序遍历建树，'null' 表示空节点"""
    if pos[0] >= len(tokens) or tokens[pos[0]] == "null":
        pos[0] += 1
        return None
    root = TreeNode(int(tokens[pos[0]]))
    pos[0] += 1
    queue = deque([root])
    while queue:
        cur = queue.popleft()
        for i in range(2):
            if pos[0] >= len(tokens):
                break
            child = None
            if tokens[pos[0]] != "null":
                child = TreeNode(int(tokens[pos[0]]))
                queue.append(child)
            if i == 0:
                cur.left = child
            else:
                cur.right = child
            pos[0] += 1
    return root


def level_order(root):
    """层序遍历输出节点值，如 [4, 7, 2]"""
    if root is None:
        return "[]"
    queue = deque([root])
    vals = []
    while queue:
        cur = queue.popleft()
        vals.append(cur.val)
        if cur.left:
            queue.append(cur.left)
        if cur.right:
            queue.append(cur.right)
    return "[" + ", ".join(map(str, vals)) + "]"


if __name__ == "__main__":
    # ACM 输入：一行层序序列，null 表示空节点
    tokens = sys.stdin.read().split()
    root = build_tree(tokens, [0])
    Solution().invertTree(root)
    print(level_order(root))
```

## 题解二：广度优先搜索（层序交换）

换一个视角：按层遍历，每取出的节点都当场交换其左右孩子，再把孩子入队。所有节点处理完时整棵树即被翻转。

- 时间复杂度：`O(n)`，每个节点入队、出队各一次
- 空间复杂度：`O(w)`，`w` 为树的最大宽度，最坏约为 `O(n/2)`

### ☕ Java

```java
import java.util.LinkedList;
import java.util.Queue;
import java.util.Scanner;

public class Solution {
    public TreeNode invertTree(TreeNode root) {
        if (root == null) {
            return root;
        }
        Queue<TreeNode> queue = new LinkedList<>();
        queue.offer(root);
        while (!queue.isEmpty()) {
            TreeNode cur = queue.poll();
            // 当场交换该节点的左右孩子
            TreeNode l = cur.left;
            cur.left = cur.right;
            cur.right = l;
            if (cur.left != null) {
                queue.offer(cur.left);
            }
            if (cur.right != null) {
                queue.offer(cur.right);
            }
        }
        return root;
    }
```

```c
    struct TreeNode* right;

// 手写队列：存节点指针
typedef struct {
    TreeNode** data;
    int head;
    int tail;
} Queue;

void queueInit(Queue* q, int cap) {
    q->data = (TreeNode**)malloc(cap * sizeof(TreeNode*));
    q->head = q->tail = 0;
}
void queuePush(Queue* q, TreeNode* node) { q->data[q->tail++] = node; }
TreeNode* queuePop(Queue* q) { return q->data[q->head++]; }
int queueEmpty(Queue* q) { return q->head == q->tail; }

struct TreeNode* invertTree(struct TreeNode* root) {
    if (root == NULL) {
        return root;
    }
    Queue queue;
    queueInit(&queue, 200);
    queuePush(&queue, root);
    while (!queueEmpty(&queue)) {
        TreeNode* cur = queuePop(&queue);
        // 当场交换该节点的左右孩子
        struct TreeNode* l = cur->left;
        cur->left = cur->right;
        cur->right = l;
        if (cur->left != NULL) {
            queuePush(&queue, cur->left);
        }
        if (cur->right != NULL) {
            queuePush(&queue, cur->right);
        }

    free(queue.data);
    return root;

    int head = 0, tail = 0;

        TreeNode* cur = queue[head++];
        printf("%d", cur->val);
        if (cur->left != NULL) {

            queue[tail++] = cur->right;
        }
        if (head < tail) {

        }
    }

}
```
