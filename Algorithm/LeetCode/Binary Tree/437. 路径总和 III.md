# [437. 路径总和 III](https://leetcode.cn/problems/path-sum-iii/)

- 🟡 **难度**：中等
- 🌳 **分类**：二叉树
- 🏷️ **标签**：`树`、`深度优先搜索`、`二叉树`

## 题目描述

给定一个二叉树的根节点 `root` ，和一个整数 `targetSum` ，求该二叉树里节点值之和等于 `targetSum` 的 **路径** 的数目。

**路径** 不需要从根节点开始，也不需要在叶子节点结束，但是路径方向必须是向下的（只能从父节点到子节点）。

**示例 1：**

![](https://assets.leetcode.com/uploads/2021/04/09/pathsum3-1-tree.jpg)

```
输入：root = [10,5,-3,3,2,null,11,3,-2,null,1], targetSum = 8
输出：3
解释：和等于 8 的路径有 3 条，如图所示。
```

**示例 2：**

```
输入：root = [5,4,8,11,null,13,4,7,2,null,null,5,1], targetSum = 22
输出：3
```

**提示:**
- 二叉树的节点个数的范围是 `[0,1000]`
- `-10^9 <= Node.val <= 10^9`
- `-1000 <= targetSum <= 1000`

> ⚠️ 注意值域：10⁹ × 深度 10³ = 10¹²，前缀和远超 `int` 上限（2.1×10⁹），全程必须使用 `long`。

## 题解一：前缀和链 + 回溯

路径必须是"祖先 → 后代"的向下链。维护一条**当前链的前缀和列表**（下标 0 是虚拟根的前缀 0，末尾是当前节点）：任意"祖先 → 我"的路径和 = `acc[我] - acc[祖先]`（前缀相减，98/560 同款思想）。每个节点入链后，让祖先指针滑过整条链做配对计数；左右子树递归完毕后**出链回溯**。

> 回溯的两行 `dfs` 与一行 `remove`：进入节点 = 链尾追加一节（并把"含自己"的前缀传给孩子）；离开节点 = 链尾缩回一节。三行合起来保证**链的内容永远精确复刻当前的递归栈**——若缺少 remove，兄弟子树的前缀会残留在链中冒充祖先，产生 `3→2` 这类不存在的"跨链路径"。

自检两处边界：① 内层扫描**排除自己**（`i < last`），否则 `target == 0` 时"自己配自己"的空路径会多计；② 链底放**虚拟 0**（空路径前缀），使"从根出发"的路径可配对（`[1], k=1` 用例）。

- 时间复杂度：`O(n·h)`，每个节点回头扫一条链，最坏 O(n²)
- 空间复杂度：`O(h)`，链长 = 树高（递归栈同深）

### ☕ Java

```java
import java.util.ArrayList;
import java.util.LinkedList;
import java.util.List;
import java.util.Queue;
import java.util.Scanner;

public class Solution {
    public static int ans;
    public static int target;
    public static List<Long> path;   // 当前链的前缀和（long 防溢出）

    public int pathSum(TreeNode root, int targetSum) {
        ans = 0;
        target = targetSum;
        path = new ArrayList<>();
        path.add(0L);                // 虚拟根：空路径前缀为 0，使"从根出发"的路径可配对
        dfs(root, 0L);
        return ans;
    }

    public static void dfs(TreeNode root, long cur) {
        if (root == null) {
            return;
        }
        path.add(cur + root.val);    // 入链：我的前缀 = 父链前缀 + 自己
        int last = path.size() - 1;
        for (int i = 0; i < last; i++) {   // 祖先滑过每一位（不含自己，排除空路径配对）
            if (target == path.get(last) - path.get(i)) {
                ans++;
            }
        }
        dfs(root.left, cur + root.val);    // 把"含自己"的前缀传给孩子
        dfs(root.right, cur + root.val);
        path.remove(path.size() - 1);      // 出链回溯：链内容 = 当前递归栈
    }

    public static void main(String[] args) {
        // ACM 输入：第一行层序序列（null 表示空节点），第二行 targetSum
        Scanner sc = new Scanner(System.in);
        java.util.List<String> tokens = new java.util.ArrayList<>();
        while (sc.hasNext()) {
            tokens.add(sc.next());
        }
        int k = Integer.parseInt(tokens.get(tokens.size() - 1));
        tokens.remove(tokens.size() - 1);
        TreeNode root = buildTree(tokens);
        System.out.println(new Solution().pathSum(root, k));
        sc.close();
    }

    // 按层序遍历读入建树，null 表示空节点
    public static TreeNode buildTree(java.util.List<String> tokens) {
        if (tokens.isEmpty() || "null".equals(tokens.get(0))) {
            return null;
        }
        TreeNode root = new TreeNode(Integer.parseInt(tokens.get(0)));
        Queue<TreeNode> queue = new LinkedList<>();
        queue.offer(root);
        int i = 1;
        while (!queue.isEmpty() && i < tokens.size()) {
            TreeNode cur = queue.poll();
            if (!"null".equals(tokens.get(i))) {
                cur.left = new TreeNode(Integer.parseInt(tokens.get(i)));
                queue.offer(cur.left);
            }
            i++;
            if (i >= tokens.size()) {
                break;
            }
            if (!"null".equals(tokens.get(i))) {
                cur.right = new TreeNode(Integer.parseInt(tokens.get(i)));
                queue.offer(cur.right);
            }
            i++;
        }
        return root;
    }
}

class TreeNode {
    int val;
    TreeNode left;
    TreeNode right;

    TreeNode() {}

    TreeNode(int val) { this.val = val; }

    TreeNode(int val, TreeNode left, TreeNode right) {
        this.val = val;
        this.left = left;
        this.right = right;
    }
}
```

```python
import sys
from collections import deque
from typing import List, Optional


class TreeNode:
    def __init__(self, val=0, left=None, right=None):
        self.val = val
        self.left = left
        self.right = right


class Solution:
    def pathSum(self, root: Optional[TreeNode], targetSum: int) -> int:
        self.ans = 0
        self.target = targetSum
        self.path = [0]              # 虚拟根：空路径前缀
        self.dfs(root, 0)
        return self.ans

    def dfs(self, node: Optional[TreeNode], cur: int) -> None:
        if node is None:
            return
        cur += node.val              # 我的前缀 = 父链前缀 + 自己
        self.path.append(cur)        # 入链
        last = len(self.path) - 1
        for i in range(last):        # 祖先滑过每一位
            if self.target == self.path[last] - self.path[i]:
                self.ans += 1
        self.dfs(node.left, cur)
        self.dfs(node.right, cur)
        self.path.pop()              # 出链回溯


def build_tree(tokens, pos):
    """按层序遍历建树，'null' 表示空节点"""
    if pos[0] >= len(tokens) or tokens[pos[0]] == "null":
        pos[0] += 1
        return None
    root = TreeNode(int(tokens[pos[0]]))
    pos[0] += 1
    queue = deque([root])
    while queue:
        cur = queue.popleft()
        for i in range(2):
            if pos[0] >= len(tokens):
                break
            child = None
            if tokens[pos[0]] != "null":
                child = TreeNode(int(tokens[pos[0]]))
                queue.append(child)
            if i == 0:
                cur.left = child
            else:
                cur.right = child
            pos[0] += 1
    return root


if __name__ == "__main__":
    # ACM 输入：第一行层序序列（null 表示空节点），第二行 targetSum
    tokens = sys.stdin.read().split()
    k = int(tokens[-1])
    root = build_tree(tokens[:-1], [0])
    print(Solution().pathSum(root, k))
```

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct TreeNode {
    int val;
    struct TreeNode* left;
    struct TreeNode* right;
} TreeNode;

static int ans = 0;
static int target = 0;

// 当前链的前缀和（long long 防溢出）
static long long path[1001];
static int pathLen = 0;

void dfs(struct TreeNode* node, long long cur) {
    if (node == NULL) {
        return;
    }
    cur += node->val;            // 我的前缀 = 父链前缀 + 自己
    path[pathLen++] = cur;       // 入链
    for (int i = 0; i < pathLen - 1; i++) {   // 祖先滑过每一位
        if (target == path[pathLen - 1] - path[i]) {
            ans++;
        }
    }
    dfs(node->left, cur);
    dfs(node->right, cur);
    pathLen--;                   // 出链回溯
}

int pathSum(struct TreeNode* root, int targetSum) {
    ans = 0;
    target = targetSum;
    pathLen = 0;
    path[pathLen++] = 0;         // 虚拟根：空路径前缀
    dfs(root, 0LL);
    return ans;
}

// 按层序遍历读入建树，"null" 表示空节点
TreeNode* buildTree(char tokens[][16], int n) {
    if (n == 0 || strcmp(tokens[0], "null") == 0) {
        return NULL;
    }
    TreeNode** queue = (TreeNode**)malloc(n * sizeof(TreeNode*));
    int head = 0, tail = 0, i = 1;
    TreeNode* root = (TreeNode*)malloc(sizeof(TreeNode));
    root->val = atoi(tokens[0]);
    root->left = root->right = NULL;
    queue[tail++] = root;
    while (head < tail && i < n) {
        TreeNode* cur = queue[head++];
        if (strcmp(tokens[i], "null") != 0) {
            cur->left = (TreeNode*)malloc(sizeof(TreeNode));
            cur->left->val = atoi(tokens[i]);
            cur->left->left = cur->left->right = NULL;
            queue[tail++] = cur->left;
        }
        i++;
        if (i >= n) {
            break;
        }
        if (strcmp(tokens[i], "null") != 0) {
            cur->right = (TreeNode*)malloc(sizeof(TreeNode));
            cur->right->val = atoi(tokens[i]);
            cur->right->left = cur->right->right = NULL;
            queue[tail++] = cur->right;
        }
        i++;
    }
    free(queue);
    return root;
}

int main(void) {
    // ACM 输入：一行层序序列（null 表示空节点），最后一个数字是 targetSum
    char tokens[2002][16];
    int n = 0;
    while (n < 2002 && scanf("%15s", tokens[n]) == 1) {
        n++;
    }
    int k = atoi(tokens[n - 1]);
    TreeNode* root = buildTree(tokens, n - 1);
    printf("%d\n", pathSum(root, k));
    return 0;
}
```

## 题解二：前缀和哈希表（O(n) 查表）

题解一的升级：回头扫祖先的 O(h) 循环，被一次 **O(1) 哈希查询**替代。`map[前缀和] → 该前缀和在当前链上出现的次数`，配对公式移项为"查 `cur - target` 有几个祖先"。核心仍是**回溯**：递归离开节点时把它对 map 的贡献撤销，保证 map 里只有当前链的祖先。

- 时间复杂度：`O(n)`，每个节点 O(1) 进出
- 空间复杂度：`O(h)`，map 最多存一条链的前缀（回溯保证）+ 递归栈

### ☕ Java

```java
import java.util.HashMap;
import java.util.LinkedList;
import java.util.Map;
import java.util.Queue;
import java.util.Scanner;

public class Solution {
    private static Map<Long, Integer> map = new HashMap<>();
    private static int target = 0;
    private static int ans = 0;

    public int pathSum(TreeNode root, int targetSum) {
        target = targetSum;
        ans = 0;
        map.clear();
        map.put(0L, 1);              // 虚拟根：空路径前缀出现 1 次
        dfs(root, 0L);
        return ans;
    }

    public static void dfs(TreeNode node, long cur) {
        if (node == null) {
            return;
        }
        cur += node.val;             // 我的前缀 = 父链前缀 + 自己
        // 先查后插（顺序反了会在 target=0 时把自己配给自己）
        ans += map.getOrDefault(cur - target, 0);
        map.merge(cur, 1, Integer::sum);      // 入表
        dfs(node.left, cur);
        dfs(node.right, cur);
        map.merge(cur, -1, Integer::sum);     // 出表回溯：离开这条链，撤销贡献
    }

    public static void main(String[] args) {
        // ACM 输入：第一行层序序列（null 表示空节点），第二行 targetSum
        Scanner sc = new Scanner(System.in);
        java.util.List<String> tokens = new java.util.ArrayList<>();
        while (sc.hasNext()) {
            tokens.add(sc.next());
        }
        int k = Integer.parseInt(tokens.get(tokens.size() - 1));
        tokens.remove(tokens.size() - 1);
        TreeNode root = buildTree(tokens);
        System.out.println(new Solution().pathSum(root, k));
        sc.close();
    }

    // 按层序遍历读入建树，null 表示空节点
    public static TreeNode buildTree(java.util.List<String> tokens) {
        if (tokens.isEmpty() || "null".equals(tokens.get(0))) {
            return null;
        }
        TreeNode root = new TreeNode(Integer.parseInt(tokens.get(0)));
        Queue<TreeNode> queue = new LinkedList<>();
        queue.offer(root);
        int i = 1;
        while (!queue.isEmpty() && i < tokens.size()) {
            TreeNode cur = queue.poll();
            if (!"null".equals(tokens.get(i))) {
                cur.left = new TreeNode(Integer.parseInt(tokens.get(i)));
                queue.offer(cur.left);
            }
            i++;
            if (i >= tokens.size()) {
                break;
            }
            if (!"null".equals(tokens.get(i))) {
                cur.right = new TreeNode(Integer.parseInt(tokens.get(i)));
                queue.offer(cur.right);
            }
            i++;
        }
        return root;
    }
}
```

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct TreeNode {
    int val;
    struct TreeNode* left;
    struct TreeNode* right;
} TreeNode;

// 手写开放寻址哈希：long long 前缀和 → 计数
#define CAP 4096
static long long hkeys[CAP];
static int hcnt[CAP];
static char hused[CAP];

static unsigned long long hashLL(long long x) {
    return (unsigned long long)x * 0x9E3779B97F4A7C15ULL >> 16;
}

static void hAdd(long long key, int delta) {
    unsigned int h = (unsigned int)(hashLL(key) & (CAP - 1));
    while (hused[h]) {
        if (hkeys[h] == key) {
            hcnt[h] += delta;
            return;
        }
        h = (h + 1) & (CAP - 1);
    }
    hused[h] = 1;
    hkeys[h] = key;
    hcnt[h] = delta;
}

static int hGet(long long key) {
    unsigned int h = (unsigned int)(hashLL(key) & (CAP - 1));
    while (hused[h]) {
        if (hkeys[h] == key) {
            return hcnt[h];
        }
        h = (h + 1) & (CAP - 1);
    }
    return 0;
}

static int target = 0;
static int ans = 0;

void dfs(struct TreeNode* node, long long cur) {
    if (node == NULL) {
        return;
    }
    cur += node->val;            // 我的前缀 = 父链前缀 + 自己
    ans += hGet(cur - target);   // 先查后插
    hAdd(cur, 1);                // 入表
    dfs(node->left, cur);
    dfs(node->right, cur);
    hAdd(cur, -1);               // 出表回溯：撤销贡献
}

int pathSum(struct TreeNode* root, int targetSum) {
    target = targetSum;
    ans = 0;
    memset(hused, 0, sizeof(hused));
    hAdd(0, 1);                  // 虚拟根：空路径前缀出现 1 次
    dfs(root, 0LL);
    return ans;
}

// 按层序遍历读入建树，"null" 表示空节点
TreeNode* buildTree(char tokens[][16], int n) {
    if (n == 0 || strcmp(tokens[0], "null") == 0) {
        return NULL;
    }
    TreeNode** queue = (TreeNode**)malloc(n * sizeof(TreeNode*));
    int head = 0, tail = 0, i = 1;
    TreeNode* root = (TreeNode*)malloc(sizeof(TreeNode));
    root->val = atoi(tokens[0]);
    root->left = root->right = NULL;
    queue[tail++] = root;
    while (head < tail && i < n) {
        TreeNode* cur = queue[head++];
        if (strcmp(tokens[i], "null") != 0) {
            cur->left = (TreeNode*)malloc(sizeof(TreeNode));
            cur->left->val = atoi(tokens[i]);
            cur->left->left = cur->left->right = NULL;
            queue[tail++] = cur->left;
        }
        i++;
        if (i >= n) {
            break;
        }
        if (strcmp(tokens[i], "null") != 0) {
            cur->right = (TreeNode*)malloc(sizeof(TreeNode));
            cur->right->val = atoi(tokens[i]);
            cur->right->left = cur->right->right = NULL;
            queue[tail++] = cur->right;
        }
        i++;
    }
    free(queue);
    return root;
}

int main(void) {
    // ACM 输入：一行层序序列（null 表示空节点），最后一个数字是 targetSum
    char tokens[2002][16];
    int n = 0;
    while (n < 2002 && scanf("%15s", tokens[n]) == 1) {
        n++;
    }
    int k = atoi(tokens[n - 1]);
    TreeNode* root = buildTree(tokens, n - 1);
    printf("%d\n", pathSum(root, k));
    return 0;
}
```
