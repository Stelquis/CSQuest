# [105. 从前序与中序遍历序列构造二叉树](https://leetcode.cn/problems/construct-binary-tree-from-preorder-and-inorder-traversal/)

- 🟡 **难度**：中等
- 🌳 **分类**：二叉树
- 🏷️ **标签**：`树`、`数组`、`哈希表`、`分治`、`二叉树`

## 题目描述

给定两个整数数组 `preorder` 和 `inorder` ，其中 `preorder` 是二叉树的**先序遍历**， `inorder` 是同一棵树的**中序遍历**，请构造二叉树并返回其根节点。

**示例 1:**

![](https://assets.leetcode.com/uploads/2021/02/19/tree.jpg)

```
输入: preorder = [3,9,20,15,7], inorder = [9,3,15,20,7]
输出: [3,9,20,null,null,15,7]
```

**示例 2:**

```
输入: preorder = [-1], inorder = [-1]
输出: [-1]
```

**提示:**
- `1 <= preorder.length <= 3000`
- `inorder.length == preorder.length`
- `-3000 <= preorder[i], inorder[i] <= 3000`
- `preorder` 和 `inorder` 均 **无重复** 元素
- `inorder` 均出现在 `preorder`
- `preorder` **保证** 为二叉树的前序遍历序列
- `inorder` **保证** 为二叉树的中序遍历序列

## 题解一：分治 + 哈希定位（填充式）

核心观察：**前序管“根是谁”，中序管“左右怎么分”**——前序首元素必是根；在无重复元素的前提下用哈希表 `值 → 中序下标` O(1) 定位根；根在中序中的位置把中序切成左、右两段，其长度 `L = idx - inStart` 决定了前序中左右子树的边界。

四个区间端点（由 `L` 推出）：

| 子问题 | 前序区间 | 中序区间 |
|---|---|---|
| 左子树 | `[preStart+1, preStart+L]` | `[inStart, rootIndex-1]` |
| 右子树 | `[preStart+L+1, preEnd]` | `[rootIndex+1, inEnd]` |

自检套路——**数格子守恒**：两个子问题的长度之和 = 本轮长度 - 1（根被挖掉），任何一条不等即端点写错。另有一个经典陷阱：`index` 是**中序**的下标，若拿去索引**前序**（如 `preorder[index+1]`）必然出错——前序取“值”（下一个根是谁），中序给“范围”（左右分界）。

- 时间复杂度：`O(n)`，每个节点建一次，哈希定位 O(1)（无哈希则每轮线性扫描退化为 O(n²)）
- 空间复杂度：`O(n)`，哈希表 + `O(h)` 递归栈

### ☕ Java

```java
import java.util.ArrayList;
import java.util.HashMap;
import java.util.LinkedList;
import java.util.List;
import java.util.Map;
import java.util.Queue;
import java.util.Scanner;

public class Solution {
    public TreeNode buildTree(int[] preorder, int[] inorder) {
        Map<Integer, Integer> map = new HashMap<>();
        for (int i = 0; i < inorder.length; i++) {
            map.put(inorder[i], i);
        }
        if (preorder.length == 0) {
            return null;
        }
        // 前序的首个元素就是根节点
        TreeNode root = new TreeNode(preorder[0]);
        build(root, preorder, inorder, map, 0, preorder.length - 1, 0, inorder.length - 1);
        return root;
    }

    public static void build(TreeNode root, int[] preorder, int[] inorder,
                             Map<Integer, Integer> map,
                             int preStart, int preEnd, int inStart, int inEnd) {
        // 定位根在中序中的位置，计算左右子树的节点个数（均为中序口径）
        int rootIndex = map.get(root.val);
        int leftCount = rootIndex - inStart;
        int rightCount = inEnd - rootIndex;
        if (leftCount > 0) {      // 有左子树才创建，避免脏节点
            root.left = new TreeNode(preorder[preStart + 1]);
            build(root.left, preorder, inorder, map,
                    preStart + 1, preStart + leftCount, inStart, rootIndex - 1);
        }
        if (rightCount > 0) {     // 有右子树才创建
            root.right = new TreeNode(preorder[preStart + leftCount + 1]);
            build(root.right, preorder, inorder, map,
                    preStart + leftCount + 1, preEnd, rootIndex + 1, inEnd);
        }
    }

    public static void main(String[] args) {
        // ACM 输入：第一行前序序列，第二行中序序列（均无 null，空格分隔）
        Scanner sc = new Scanner(System.in);
        int[] preorder = readArray(sc.nextLine());
        int[] inorder = sc.hasNextLine() ? readArray(sc.nextLine()) : new int[0];
        TreeNode root = new Solution().buildTree(preorder, inorder);
        System.out.println(levelOrder(root));
        sc.close();
    }

    public static int[] readArray(String line) {
        List<Integer> list = new ArrayList<>();
        for (String s : line.trim().split("\\s+")) {
            list.add(Integer.parseInt(s));
        }
        int[] arr = new int[list.size()];
        for (int i = 0; i < arr.length; i++) {
            arr[i] = list.get(i);
        }
        return arr;
    }

    // 层序输出（含 null 占位，修剪末尾），如 [3, 9, 20, null, null, 15, 7]
    public static String levelOrder(TreeNode root) {
        if (root == null) {
            return "[]";
        }
        Queue<TreeNode> queue = new LinkedList<>();
        queue.offer(root);
        List<String> items = new ArrayList<>();
        items.add(String.valueOf(root.val));
        while (!queue.isEmpty()) {
            TreeNode cur = queue.poll();
            for (TreeNode child : new TreeNode[]{cur.left, cur.right}) {
                if (child != null) {
                    items.add(String.valueOf(child.val));
                    queue.offer(child);
                } else {
                    items.add("null");
                }
            }
        }
        int end = items.size();
        while (end > 0 && items.get(end - 1).equals("null")) {
            end--;
        }
        return items.subList(0, end).toString().replace(" ", "");
    }
}

class TreeNode {
    int val;
    TreeNode left;
    TreeNode right;

    TreeNode() {}

    TreeNode(int val) { this.val = val; }

    TreeNode(int val, TreeNode left, TreeNode right) {
        this.val = val;
        this.left = left;
        this.right = right;
    }
}
```

```python
from collections import deque
from typing import List, Optional


class TreeNode:
    def __init__(self, val=0, left=None, right=None):
        self.val = val
        self.left = left
        self.right = right


class Solution:
    def buildTree(self, preorder: List[int], inorder: List[int]) -> Optional[TreeNode]:
        pos = {v: i for i, v in enumerate(inorder)}   # 值 → 中序下标
        if not preorder:
            return None
        root = TreeNode(preorder[0])

        def build(node, pre_start, pre_end, in_start, in_end):
            root_index = pos[node.val]
            left_count = root_index - in_start
            right_count = in_end - root_index
            if left_count > 0:
                node.left = TreeNode(preorder[pre_start + 1])
                build(node.left, pre_start + 1, pre_start + left_count,
                      in_start, root_index - 1)
            if right_count > 0:
                node.right = TreeNode(preorder[pre_start + left_count + 1])
                build(node.right, pre_start + left_count + 1,
                      pre_end, root_index + 1, in_end)

        build(root, 0, len(preorder) - 1, 0, len(inorder) - 1)
        return root

                queue.append(child)
            else:

    while end > 0 and items[end - 1] == "null":
        end -= 1
    return "[" + ", ".join(items[:end]) + "]"
```

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct TreeNode {
    int val;
    struct TreeNode* left;
    struct TreeNode* right;
} TreeNode;

// 值范围 -3000..3000，用偏移数组当哈希：值 → 中序下标
static int posTable[6001];
#define OFFSET 3000

void build(struct TreeNode* root, int* preorder,
           int preStart, int preEnd, int inStart, int inEnd) {
    int rootIndex = posTable[root->val + OFFSET];
    int leftCount = rootIndex - inStart;
    int rightCount = inEnd - rootIndex;
    if (leftCount > 0) {
        root->left = (TreeNode*)malloc(sizeof(TreeNode));
        root->left->val = preorder[preStart + 1];
        root->left->left = root->left->right = NULL;
        build(root->left, preorder, preStart + 1, preStart + leftCount, inStart, rootIndex - 1);
    }
    if (rightCount > 0) {
        root->right = (TreeNode*)malloc(sizeof(TreeNode));
        root->right->val = preorder[preStart + leftCount + 1];
        root->right->left = root->right->right = NULL;
        build(root->right, preorder, preStart + leftCount + 1, preEnd, rootIndex + 1, inEnd);
    }
}

TreeNode* buildTree(int* preorder, int* inorder, int n) {
    for (int i = 0; i < n; i++) {
        posTable[inorder[i] + OFFSET] = i;
    }
    if (n == 0) {
        return NULL;
    }
    TreeNode* root = (TreeNode*)malloc(sizeof(TreeNode));
    root->val = preorder[0];
    root->left = root->right = NULL;
    build(root, preorder, 0, n - 1, 0, n - 1);
    return root;
}

// 层序输出（含 null 占位，修剪末尾）
void levelOrder(TreeNode* root) {
    if (root == NULL) {
        printf("[]\n");
        return;
    }
    TreeNode** queue = (TreeNode**)malloc(6000 * sizeof(TreeNode*));
    char* items[12000];
    int head = 0, tail = 0, cnt = 0;
    char buf[16];
    queue[tail++] = root;
    sprintf(buf, "%d", root->val);
    items[cnt++] = strdup(buf);
    while (head < tail) {
        TreeNode* cur = queue[head++];
        TreeNode* kids[2] = {cur->left, cur->right};
        for (int i = 0; i < 2; i++) {
            if (kids[i] != NULL) {
                sprintf(buf, "%d", kids[i]->val);
                items[cnt++] = strdup(buf);
                queue[tail++] = kids[i];
            } else {
                items[cnt++] = "null";
            }
        }
    }
    int end = cnt;
    while (end > 0 && strcmp(items[end - 1], "null") == 0) {
        end--;
    }
    printf("[");
    for (int i = 0; i < end; i++) {
        printf("%s%s", items[i], i + 1 < end ? ", " : "");
    }
    printf("]\n");
}

    int np = 0, ni = 0;

    if (fgets(line, sizeof(line), stdin) != NULL) {

            tok = strtok(NULL, " \n");

            inorder[ni++] = atoi(tok);
            tok = strtok(NULL, " \n");
        }
    }
    TreeNode* root = buildTree(preorder, inorder, np);
    levelOrder(root);
    return 0;
}
```

## 题解二：返回式分治（区间空即出口）

同一思路的更简洁形态：函数**自己创建并返回**子树根，空区间（`preL > preR`）直接返回 `null`——出口天然覆盖“无左/无右子树”，不需要 `leftCount > 0` 守卫和预创建节点，也避免了“传入待填节点”带来的耦合。

- 时间复杂度：`O(n)`
- 空间复杂度：`O(n)`（哈希）+ `O(h)` 递归栈

### ☕ Java

```java
import java.util.ArrayList;
import java.util.HashMap;
import java.util.LinkedList;
import java.util.List;
import java.util.Map;
import java.util.Queue;
import java.util.Scanner;

public class Solution {
    private static Map<Integer, Integer> pos = new HashMap<>();

    public TreeNode buildTree(int[] preorder, int[] inorder) {
        for (int i = 0; i < inorder.length; i++) {
            pos.put(inorder[i], i);
        }
        return build(preorder, 0, preorder.length - 1, 0, inorder.length - 1);
    }

    // 返回由 前序[preL..preR] + 中序[inL..inR] 建成的子树根；空区间返回 null
    public static TreeNode build(int[] preorder, int preL, int preR, int inL, int inR) {
        if (preL > preR) {
            return null;                        // 区间空 → 空子树
        }
        TreeNode root = new TreeNode(preorder[preL]);   // 前序首 = 根
        int idx = pos.get(root.val);                    // 根在中序的位置
        int leftCount = idx - inL;
        root.left = build(preorder, preL + 1, preL + leftCount, inL, idx - 1);
        root.right = build(preorder, preL + leftCount + 1, preR, idx + 1, inR);
        return root;
    }

    public static void main(String[] args) {
        // ACM 输入：第一行前序序列，第二行中序序列（均无 null，空格分隔）
        Scanner sc = new Scanner(System.in);
        int[] preorder = readArray(sc.nextLine());
        int[] inorder = sc.hasNextLine() ? readArray(sc.nextLine()) : new int[0];
        TreeNode root = new Solution().buildTree(preorder, inorder);
        System.out.println(levelOrder(root));
        sc.close();
    }

    public static int[] readArray(String line) {
        List<Integer> list = new ArrayList<>();
        for (String s : line.trim().split("\\s+")) {
            list.add(Integer.parseInt(s));
        }
        int[] arr = new int[list.size()];
        for (int i = 0; i < arr.length; i++) {
            arr[i] = list.get(i);
        }
        return arr;
    }

    // 层序输出（含 null 占位，修剪末尾）
    public static String levelOrder(TreeNode root) {
        if (root == null) {
            return "[]";
        }
        Queue<TreeNode> queue = new LinkedList<>();
        queue.offer(root);
        List<String> items = new ArrayList<>();
        items.add(String.valueOf(root.val));
        while (!queue.isEmpty()) {
            TreeNode cur = queue.poll();
            for (TreeNode child : new TreeNode[]{cur.left, cur.right}) {
                if (child != null) {
                    items.add(String.valueOf(child.val));
                    queue.offer(child);
                } else {
                    items.add("null");
                }
            }
        }
        int end = items.size();
        while (end > 0 && items.get(end - 1).equals("null")) {
            end--;
        }
        return items.subList(0, end).toString().replace(" ", "");
    }
}

    TreeNode right;
```

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct TreeNode {
    int val;
    struct TreeNode* left;
    struct TreeNode* right;
} TreeNode;

#define OFFSET 3000
static int posTable[6001];   // 值 → 中序下标

// 返回式分治：空区间返回 NULL
TreeNode* build(int* preorder, int preL, int preR, int inL, int inR) {
    if (preL > preR) {
        return NULL;
    }
    TreeNode* root = (TreeNode*)malloc(sizeof(TreeNode));
    root->val = preorder[preL];
    root->left = root->right = NULL;
    int idx = posTable[root->val + OFFSET];
    int leftCount = idx - inL;
    root->left = build(preorder, preL + 1, preL + leftCount, inL, idx - 1);
    root->right = build(preorder, preL + leftCount + 1, preR, idx + 1, inR);
    return root;
}

TreeNode* buildTree(int* preorder, int* inorder, int n) {
    for (int i = 0; i < n; i++) {
        posTable[inorder[i] + OFFSET] = i;
    }
    return build(preorder, 0, n - 1, 0, n - 1);
}

void levelOrder(TreeNode* root) {
    if (root == NULL) {
        printf("[]\n");
        return;
    }
    TreeNode** queue = (TreeNode**)malloc(6000 * sizeof(TreeNode*));
    char* items[12000];
    int head = 0, tail = 0, cnt = 0;
    char buf[16];
    queue[tail++] = root;
    sprintf(buf, "%d", root->val);
    items[cnt++] = strdup(buf);
    while (head < tail) {
        TreeNode* cur = queue[head++];
        TreeNode* kids[2] = {cur->left, cur->right};
        for (int i = 0; i < 2; i++) {
            if (kids[i] != NULL) {
                sprintf(buf, "%d", kids[i]->val);
                items[cnt++] = strdup(buf);
                queue[tail++] = kids[i];
            } else {
                items[cnt++] = "null";
            }
        }
    }
    int end = cnt;
    while (end > 0 && strcmp(items[end - 1], "null") == 0) {
        end--;
    }
    printf("[");
    for (int i = 0; i < end; i++) {
        printf("%s%s", items[i], i + 1 < end ? ", " : "");
    }
    printf("]\n");
}

int main(void) {
    // ACM 输入：第一行前序序列，第二行中序序列
    int preorder[3000], inorder[3000];
    int np = 0, ni = 0;
    char line[30000];
    if (fgets(line, sizeof(line), stdin) != NULL) {
        char* tok = strtok(line, " \n");
        while (tok != NULL && np < 3000) {
            preorder[np++] = atoi(tok);
            tok = strtok(NULL, " \n");
        }
    }
    if (fgets(line, sizeof(line), stdin) != NULL) {
        char* tok = strtok(line, " \n");
        while (tok != NULL && ni < 3000) {
            inorder[ni++] = atoi(tok);
            tok = strtok(NULL, " \n");
        }
    }
    TreeNode* root = buildTree(preorder, inorder, np);
    levelOrder(root);
    return 0;
}
```
