# [102. 二叉树的层序遍历](https://leetcode.cn/problems/binary-tree-level-order-traversal/)

- 🟡 **难度**：中等
- 🌳 **分类**：二叉树
- 🏷️ **标签**：`树`、`广度优先搜索`、`二叉树`

## 题目描述

给你二叉树的根节点 `root` ，返回其节点值的 **层序遍历** 。 （即逐层地，从左到右访问所有节点）。

**示例 1：**

![](https://assets.leetcode.com/uploads/2021/02/19/tree1.jpg)

```
输入：root = [3,9,20,null,null,15,7]
输出：[[3],[9,20],[15,7]]
```

**示例 2：**

```
输入：root = [1]
输出：[[1]]
```

**示例 3：**

```
输入：root = []
输出：[]
```

**提示：**
- 树中节点数目在范围 `[0, 2000]` 内
- `-1000 <= Node.val <= 1000`

## 题解一：广度优先搜索（队列 + 分层处理）

队列天然维护"先来先访问"：出队一个节点时把它的孩子排到队尾，第 k 层的节点全部出队后，队列里恰好是第 k+1 层的全部节点。**进入循环时先固化 `size = queue.size()`**，一次性处理完这一批，即可把层与层切开。

- 时间复杂度：`O(n)`，每个节点入队、出队各一次
- 空间复杂度：`O(w)`，`w` 为树的最大宽度，队列最坏存一整层（约 n/2）

### ☕ Java

```java
import java.util.ArrayDeque;
import java.util.ArrayList;
import java.util.Deque;
import java.util.LinkedList;
import java.util.List;
import java.util.Queue;
import java.util.Scanner;

public class Solution {
    public List<List<Integer>> levelOrder(TreeNode root) {
        if (root == null) {
            return new ArrayList<>();
        }
        List<List<Integer>> ans = new ArrayList<>();
        Queue<TreeNode> queue = new ArrayDeque<>();
        queue.offer(root);
        while (!queue.isEmpty()) {
            List<Integer> level = new ArrayList<>();
            int size = queue.size();          // 固化本层人数
            for (int i = 0; i < size; i++) {
                TreeNode node = queue.poll();
                level.add(node.val);
                if (node.left != null) {
                    queue.offer(node.left);
                }
                if (node.right != null) {
                    queue.offer(node.right);
                }
            }
            ans.add(level);                   // 每层交货
        }
        return ans;
    }

    public static void main(String[] args) {
        // ACM 输入：一行层序序列，null 表示空节点
        Scanner sc = new Scanner(System.in);
        TreeNode root = buildTree(sc);
        System.out.println(new Solution().levelOrder(root));
        sc.close();
    }

    // 按层序遍历读入建树，null 表示空节点
    public static TreeNode buildTree(Scanner sc) {
        if (!sc.hasNext()) {
            return null;
        }
        String token = sc.next();
        if ("null".equals(token)) {
            return null;
        }
        TreeNode root = new TreeNode(Integer.parseInt(token));
        Queue<TreeNode> queue = new LinkedList<>();
        queue.offer(root);
        while (!queue.isEmpty()) {
            TreeNode cur = queue.poll();
            if (!sc.hasNext()) {
                break;
            }
            String left = sc.next();
            if (!"null".equals(left)) {
                cur.left = new TreeNode(Integer.parseInt(left));
                queue.offer(cur.left);
            }
            if (!sc.hasNext()) {
                break;
            }
            String right = sc.next();
            if (!"null".equals(right)) {
                cur.right = new TreeNode(Integer.parseInt(right));
                queue.offer(cur.right);
            }
        }
        return root;
    }
}

class TreeNode {
    int val;
    TreeNode left;
    TreeNode right;

    TreeNode() {}

    TreeNode(int val) { this.val = val; }

    TreeNode(int val, TreeNode left, TreeNode right) {
        this.val = val;
        this.left = left;
        this.right = right;
    }
}
```

```python
            size = len(queue)          # 固化本层人数

            for _ in range(size):

def build_tree(tokens, pos):
    """按层序遍历建树，'null' 表示空节点"""
    if pos[0] >= len(tokens) or tokens[pos[0]] == "null":
        pos[0] += 1
        return None
    root = TreeNode(int(tokens[pos[0]]))
    pos[0] += 1
    queue = deque([root])
    while queue:
        cur = queue.popleft()
        for i in range(2):
            if pos[0] >= len(tokens):
                break

            if i == 0:
                cur.left = child
            else:
                cur.right = child
            pos[0] += 1
    return root
```

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct TreeNode {
    int val;
    struct TreeNode* left;
    struct TreeNode* right;
} TreeNode;

// 手写队列
typedef struct {
    TreeNode** data;
    int head;
    int tail;
} Queue;

void queueInit(Queue* q, int cap) {
    q->data = (TreeNode**)malloc(cap * sizeof(TreeNode*));
    q->head = q->tail = 0;
}
void queuePush(Queue* q, TreeNode* node) { q->data[q->tail++] = node; }
TreeNode* queuePop(Queue* q) { return q->data[q->head++]; }
int queueEmpty(Queue* q) { return q->head == q->tail; }

#define MAXN 4000

int main(void) {
    // ACM 输入：一行层序序列，null 表示空节点
    char tokens[4000][16];
    int n = 0;
    while (n < 4000 && scanf("%15s", tokens[n]) == 1) {
        n++;
    }
    if (n == 0 || strcmp(tokens[0], "null") == 0) {
        printf("[]\n");
        return 0;
    }

    // 建树
    Queue queue;
    queueInit(&queue, MAXN);
    TreeNode* root = (TreeNode*)malloc(sizeof(TreeNode));
    root->val = atoi(tokens[0]);
    root->left = root->right = NULL;
    queuePush(&queue, root);
    int i = 1;
    while (!queueEmpty(&queue) && i < n) {
        TreeNode* cur = queuePop(&queue);
        if (strcmp(tokens[i], "null") != 0) {
            cur->left = (TreeNode*)malloc(sizeof(TreeNode));
            cur->left->val = atoi(tokens[i]);
            cur->left->left = cur->left->right = NULL;
            queuePush(&queue, cur->left);
        }
        i++;
        if (i >= n) {
            break;
        }
        if (strcmp(tokens[i], "null") != 0) {
            cur->right = (TreeNode*)malloc(sizeof(TreeNode));
            cur->right->val = atoi(tokens[i]);
            cur->right->left = cur->right->right = NULL;
            queuePush(&queue, cur->right);
        }
        i++;
    }

    // 层序遍历：建树时队列已耗尽，重置后从根重新开始
    queue.head = queue.tail = 0;
    queuePush(&queue, root);
    int first = 1;
    printf("[");
    while (!queueEmpty(&queue)) {
        int size = queue.tail - queue.head;   // 本层人数
        printf(first ? "[" : ", [");
        first = 0;
        for (int j = 0; j < size; j++) {
            TreeNode* node = queuePop(&queue);
            printf("%d%s", node->val, j + 1 < size ? ", " : "");
            if (node->left != NULL) {
                queuePush(&queue, node->left);
            }
            if (node->right != NULL) {
                queuePush(&queue, node->right);
            }
        }
        printf("]");
    }
    printf("]\n");
    return 0;
```

## 题解二：深度优先搜索（按层挂载）

换个视角：DFS 也能分层。递归时携带**深度参数**，每个节点的值按自己的深度挂到对应层的列表上（第 0 层 = 根）。深层的列表不存在时先创建。

- 时间复杂度：`O(n)`
- 空间复杂度：`O(h)`，递归栈；输出本身 O(n)
- 对比题解一：DFS 顺序是"纵深"的，靠深度参数**事后归位**；BFS 顺序天然分层。两者产出相同

### ☕ Java

```java
import java.util.ArrayList;
import java.util.LinkedList;
import java.util.List;
import java.util.Queue;
import java.util.Scanner;

public class Solution {
    public List<List<Integer>> levelOrder(TreeNode root) {
        List<List<Integer>> ans = new ArrayList<>();
        dfs(root, 0, ans);
        return ans;
    }

    // depth：当前节点所在层（根为第 0 层）
    public static void dfs(TreeNode node, int depth, List<List<Integer>> ans) {
        if (node == null) {
            return;
        }
        if (depth == ans.size()) {
            ans.add(new ArrayList<>());     // 第一次踏足这一层，先建列表
        }
        ans.get(depth).add(node.val);       // 挂到对应层
        dfs(node.left, depth + 1, ans);
        dfs(node.right, depth + 1, ans);
    }

    public static void main(String[] args) {
        // ACM 输入：一行层序序列，null 表示空节点
        Scanner sc = new Scanner(System.in);
        TreeNode root = buildTree(sc);
        System.out.println(new Solution().levelOrder(root));
        sc.close();
    }

    // 按层序遍历读入建树，null 表示空节点
    public static TreeNode buildTree(Scanner sc) {
        if (!sc.hasNext()) {
            return null;
        }
        String token = sc.next();
        if ("null".equals(token)) {
            return null;
        }
        TreeNode root = new TreeNode(Integer.parseInt(token));
        Queue<TreeNode> queue = new LinkedList<>();
        queue.offer(root);
        while (!queue.isEmpty()) {
            TreeNode cur = queue.poll();
            if (!sc.hasNext()) {
                break;
            }
            String left = sc.next();
            if (!"null".equals(left)) {
                cur.left = new TreeNode(Integer.parseInt(left));
                queue.offer(cur.left);
            }
            if (!sc.hasNext()) {
                break;
            }
            String right = sc.next();
            if (!"null".equals(right)) {
                cur.right = new TreeNode(Integer.parseInt(right));
                queue.offer(cur.right);
            }
        }
        return root;
    }
}
```

### ⚙️ C

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct TreeNode {
    int val;
    struct TreeNode* left;
    struct TreeNode* right;
} TreeNode;

#define MAXDEPTH 2000

// 按层收集：全局二维结果
static int* levelVals[MAXDEPTH];
static int levelCnt[MAXDEPTH];

void dfs(struct TreeNode* node, int depth) {
    if (node == NULL) {
        return;
    }
    if (levelVals[depth] == NULL) {
        levelVals[depth] = (int*)malloc(2000 * sizeof(int));
        levelCnt[depth] = 0;
    }
    levelVals[depth][levelCnt[depth]++] = node->val;   // 挂到对应层
    dfs(node->left, depth + 1);
    dfs(node->right, depth + 1);
}

// 按层序遍历读入建树，"null" 表示空节点
TreeNode* buildTree(char tokens[][16], int n) {
    if (n == 0 || strcmp(tokens[0], "null") == 0) {
        return NULL;
    }
    TreeNode** queue = (TreeNode**)malloc(n * sizeof(TreeNode*));
    int head = 0, tail = 0, i = 1;
    TreeNode* root = (TreeNode*)malloc(sizeof(TreeNode));
    root->val = atoi(tokens[0]);
    root->left = root->right = NULL;
    queue[tail++] = root;
    while (head < tail && i < n) {
        TreeNode* cur = queue[head++];
        if (strcmp(tokens[i], "null") != 0) {
            cur->left = (TreeNode*)malloc(sizeof(TreeNode));
            cur->left->val = atoi(tokens[i]);
            cur->left->left = cur->left->right = NULL;
            queue[tail++] = cur->left;
        }
        i++;
        if (i >= n) {
            break;
        }
        if (strcmp(tokens[i], "null") != 0) {
            cur->right = (TreeNode*)malloc(sizeof(TreeNode));
            cur->right->val = atoi(tokens[i]);
            cur->right->left = cur->right->right = NULL;
            queue[tail++] = cur->right;
        }
        i++;
    }
    free(queue);
    return root;
}

int main(void) {
    // ACM 输入：一行层序序列，null 表示空节点
    char tokens[4000][16];
    int n = 0;
    while (n < 4000 && scanf("%15s", tokens[n]) == 1) {
        n++;
    }
    if (n == 0 || strcmp(tokens[0], "null") == 0) {
        printf("[]\n");
        return 0;
    }
    TreeNode* root = buildTree(tokens, n);
    dfs(root, 0);
    // 找出实际层数并输出
    printf("[");
    int printed = 0;
    for (int d = 0; d < MAXDEPTH; d++) {
        if (levelVals[d] == NULL) {
            break;
        }
        printf(printed ? ", [" : "[");
        for (int j = 0; j < levelCnt[d]; j++) {
            printf("%d%s", levelVals[d][j], j + 1 < levelCnt[d] ? ", " : "");
        }
        printf("]");
        printed = 1;
    }
    printf("]\n");
    return 0;
}
```
