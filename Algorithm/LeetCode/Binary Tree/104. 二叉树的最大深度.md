# [104. 二叉树的最大深度](https://leetcode.cn/problems/maximum-depth-of-binary-tree/)

- 🟢 **难度**：简单
- 🌳 **分类**：二叉树
- 🏷️ **标签**：`树`、`深度优先搜索`、`广度优先搜索`、`二叉树`

## 题目描述

给定一个二叉树 `root` ，返回其最大深度。

二叉树的 **最大深度** 是指从根节点到最远叶子节点的最长路径上的节点数。

**示例 1：**

![](https://assets.leetcode.com/uploads/2020/11/26/tmp-tree.jpg)

```
输入：root = [3,9,20,null,null,15,7]
输出：3
```

**示例 2：**

```
输入：root = [1,null,2]
输出：2
```

**提示：**
- 树中节点的数量在 `[0, 10^4]` 区间内。
- `-100 <= Node.val <= 100`

## 题解一：深度优先搜索（递归）

树的深度有天然的递归定义：空树深度为 `0`，非空树的深度为 `1 + max(左子树深度, 右子树深度)`。整个问题被规约为两个结构相同、规模更小的子问题。

- 时间复杂度：`O(n)`，每个节点恰好访问一次
- 空间复杂度：`O(h)`，递归栈深度为树高，最坏（链状树）为 `O(n)`

### ☕ Java

```java
import java.util.LinkedList;
import java.util.Queue;
import java.util.Scanner;

public class Solution {
    public int maxDepth(TreeNode root) {
        if (root == null) {
            return 0;
        }
        return 1 + Math.max(maxDepth(root.left), maxDepth(root.right));
    }

    public static void main(String[] args) {
        // ACM 输入：一行层序序列，null 表示空节点
        Scanner sc = new Scanner(System.in);
        TreeNode root = buildTree(sc);
        System.out.println(new Solution().maxDepth(root));
        sc.close();
    }

    // 按层序遍历读入建树，null 表示空节点
    public static TreeNode buildTree(Scanner sc) {
        if (!sc.hasNext()) {
            return null;
        }
        String token = sc.next();
        if ("null".equals(token)) {
            return null;
        }
        TreeNode root = new TreeNode(Integer.parseInt(token));
        Queue<TreeNode> queue = new LinkedList<>();
        queue.offer(root);
        while (!queue.isEmpty()) {
            TreeNode cur = queue.poll();
            if (!sc.hasNext()) {
                break;
            }
            String left = sc.next();
            if (!"null".equals(left)) {
                cur.left = new TreeNode(Integer.parseInt(left));
                queue.offer(cur.left);
            }
            if (!sc.hasNext()) {
                break;
            }
            String right = sc.next();
            if (!"null".equals(right)) {
                cur.right = new TreeNode(Integer.parseInt(right));
                queue.offer(cur.right);
            }
        }
        return root;
    }
}

class TreeNode {
    int val;
    TreeNode left;
    TreeNode right;

    TreeNode() {}

    TreeNode(int val) { this.val = val; }

    TreeNode(int val, TreeNode left, TreeNode right) {
        this.val = val;
        this.left = left;
        this.right = right;
    }
}
```

### 🐍 Python

```python
import sys
from typing import Optional


class TreeNode:
    def __init__(self, val=0, left=None, right=None):
        self.val = val
        self.left = left
        self.right = right


class Solution:
    def maxDepth(self, root: Optional[TreeNode]) -> int:
        if root is None:
            return 0
        return 1 + max(self.maxDepth(root.left), self.maxDepth(root.right))


def build_tree(tokens, pos):
    """按层序遍历建树，'null' 表示空节点"""
    if pos[0] >= len(tokens) or tokens[pos[0]] == "null":
        pos[0] += 1
        return None
    root = TreeNode(int(tokens[pos[0]]))
    pos[0] += 1
    queue = [root]
    while queue:
        cur = queue.pop(0)
        for _ in range(2):
            if pos[0] >= len(tokens):
                break
            child = None
            if tokens[pos[0]] != "null":
                child = TreeNode(int(tokens[pos[0]]))
                queue.append(child)
            if _ == 0:
                cur.left = child
            else:
                cur.right = child
            pos[0] += 1
    return root


if __name__ == "__main__":
    # ACM 输入：一行层序序列，null 表示空节点
    tokens = sys.stdin.read().split()
    root = build_tree(tokens, [0])
    print(Solution().maxDepth(root))
```

### ⚙️ C

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct TreeNode {
    int val;
    struct TreeNode* left;
    struct TreeNode* right;
} TreeNode;

int maxDepth(struct TreeNode* root) {
    if (root == NULL) {
        return 0;
    }
    int left = maxDepth(root->left);
    int right = maxDepth(root->right);
    return 1 + (left > right ? left : right);
}

// 按层序遍历读入建树，"null" 表示空节点
TreeNode* buildTree(char tokens[][16], int n) {
    if (n == 0 || strcmp(tokens[0], "null") == 0) {
        return NULL;
    }
    TreeNode** queue = (TreeNode**)malloc(n * sizeof(TreeNode*));
    int head = 0, tail = 0, i = 1;
    TreeNode* root = (TreeNode*)malloc(sizeof(TreeNode));
    root->val = atoi(tokens[0]);
    root->left = root->right = NULL;
    queue[tail++] = root;
    while (head < tail && i < n) {
        TreeNode* cur = queue[head++];
        if (strcmp(tokens[i], "null") != 0) {
            cur->left = (TreeNode*)malloc(sizeof(TreeNode));
            cur->left->val = atoi(tokens[i]);
            cur->left->left = cur->left->right = NULL;
            queue[tail++] = cur->left;
        }
        i++;
        if (i >= n) {
            break;
        }
        if (strcmp(tokens[i], "null") != 0) {
            cur->right = (TreeNode*)malloc(sizeof(TreeNode));
            cur->right->val = atoi(tokens[i]);
            cur->right->left = cur->right->right = NULL;
            queue[tail++] = cur->right;
        }
        i++;
    }
    free(queue);
    return root;
}

int main(void) {
    // ACM 输入：一行层序序列，null 表示空节点
    char tokens[4096][16];
    int n = 0;
    while (n < 4096 && scanf("%15s", tokens[n]) == 1) {
        n++;
    }
    TreeNode* root = buildTree(tokens, n);
    printf("%d\n", maxDepth(root));
    return 0;
}
```

## 题解二：广度优先搜索（层序遍历）

深度也可以横着数：逐层遍历，每处理完一层深度加一，队列清空时的层数即为最大深度。

- 时间复杂度：`O(n)`，每个节点入队、出队各一次
- 空间复杂度：`O(w)`，`w` 为树的最大宽度，最坏（满二叉树底层）约为 `O(n/2)`

### ☕ Java

```java
import java.util.LinkedList;
import java.util.Queue;
import java.util.Scanner;

public class Solution {
    public int maxDepth(TreeNode root) {
        if (root == null) {
            return 0;
        }
        Queue<TreeNode> queue = new LinkedList<>();
        queue.offer(root);
        int depth = 0;
        while (!queue.isEmpty()) {
            int size = queue.size();
            // 一次处理完当前层的所有节点
            for (int i = 0; i < size; i++) {
                TreeNode cur = queue.poll();
                if (cur.left != null) {
                    queue.offer(cur.left);
                }
                if (cur.right != null) {
                    queue.offer(cur.right);
                }
            }
            depth++;
        }
        return depth;
    }

    public static void main(String[] args) {
        // ACM 输入：一行层序序列，null 表示空节点
        Scanner sc = new Scanner(System.in);
        TreeNode root = buildTree(sc);
        System.out.println(new Solution().maxDepth(root));
        sc.close();
    }

    // 按层序遍历读入建树，null 表示空节点
    public static TreeNode buildTree(Scanner sc) {
        if (!sc.hasNext()) {
            return null;
        }
        String token = sc.next();

            return null;
        }

        while (!queue.isEmpty()) {

                break;
            }

            if (!"null".equals(left)) {
                cur.left = new TreeNode(Integer.parseInt(left));

            }
            if (!sc.hasNext()) {
                break;
            }
            String right = sc.next();
            if (!"null".equals(right)) {
                cur.right = new TreeNode(Integer.parseInt(right));
                queue.offer(cur.right);
            }
        }
        return root;
    }
}
```

```python
        self.right = right

            return 0

                if cur.right:
```

```c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef struct TreeNode {
    int val;
    struct TreeNode* left;
    struct TreeNode* right;
} TreeNode;

// 手写队列：存节点指针
typedef struct {
    TreeNode** data;
    int head;
    int tail;
} Queue;

void queueInit(Queue* q, int cap) {
    q->data = (TreeNode**)malloc(cap * sizeof(TreeNode*));
    q->head = q->tail = 0;
}
void queuePush(Queue* q, TreeNode* node) { q->data[q->tail++] = node; }
TreeNode* queuePop(Queue* q) { return q->data[q->head++]; }
int queueEmpty(Queue* q) { return q->head == q->tail; }

int maxDepth(struct TreeNode* root) {
    if (root == NULL) {
        return 0;
    }
    Queue queue;
    queueInit(&queue, 10000);
    queuePush(&queue, root);
    int depth = 0;
    while (!queueEmpty(&queue)) {
        int size = queue.tail - queue.head;
        // 一次处理完当前层的所有节点
        for (int i = 0; i < size; i++) {
            TreeNode* cur = queuePop(&queue);
            if (cur->left != NULL) {
                queuePush(&queue, cur->left);
            }
            if (cur->right != NULL) {
                queuePush(&queue, cur->right);
            }
        }
        depth++;
    }
    free(queue.data);
    return depth;
}

// 按层序遍历读入建树，"null" 表示空节点
```
