# [45. 跳跃游戏 II](https://leetcode.cn/problems/jump-game-ii/)

- 🟡 **难度**：中等
- 🤑 **分类**：贪心算法
- 🏷️ **标签**：`贪心`、`数组`、`动态规划`

## 题目描述

给定一个长度为 `n` 的 **0 索引**整数数组 `nums`。初始位置在下标 0。

每个元素 `nums[i]` 表示从索引 `i` 向后跳转的最大长度。换句话说，如果你在索引 `i` 处，你可以跳转到任意 `(i + j)` 处：
- `0 <= j <= nums[i]` 且
- `i + j < n`

返回到达 `n - 1` 的最小跳跃次数。测试用例保证可以到达 `n - 1`。

**示例 1:**

```
输入: nums = [2,3,1,1,4]
输出: 2
解释: 跳到最后一个位置的最小跳跃数是 2。
     从下标为 0 跳到下标为 1 的位置，跳 1 步，然后跳 3 步到达数组的最后一个位置。
```

**示例 2:**

```
输入: nums = [2,3,0,1,4]
输出: 2
```

**提示:**
- `1 <= nums.length <= 10^4`
- `0 <= nums[i] <= 1000`
- 题目保证可以到达 `n - 1`

## 题解一：动态规划（O(n²) / O(n)）

状态定义：`dp[i]` = 从起点 `0` 到达下标 `i` 所需的**最少跳跃次数**。

- 初始：`dp[0] = 0`，其余为无穷大（暂不可达）。
- 转移：枚举所有可能在**一步之内**到达 `i` 的出发点 `j`（`0 <= j < i` 且 `j + nums[j] >= i`），则 `dp[i] = min(dp[i], dp[j] + 1)`。
- 答案：`dp[n - 1]`。

正确性：到达 `i` 的最优路径，其最后一跳一定从某个满足 `j + nums[j] >= i` 的 `j` 出发；而该路径的前缀必然也是到达 `j` 的最优解（否则替换前缀更优），因此由 `dp[j] + 1` 取最小即可。

- 时间复杂度：`O(n^2)`，双重循环
- 空间复杂度：`O(n)`，`dp` 数组

### ☕ Java

```java
import java.util.Scanner;

public class Solution {
    public int jump(int[] nums) {
        int n = nums.length;
        if (n == 1) {
            return 0;                       // 已在终点，无需跳跃
        }
        int[] dp = new int[n];              // dp[i] = 到达下标 i 的最少跳跃次数
        dp[0] = 0;
        for (int i = 1; i < n; i++) {
            dp[i] = Integer.MAX_VALUE;
            for (int j = 0; j < i; j++) {   // 枚举所有能一步到达 i 的出发点 j
                if (j + nums[j] >= i && dp[j] != Integer.MAX_VALUE) {
                    dp[i] = Math.min(dp[i], dp[j] + 1);
                }
            }
        }
        return dp[n - 1];
    }

    public static void main(String[] args) {
        // ACM 输入：第一行 n，第二行 n 个整数
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] nums = new int[n];
        for (int i = 0; i < n; i++) {
            nums[i] = sc.nextInt();
        }
        System.out.println(new Solution().jump(nums));
        sc.close();
    }
}
```

### 🐍 Python

```python
import sys
from typing import List


class Solution:
    def jump(self, nums: List[int]) -> int:
        n = len(nums)
        if n == 1:
            return 0                         # 已在终点，无需跳跃
        INF = float("inf")
        dp = [INF] * n                       # dp[i] = 到达下标 i 的最少跳跃次数
        dp[0] = 0
        for i in range(1, n):
            for j in range(i):               # 枚举所有能一步到达 i 的出发点 j
                if j + nums[j] >= i and dp[j] + 1 < dp[i]:
                    dp[i] = dp[j] + 1
        return dp[n - 1]


if __name__ == "__main__":
    # ACM 输入：第一行 n，第二行 n 个整数
    data = sys.stdin.read().split()
    n = int(data[0])
    nums = [int(x) for x in data[1:1 + n]]
    print(Solution().jump(nums))
```

### ➕ C++

```cpp
#include <climits>
#include <vector>
#include <iostream>
using namespace std;

class Solution {
public:
    int jump(vector<int>& nums) {
        int n = (int)nums.size();
        if (n == 1) {
            return 0;                        // 已在终点，无需跳跃
        }
        vector<int> dp(n, INT_MAX);          // dp[i] = 到达下标 i 的最少跳跃次数
        dp[0] = 0;
        for (int i = 1; i < n; i++) {
            for (int j = 0; j < i; j++) {    // 枚举所有能一步到达 i 的出发点 j
                if (j + nums[j] >= i && dp[j] != INT_MAX) {
                    dp[i] = min(dp[i], dp[j] + 1);
                }
            }
        }
        return dp[n - 1];
    }
};

int main() {
    // ACM 输入：第一行 n，第二行 n 个整数
    int n;
    cin >> n;
    vector<int> nums(n);
    for (int i = 0; i < n; i++) {
        cin >> nums[i];
    }
    cout << Solution().jump(nums) << endl;
    return 0;
}
```

### ⚙️ C

```c
#include <limits.h>
#include <stdio.h>
#include <stdlib.h>

int jump(int* nums, int numsSize) {
    if (numsSize == 1) {
        return 0;                            // 已在终点，无需跳跃
    }
    int* dp = (int*)malloc(numsSize * sizeof(int));   // dp[i] = 到达下标 i 的最少跳跃次数
    dp[0] = 0;
    for (int i = 1; i < numsSize; i++) {
        dp[i] = INT_MAX;
        for (int j = 0; j < i; j++) {        // 枚举所有能一步到达 i 的出发点 j
            if (j + nums[j] >= i && dp[j] != INT_MAX && dp[j] + 1 < dp[i]) {
                dp[i] = dp[j] + 1;
            }
        }
    }
    int ans = dp[numsSize - 1];
    free(dp);
    return ans;
}

int main(void) {
    // ACM 输入：第一行 n，第二行 n 个整数
    int n;
    if (scanf("%d", &n) != 1) {
        return 0;
    }
    int* nums = (int*)malloc(n * sizeof(int));
    for (int i = 0; i < n; i++) {
        scanf("%d", &nums[i]);
    }
    printf("%d\n", jump(nums, n));
    free(nums);
    return 0;
}
```
