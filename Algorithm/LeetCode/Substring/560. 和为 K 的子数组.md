# [560. 和为 K 的子数组](https://leetcode.cn/problems/subarray-sum-equals-k/)

- 🟡 **难度**：中等
- 🔤 **分类**：子串
- 🏷️ **标签**：`数组`、`哈希表`、`前缀和`

## 题目描述

给你一个整数数组 `nums` 和一个整数 `k` ，请你统计并返回 *该数组中和为 `k` 的子数组的个数* 。

子数组是数组中元素的连续非空序列。

**示例 1：**

```
输入：nums = [1,1,1], k = 2
输出：2
```

**示例 2：**

```
输入：nums = [1,2,3], k = 3
输出：2
```

**提示：**
- `1 <= nums.length <= 2 * 10^4`
- `-1000 <= nums[i] <= 1000`
- `-10^7 <= k <= 10^7`

## 题解一：暴力枚举

### ☕ Java

```java
import java.util.Scanner;

class Solution {
    public int subarraySum(int[] nums, int k) {
        int ans = 0;
        for (int i = 0; i < nums.length; i++) {
            if (nums[i] == k) {
                ans++;
            }
            int sum = nums[i];
            for (int j = i + 1; j < nums.length; j++) {
                sum += nums[j];
                if (sum == k) {
                    ans++;
                }
            }
        }
        return ans;
    }

    public static void main(String[] args) {
        // ACM 输入：第一行 n k，第二行 n 个整数
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        int[] nums = new int[n];
        for (int i = 0; i < n; i++) {
            nums[i] = sc.nextInt();
        }
        System.out.println(new Solution().subarraySum(nums, k));
        sc.close();
    }
}
```

## 题解二：前缀和 + 哈希表

### ☕ Java

```java
import java.util.HashMap;
import java.util.Map;
import java.util.Scanner;

class Solution {
    public int subarraySum(int[] nums, int k) {
        int ans = 0;
        // 前缀和 -> 出现次数；空前缀和为 0，记 1 次
        Map<Integer, Integer> count = new HashMap<>();
        count.put(0, 1);

        int pre = 0;
        for (int x : nums) {
            pre += x;
            // 以当前元素结尾、和为 k 的子数组个数 = 前面有多少个前缀和等于 pre - k
            ans += count.getOrDefault(pre - k, 0);
            count.put(pre, count.getOrDefault(pre, 0) + 1);
        }
        return ans;
    }

    public static void main(String[] args) {
        // ACM 输入：第一行 n k，第二行 n 个整数
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int k = sc.nextInt();
        int[] nums = new int[n];
        for (int i = 0; i < n; i++) {
            nums[i] = sc.nextInt();
        }
        System.out.println(new Solution().subarraySum(nums, k));
        sc.close();
    }
}
```

### ➕ C++

```cpp
#include <unordered_map>
#include <vector>
#include <iostream>
using namespace std;

class Solution {
public:
    int subarraySum(vector<int>& nums, int k) {
        int ans = 0;
        // 前缀和 -> 出现次数；空前缀和为 0，记 1 次
        unordered_map<int, int> count;
        count[0] = 1;

        int pre = 0;
        for (int x : nums) {
            pre += x;
            // 以当前元素结尾、和为 k 的子数组个数 = 前面有多少个前缀和等于 pre - k
            int need = pre - k;
            auto it = count.find(need);
            if (it != count.end()) {
                ans += it->second;
            }
            count[pre]++;
        }
        return ans;
    }
};

int main() {
    // ACM 输入：第一行 n k，第二行 n 个整数
    int n, k;
    cin >> n >> k;
    vector<int> nums(n);
    for (int i = 0; i < n; i++) {
        cin >> nums[i];
    }
    cout << Solution().subarraySum(nums, k) << endl;
    return 0;
}
```

### ⚙️ C

```c
#include <stdio.h>
#include <stdlib.h>

static unsigned int hashInt(int v, int cap) {
    return ((unsigned int)v * 2654435761u) & (cap - 1);
}

static void put(int* keys, int* vals, char* used, int cap, int key) {
    unsigned int h = hashInt(key, cap);
    while (used[h]) {
        if (keys[h] == key) {
            vals[h]++;
            return;
        }
        h = (h + 1) & (cap - 1);
    }
    used[h] = 1;
    keys[h] = key;
    vals[h] = 1;
}

static int get(int* keys, int* vals, char* used, int cap, int key) {
    unsigned int h = hashInt(key, cap);
    while (used[h]) {
        if (keys[h] == key) {
            return vals[h];
        }
        h = (h + 1) & (cap - 1);
    }
    return 0;
}

int subarraySum(int* nums, int numsSize, int k) {
    // 开放寻址哈希表：前缀和 -> 出现次数
    int cap = 1;
    while (cap < (numsSize + 1) * 2) {
        cap <<= 1;
    }
    int* keys = (int*)malloc(cap * sizeof(int));
    int* vals = (int*)calloc(cap, sizeof(int));
    char* used = (char*)calloc(cap, sizeof(char));

    int ans = 0;
    put(keys, vals, used, cap, 0);   // 空前缀和为 0，记 1 次

    int pre = 0;
    for (int i = 0; i < numsSize; i++) {
        pre += nums[i];
        // 以当前元素结尾、和为 k 的子数组个数 = 前面有多少个前缀和等于 pre - k
        ans += get(keys, vals, used, cap, pre - k);
        put(keys, vals, used, cap, pre);
    }

    free(keys);
    free(vals);
    free(used);
    return ans;
}

int main(void) {
    // ACM 输入：第一行 n k，第二行 n 个整数
    int n, k;
    if (scanf("%d %d", &n, &k) != 2) {
        return 0;
    }
    int* nums = (int*)malloc(n * sizeof(int));
    for (int i = 0; i < n; i++) {
        scanf("%d", &nums[i]);
    }
    printf("%d\n", subarraySum(nums, n, k));
    free(nums);
    return 0;
}
```
