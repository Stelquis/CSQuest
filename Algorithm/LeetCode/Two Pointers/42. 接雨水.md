# [42. 接雨水](https://leetcode.cn/problems/trapping-rain-water/)

- 🔴 **难度**：困难
- ↔️ **分类**：双指针
- 🏷️ **标签**：`栈`、`数组`、`双指针`、`动态规划`、`单调栈`

## 题目描述

给定 `n` 个非负整数表示每个宽度为 `1` 的柱子的高度图，计算按此排列的柱子，下雨之后能接多少雨水。

**示例 1：**

![](https://assets.leetcode.cn/aliyun-lc-upload/uploads/2018/10/22/rainwatertrap.png)

```
输入：height = [0,1,0,2,1,0,1,3,2,1,2,1]
输出：6
解释：上面是由数组 [0,1,0,2,1,0,1,3,2,1,2,1] 表示的高度图，在这种情况下，可以接 6 个单位的雨水（蓝色部分表示雨水）。 
```

**示例 2：**

```
输入：height = [4,2,0,3,2,5]
输出：9
```

**提示：**
- `n == height.length`
- `1 <= n <= 2 * 10^4`
- `0 <= height[i] <= 10^5`

## 题解：预处理左右最大值

### ☕ Java

```java
import java.util.Scanner;

class Solution {
    public int trap(int[] height) {
        int ans = 0;
        // 左侧最大值
        int[] l = new int[height.length];
        l[0] = height[0];
        for (int i = 1; i < height.length; i++) {
            l[i] = Math.max(l[i - 1], height[i]);
        }
        // 右侧最大值
        int[] r = new int[height.length];
        r[height.length - 1] = height[height.length - 1];
        for (int i = height.length - 2; i >= 0; i--) {
            r[i] = Math.max(r[i + 1], height[i]);
        }
        for (int i = 1; i < height.length - 1; i++) {
            int min = Math.min(l[i], r[i]);
            ans += min > height[i] ? (min - height[i]) : 0;
        }
        return ans;
    }

    public static void main(String[] args) {
        // ACM 输入：第一行 n，第二行 n 个整数
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] height = new int[n];
        for (int i = 0; i < n; i++) {
            height[i] = sc.nextInt();
        }
        System.out.println(new Solution().trap(height));
        sc.close();
    }
}
```

### 🐍 Python

```python
import sys
from typing import List


class Solution:
    def trap(self, height: List[int]) -> int:
        n = len(height)
        ans = 0

        # 左侧最大值
        left = [0] * n
        left[0] = height[0]
        for i in range(1, n):
            left[i] = max(left[i - 1], height[i])

        # 右侧最大值
        right = [0] * n
        right[n - 1] = height[n - 1]
        for i in range(n - 2, -1, -1):
            right[i] = max(right[i + 1], height[i])

        # 每根柱子能接的水 = min(左最大, 右最大) - 自身高度
        for i in range(1, n - 1):
            m = min(left[i], right[i])
            ans += (m - height[i]) if m > height[i] else 0

        return ans


if __name__ == "__main__":
    # ACM 输入：第一行 n，第二行 n 个整数
    data = sys.stdin.read().split()
    n = int(data[0])
    height = [int(x) for x in data[1:1 + n]]
    print(Solution().trap(height))
```

```cpp
#include <vector>
#include <algorithm>
#include <iostream>
using namespace std;

class Solution {
public:
    int trap(vector<int>& height) {
        int n = height.size();
        int ans = 0;

        // 左侧最大值
        vector<int> lv(n);
        lv[0] = height[0];
        for (int i = 1; i < n; i++) {
            lv[i] = max(lv[i - 1], height[i]);
        }

        // 右侧最大值
        vector<int> rv(n);
        rv[n - 1] = height[n - 1];
        for (int i = n - 2; i >= 0; i--) {
            rv[i] = max(rv[i + 1], height[i]);
        }

        // 每根柱子能接的水 = min(左最大, 右最大) - 自身高度
        for (int i = 1; i < n - 1; i++) {
            int m = min(lv[i], rv[i]);
            if (m > height[i]) {
                ans += m - height[i];
            }
        }
        return ans;
    }
};

int main() {
    // ACM 输入：第一行 n，第二行 n 个整数
    int n;
    cin >> n;
    vector<int> height(n);
    for (int i = 0; i < n; i++) {
        cin >> height[i];
    }
    cout << Solution().trap(height) << endl;
    return 0;
}
```

### ⚙️ C

```c
#include <stdio.h>
#include <stdlib.h>

int trap(int* height, int heightSize) {
    if (heightSize <= 2) {
        return 0;
    }
    int ans = 0;

    // 左侧最大值
    int* lv = (int*)malloc(heightSize * sizeof(int));
    lv[0] = height[0];
    for (int i = 1; i < heightSize; i++) {
        lv[i] = lv[i - 1] > height[i] ? lv[i - 1] : height[i];
    }

    // 右侧最大值
    int* rv = (int*)malloc(heightSize * sizeof(int));
    rv[heightSize - 1] = height[heightSize - 1];
    for (int i = heightSize - 2; i >= 0; i--) {
        rv[i] = rv[i + 1] > height[i] ? rv[i + 1] : height[i];
    }

    // 每根柱子能接的水 = min(左最大, 右最大) - 自身高度
    for (int i = 1; i < heightSize - 1; i++) {
        int m = lv[i] < rv[i] ? lv[i] : rv[i];
        if (m > height[i]) {
            ans += m - height[i];
        }
    }

    free(lv);
    free(rv);
    return ans;
}

int main(void) {
    // ACM 输入：第一行 n，第二行 n 个整数
    int n;
    if (scanf("%d", &n) != 1) {
        return 0;
    }
    int* height = (int*)malloc(n * sizeof(int));
    for (int i = 0; i < n; i++) {
        scanf("%d", &height[i]);
    }
    printf("%d\n", trap(height, n));
    free(height);
    return 0;
}
```
