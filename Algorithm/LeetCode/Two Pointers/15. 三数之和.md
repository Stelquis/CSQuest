# [15. 三数之和](https://leetcode.cn/problems/3sum/)

- 🟡 **难度**：中等
- ↔️ **分类**：双指针
- 🏷️ **标签**：`数组`、`双指针`、`排序`

## 题目描述

给你一个整数数组 `nums` ，判断是否存在三元组 `[nums[i], nums[j], nums[k]]` 满足 `i != j`、`i != k` 且 `j != k` ，同时还满足 `nums[i] + nums[j] + nums[k] == 0` 。请你返回所有和为 `0` 且不重复的三元组。

**注意：** 答案中不可以包含重复的三元组。

**示例 1：**

```
输入：nums = [-1,0,1,2,-1,-4]
输出：[[-1,-1,2],[-1,0,1]]
解释：
nums[0] + nums[1] + nums[2] = (-1) + 0 + 1 = 0 。
nums[1] + nums[2] + nums[4] = 0 + 1 + (-1) = 0 。
nums[0] + nums[3] + nums[4] = (-1) + 2 + (-1) = 0 。
不同的三元组是 [-1,0,1] 和 [-1,-1,2] 。
注意，输出的顺序和三元组的顺序并不重要。
```

**示例 2：**

```
输入：nums = [0,1,1]
输出：[]
解释：唯一可能的三元组和不为 0 。
```

**示例 3：**

```
输入：nums = [0,0,0]
输出：[[0,0,0]]
解释：唯一可能的三元组和为 0 。
```

**提示：**
- `3 <= nums.length <= 3000`
- `-10^5 <= nums[i] <= 10^5`

## 题解：排序 + 双指针

### ☕ Java

```java
import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;
import java.util.Scanner;

class Solution {
    public List<List<Integer>> threeSum(int[] nums) {
        List<List<Integer>> ans = new ArrayList<>();
        Arrays.sort(nums);
        for (int i = 0; i < nums.length - 2; i++) {
            if (i > 0 && nums[i] == nums[i - 1]) {
                continue;
            }
            int l = i + 1, r = nums.length - 1;
            while (l < r) {
                int sum = nums[i] + nums[r] + nums[l];
                if (sum == 0) {
                    ans.add(Arrays.asList(nums[i], nums[l], nums[r]));
                    while (l < r && nums[l] == nums[l + 1]) l++;
                    while (l < r && nums[r] == nums[r - 1]) r--;
                    l++;
                    r--;
                } else if (sum < 0) {
                    l++;
                } else if (sum > 0) {
                    r--;
                }
            }
        }
        return ans;
    }

    public static void main(String[] args) {
        // ACM 输入：第一行 n，第二行 n 个整数
        Scanner sc = new Scanner(System.in);
        int n = sc.nextInt();
        int[] nums = new int[n];
        for (int i = 0; i < n; i++) {
            nums[i] = sc.nextInt();
        }
        List<List<Integer>> res = new Solution().threeSum(nums);
        for (List<Integer> tri : res) {
            StringBuilder sb = new StringBuilder();
            for (int i = 0; i < tri.size(); i++) {
                if (i > 0) sb.append(' ');
                sb.append(tri.get(i));
            }
            System.out.println(sb);
        }
        sc.close();
    }
}
```

```python
import sys
from typing import List


class Solution:
    def threeSum(self, nums: List[int]) -> List[List[int]]:
        ans = []
        nums.sort()
        for i in range(len(nums) - 2):
            if i > 0 and nums[i] == nums[i - 1]:
                continue
            l, r = i + 1, len(nums) - 1
            while l < r:
                total = nums[i] + nums[l] + nums[r]
                if total == 0:
                    ans.append([nums[i], nums[l], nums[r]])
                    while l < r and nums[l] == nums[l + 1]:
                        l += 1
                    while l < r and nums[r] == nums[r - 1]:
                        r -= 1
                    l += 1
                    r -= 1
                elif total < 0:
                    l += 1
                else:
                    r -= 1
        return ans


if __name__ == "__main__":
    # ACM 输入：第一行 n，第二行 n 个整数
    data = sys.stdin.read().split()
    n = int(data[0])
    nums = [int(x) for x in data[1:1 + n]]
    res = Solution().threeSum(nums)
    for tri in res:
        print(" ".join(map(str, tri)))
```

```cpp
#include <algorithm>

class Solution {

    vector<vector<int>> threeSum(vector<int>& nums) {
        vector<vector<int>> ans;
        sort(nums.begin(), nums.end());
        for (int i = 0; i < (int)nums.size() - 2; i++) {
            if (i > 0 && nums[i] == nums[i - 1]) {
                continue;
            }
            int l = i + 1, r = (int)nums.size() - 1;
            while (l < r) {
                int sum = nums[i] + nums[l] + nums[r];

                    ans.push_back({nums[i], nums[l], nums[r]});
                    while (l < r && nums[l] == nums[l + 1]) l++;
                    while (l < r && nums[r] == nums[r - 1]) r--;
                    l++;
                    r--;
                } else if (sum < 0) {
                    l++;
                } else {
                    r--;
                }
            }
        }
        return ans;
    }
};

int main() {
    // ACM 输入：第一行 n，第二行 n 个整数
    int n;
    cin >> n;
    vector<int> nums(n);
    for (int i = 0; i < n; i++) {
```

### ⚙️ C

```c
#include <stdio.h>
#include <stdlib.h>

static int cmpInt(const void* a, const void* b) {
    return (*(const int*)a) - (*(const int*)b);
}

/**
 * Return an array of arrays of size *returnSize.
 * The sizes of the arrays are returned as *returnColumnSizes array.
 * Note: Both returned array and *columnSizes array must be malloced, assume caller calls free().
 */
int** threeSum(int* nums, int numsSize, int* returnSize, int** returnColumnSizes) {
    qsort(nums, numsSize, sizeof(int), cmpInt);

    int cap = 16;
    int count = 0;
    int** res = (int**)malloc(cap * sizeof(int*));
    int* colSizes = (int*)malloc(cap * sizeof(int));

    for (int i = 0; i < numsSize - 2; i++) {
        if (i > 0 && nums[i] == nums[i - 1]) {
            continue;
        }
        int l = i + 1, r = numsSize - 1;
        while (l < r) {
            int sum = nums[i] + nums[l] + nums[r];
            if (sum == 0) {
                if (count == cap) {
                    cap <<= 1;
                    res = (int**)realloc(res, cap * sizeof(int*));
                    colSizes = (int*)realloc(colSizes, cap * sizeof(int));
                }
                int* tri = (int*)malloc(3 * sizeof(int));
                tri[0] = nums[i];
                tri[1] = nums[l];
                tri[2] = nums[r];
                res[count] = tri;
                colSizes[count] = 3;
                count++;

                while (l < r && nums[l] == nums[l + 1]) l++;
                while (l < r && nums[r] == nums[r - 1]) r--;
                l++;
                r--;
            } else if (sum < 0) {
                l++;
            } else {
                r--;
            }
        }
    }

    *returnSize = count;
    *returnColumnSizes = colSizes;
    return res;
}

int main(void) {
    // ACM 输入：第一行 n，第二行 n 个整数
    int n;
    if (scanf("%d", &n) != 1) {
        return 0;
    }
    int* nums = (int*)malloc(n * sizeof(int));
    for (int i = 0; i < n; i++) {
        scanf("%d", &nums[i]);
    }

    int returnSize = 0;
    int* returnColumnSizes = NULL;
    int** res = threeSum(nums, n, &returnSize, &returnColumnSizes);

    // 输出：每组一行，组内空格分隔
    for (int i = 0; i < returnSize; i++) {
        for (int j = 0; j < returnColumnSizes[i]; j++) {
            if (j) printf(" ");
            printf("%d", res[i][j]);
        }
        printf("\n");
    }

    for (int i = 0; i < returnSize; i++) {
        free(res[i]);
    }
    free(res);
    free(returnColumnSizes);
    free(nums);
    return 0;
}
```
